   ───────────────────────────────────────────────────────────────
   🎯 Rôle : porter l'état défaut d'un composant, applicable dans TOUS les cas

   ======================================================================= *)

(* =======================================================================
   🧩 FB_FaultCore — Socle de remplissage de ST_Fault (vue live + vue latchée)
   ───────────────────────────────────────────────────────────────

   🔒 Sécurité :
      - interlock TOUJOURS sur `Causes[i].Active` (cause brute), jamais sur le latch.
      - Reset TOUJOURS effectif, jamais conditionné (std §9) — agit même Enable=FALSE.
      - Latching=FALSE → cause visible en LIVE seulement (fail-safe : reste visible).
      - un défaut laté non acquitté reste publié même FB désactivé (pas d'effacement
        silencieux sur bascule Enable OFF→ON sans Reset).

   🧬 Un seul socle, réutilisé par tout FB `standard` qui expose un ST_Fault.
      Pas de State, pas de Warning, pas de texte — hors périmètre de ce socle.
   📄 Doc : DOC/STDS/CODE_QUALITY_STANDARDS.md §2quinquies / §9

    Causes  : ARRAY[0..15] OF ST_FaultCause; // --> [CMD] Liste des causes en clair (Active / Latching / Texte)
END_VAR
VAR_OUTPUT

    i         : INT;                         // . [LOC] Index de parcours de la liste
END_VAR

// Front Reset : efface tous les latches. Jamais conditionné, agit même Enable=FALSE (§9).
Ready := Enable;
ResetEdge(CLK := Reset);
IF ResetEdge.Q THEN
    FOR i := 0 TO 15 DO Latch[i] := FALSE; END_FOR;
END_IF;
{endregion}

{region "§2 Vue LIVE — Error / ErrorId"}

// Non évaluée si Enable=FALSE (pas de lecture de cause hors autorisation).

    FOR i := 0 TO 15 DO
        IF Causes[i].Active THEN
            Fault.ErrorId := Fault.ErrorId OR SHL(WORD#1, INT_TO_DWORD(i));
        END_IF;
    END_FOR;
END_IF;

{endregion}

// même si la cause disparaît. Réapparition => ré-armement (ré-alarme).

    FOR i := 0 TO 15 DO
        IF Causes[i].Active AND Causes[i].Latching THEN
            Latch[i] := TRUE;
        END_IF;
    END_FOR;
END_IF;
{endregion}

END_FOR;

{endregion}

(* =======================================================================

   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul du dt réel entre deux exécutions successives
   ⚙️ Calcul : Delta horloge système TIME() converti en secondes
   🛡️ Robustesse : dt borné (0 < dt <= CST_MaxCycleDeltaMs) sinon valeur de secours —
      neutralise les artefacts de rebouclage TIME() (49,7 j) et de reprise (online-change, debug)
   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md
   ======================================================================= *)
FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

    Hyst3                           : HYSTERESIS;                     // * [INST] Transition palier 3 vers 4
    Hyst4                           : HYSTERESIS;                     // * [INST] Transition palier 4 vers 5

    SpeedTgtInt                     : INT;                            // . [LOC] Consigne convertie en INT pour Hysteresis

// === IMPLEMENTATION (à coller dans le corps ST du POU FB_SpeedStep) ===

{region "§0 Validation configuration"}
// ─────────────────────────────────────────────────────────────────────────
// 🛡️ 0. VALIDATION CONFIGURATION (P0.2) — exécutée CHAQUE cycle pour détecter
//    toute corruption IHM/PERSISTENT, même siEnable=FALSE (sorties sûres forcée)
// ─────────────────────────────────────────────────────────────────────────

// 🔧 Bornage systématique MaxStepNumber ∈ [1..5]
IF MaxStepNumber < 1 THEN
    _MaxStepClamped := 1;
ELSIF MaxStepNumber > 5 THEN
    _MaxStepClamped := 5;
ELSE
    _MaxStepClamped := MaxStepNumber;
END_IF;

// 🔍 Validation table ST_SpeedStepTable
// Critères (Partie 9 §2) :
// 1. Seuils croissants stricts : StepThreshold_Pct[1] < [2] < [3] < [4] < [5]
// 2. Seuils dans [0..100] %
// 3. HystMargin > 0
// 4. Pour chaque palier 2.._MaxStepClamped : au moins 1 contacteur TRUE (cohérence câblage)
//    Le palier 1 peut légitimement avoir les 4 contacteurs FALSE : résistances rotoriques
//    toutes insérées = vitesse minimale. On ne valide pas les paliers jamais atteignables.
_ConfigValid := TRUE;
_ConfigErrorIdInt := 16#0000;

// Critère 1 : seuils croissants
IF NOT (Table.StepThreshold_Pct[1] < Table.StepThreshold_Pct[2] AND
        Table.StepThreshold_Pct[2] < Table.StepThreshold_Pct[3] AND
        Table.StepThreshold_Pct[3] < Table.StepThreshold_Pct[4] AND
        Table.StepThreshold_Pct[4] < Table.StepThreshold_Pct[5]) THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;  // bit0
END_IF;

// Critère 2 : seuils dans [0..100]
IF (Table.StepThreshold_Pct[1] < 0.0) OR (Table.StepThreshold_Pct[5] > 100.0) THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
END_IF;

// Critère 3 : hystérésis > 0
IF HystMargin <= 0.0 THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
END_IF;

// Critère 4 : cohérence contacteurs par palier (seulement jusqu'au plafond actif)
IF _ConfigValid THEN
    // Palier 1 : les 4 sorties FALSE sont une configuration valide de vitesse minimale.
    // Palier 2
    IF _MaxStepClamped >= 2 THEN
        IF NOT (Table.P2R1 OR Table.P2R2 OR Table.P2R3 OR Table.P2R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 3
    IF _MaxStepClamped >= 3 THEN
        IF NOT (Table.P3R1 OR Table.P3R2 OR Table.P3R3 OR Table.P3R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 4
    IF _MaxStepClamped >= 4 THEN
        IF NOT (Table.P4R1 OR Table.P4R2 OR Table.P4R3 OR Table.P4R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 5
    IF _MaxStepClamped >= 5 THEN
        IF NOT (Table.P5R1 OR Table.P5R2 OR Table.P5R3 OR Table.P5R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
END_IF;

// 🛡️ Gestion erreur config : front montant + reset sur front Reset (cause disparue)
// Note : pas d'entrée Reset dédiée (brique réduite) → on réarme quand config redevient valide
//        (front descendant de _ConfigValid). Conforme Partie 3 §5 : cause disparue + front.
_ConfigErrorTrig(CLK := NOT _ConfigValid);  // Front montant = apparition erreur

// Mémoire erreur config : reste TRUE tant que config invalide OU pas de réarmement
// Réarmement = front montant de _ConfigValid (cause disparue) — détecté via _ConfigValidPrev
IF _ConfigErrorTrig.Q THEN
    // Nouvelle erreur : on la mémorise
    _ConfigValidPrev := FALSE;  // Force détection front montant au cycle suivant si redevient valide
ELSIF _ConfigValid AND NOT _ConfigValidPrev THEN
    // Config redevenue valide : front montant → on efface l'erreur (reset sur front)
    _ConfigValidPrev := TRUE;
ELSIF NOT _ConfigValid THEN
    _ConfigValidPrev := FALSE;
END_IF;

// Sorties erreur config (miroir standard Partie 3)
ConfigErrorId := _ConfigErrorIdInt;
ConfigError   := (ConfigErrorId <> 16#0000);
{endregion}

{region "§1 Gate"}
// ─────────────────────────────────────────────────────────────────────────
// 🛑 1. GATE — Enable FALSE OU config invalide = palier 0, sorties sûres
// ─────────────────────────────────────────────────────────────────────────
IF (NOT Enable) OR (NOT _ConfigValid) THEN
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    StepNumber := 0;
    SpeedGuardLimited := FALSE;
    RETURN;
END_IF;
{endregion}

{region "§2 Sélection palier"}
// ─────────────────────────────────────────────────────────────────────────
// 🪜 2. SÉLECTION DU PALIER (HYSTERESIS lib Util : IN/HIGH/LOW → OUT)
//    OUT = TRUE tant que IN < LOW (régime "bas") ; OUT bascule à FALSE une fois
//    IN > HIGH (régime "haut" confirmé) ; maintien entre LOW et HIGH (anti-battement).
//    => NOT Hyst_n.OUT = "on a franchi le seuil n vers le haut" (palier n+1 atteint).
// ─────────────────────────────────────────────────────────────────────────
SpeedTgtInt := REAL_TO_INT(SpeedTgt_Pct);

Hyst1(IN := SpeedTgtInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[1] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[1] - HystMargin));
Hyst2(IN := SpeedTgtInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[2] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[2] - HystMargin));
Hyst3(IN := SpeedTgtInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[3] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[3] - HystMargin));
Hyst4(IN := SpeedTgtInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[4] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[4] - HystMargin));

StepNumber := 1;                          // Palier mini dès qu'Enable est actif
IF NOT Hyst1.OUT THEN StepNumber := 2; END_IF;
IF NOT Hyst2.OUT THEN StepNumber := 3; END_IF;
IF NOT Hyst3.OUT THEN StepNumber := 4; END_IF;
IF NOT Hyst4.OUT THEN StepNumber := 5; END_IF;

// ─────────────────────────────────────────────────────────────────────────
// 2bis. PLAFOND — appliqué APRÈS la sélection hystérésis,
// AVANT la lecture table : le 0..100% de la consigne peut réclamer le palier 5,
// mais ne dépassera jamais _MaxStepClamped (ex. limitation couple en descente)
// ─────────────────────────────────────────────────────────────────────────
IF StepNumber > _MaxStepClamped THEN
    StepNumber := _MaxStepClamped;
END_IF;
{endregion}

{region "§2ter Garde-fou vitesse"}
// ─────────────────────────────────────────────────────────────────────────
// 🛡️ 2ter. GARDE-FOU VITESSE RÉELLE — limite uniquement la consigne de palier.
// Si la vitesse n'est pas validée stable, seul le palier 1 reste autorisé.
// ─────────────────────────────────────────────────────────────────────────
SpeedGuardLimited := FALSE;
IF SpeedGuardEnable THEN
    IF NOT SpeedGuardReady THEN
        IF StepNumber > 1 THEN
            StepNumber := 1;
            SpeedGuardLimited := TRUE;
        END_IF;
    ELSIF (MeasuredSpeedBand >= 1) AND (MeasuredSpeedBand <= 5) AND (StepNumber > MeasuredSpeedBand) THEN
        StepNumber := MeasuredSpeedBand;
        SpeedGuardLimited := TRUE;
    END_IF;
END_IF;
{endregion}

{region "§3 Contacteurs"}
// ─────────────────────────────────────────────────────────────────────────
// 📤 3. CONTACTEURS — lus dans la table (donnée, jamais codé en dur)
//    CASE couvre explicitement 0..5 : StepNumber=0 → tous FALSE (sécurité)
// ─────────────────────────────────────────────────────────────────────────
CASE StepNumber OF
    0:  // Arrêt / config invalide / Enable=FALSE — sécurité positive
        Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    1:  Contactor1 := Table.P1R1; Contactor2 := Table.P1R2; Contactor3 := Table.P1R3; Contactor4 := Table.P1R4;
    2:  Contactor1 := Table.P2R1; Contactor2 := Table.P2R2; Contactor3 := Table.P2R3; Contactor4 := Table.P2R4;
    3:  Contactor1 := Table.P3R1; Contactor2 := Table.P3R2; Contactor3 := Table.P3R3; Contactor4 := Table.P3R4;
    4:  Contactor1 := Table.P4R1; Contactor2 := Table.P4R2; Contactor3 := Table.P4R3; Contactor4 := Table.P4R4;
    5:  Contactor1 := Table.P5R1; Contactor2 := Table.P5R2; Contactor3 := Table.P5R3; Contactor4 := Table.P5R4;
ELSE
    // Défense en profondeur : valeur StepNumber inattendue → sécurité
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    StepNumber := 0;
END_CASE;
{endregion}

VAR_INPUT

    Reset                           : BOOL;                           // --> [CMD] Acquittement defaut sur front

    PowerContactorEngaged           : BOOL;                           // --> [SAFE] Chaine AU rearmee et puissance OK

    FwdRevSpeedFeedbackOff          : BOOL;                           // --> [HW] Feedback retombe tous contacteurs sens/vitesse

    RequestedContactor3             : BOOL;                           // --> [CMD] Demande contacteur vitesse palier 3
    RequestedContactor4             : BOOL;                           // --> [CMD] Demande contacteur vitesse palier 4
    RequestedStep                   : INT;                            // --> [CMD] Palier de vitesse demande

    // === TEMPS MORT DIRECTIONNEL (TC-P10-021/022) ===
    DeadTimeSameDir                 : TIME := T#1s;                  // --> [CFG] Temps mort redemarrage meme sens apres arret

    Ready                           : BOOL;                           // <-- [STAT] FB pret a fonctionner
    Busy                            : BOOL;                           // <-- [STAT] Interlock actif

    Error                           : BOOL;                           // <-- [STAT] Synthese defaut
    ErrorId                         : WORD;                           // <-- [DIAG] Code erreur (bit0: timeout frein)
    State                           : E_WinchFinalInterlockState;     // <-- [STAT] Etat courant machine d'etat interlock
    StateAtError                    : E_WinchFinalInterlockState;     // <-- [DIAG] Etat memorise lors du dernier defaut

    AuthorizedStep                  : INT;                            // <-- [ACT] Palier de vitesse autorise
    RelayFwd                        : BOOL;                           // <-- [ACT] Ordre contacteur sens montee valide

    Contactor2                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 2 valide

    StepDelayElapsed                : TIME;                           // <-- [DIAG] Ecoule temporisation palier

    BrakeTimeoutCauseEdge           : R_TRIG;                         // . [LOC] Detecteur front cause timeout frein
    BrakeTimeout                    : TON;                            // . [LOC] Temporisation watchdog frein
    StepDelay                       : TON;                            // . [LOC] Temporisation changement de palier
    RestartDelay                    : TON;                            // . [LOC] Temporisation anti-court-cycle redemarrage
    DeadTime                        : TON;                            // . [LOC] Temporisation temps mort directionnel

    // === VARIABLES LOCALES CALCULÉES ===
    RestartRequired                 : BOOL;                           // . [LOC] Exigence de retour au neutre pour redemarrage
    ResetRequired                   : BOOL;                           // . [LOC] Exigence d'acquittement Reset conscient
    NeutralRequestSeen              : BOOL;                           // . [LOC] Observation passage consigne au neutre
    RequestedStepClamped            : INT;                            // . [LOC] Palier demande borne dans [0..5]
    StepDelayTarget                 : INT;                            // . [LOC] Cible palier memorisee
    MotorRequest                    : BOOL;                           // . [LOC] Demande mouvement moteur active
    BrakeTimeoutCause               : BOOL;                           // . [LOC] Cause timeout frein presente
    BrakeTimeoutAck                 : BOOL := TRUE;                  // . [LOC] Acquittement cause timeout frein

    DeadTimePending                 : BOOL;                           // . [LOC] Temps mort directionnel en attente
    NewDirection                    : INT;                            // . [LOC] Sens de la nouvelle demande
    RequiredDeadTime                : TIME;                           // . [LOC] Temps mort requis selon sens de la demande

// ── 1. FRONT RESET AVANT GATE ───────────────────────────────────────────
// Le front est échantillonné même pendant Enable/AU faux : un Reset maintenu
// pendant une gate ne devient jamais un faux acquittement au retour.
ResetEdge(CLK := Reset);

// ── 2. GATE : neutralisation sûre ───────────────────────────────────────
// ErrorId, RestartInhibit et ResetRequired ne sont jamais effacés par une perte
// Enable/AU : celle-ci coupe uniquement les sorties. Le retour exige toujours un
// acquittement conscient puis une nouvelle demande opérateur.
IF NOT Enable OR NOT PowerContactorEngaged THEN
    BrakeTimeout(IN := FALSE, PT := T#500ms);
    StepDelay(IN := FALSE, PT := T#1s250ms);
    RestartDelay(IN := FALSE, PT := T#1000ms);
    DeadTime(IN := FALSE);
    DeadTimePending := FALSE;
    DeadTimeElapsed := T#0ms;
    RelayFwd := FALSE; RelayRev := FALSE;
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    BrakeCmd := FALSE; AuthorizedStep := 0;
    BrakeTimeoutElapsed := T#0ms; RestartDelayElapsed := T#0ms; StepDelayElapsed := T#0ms;
    Ready := FALSE; Busy := FALSE; Done := FALSE;
    Error := (ErrorId <> 16#0000);
    State := E_WinchFinalInterlockState.DISABLED;
    RestartRequired := TRUE;
    // 🔒 Sorties de diagnostic et latches de sûreté VOLONTAIREMENT conservés hors Enable (AC3 / §9) :
    // - ErrorId & RestartInhibit : latches défaut et anti-redémarrage auto (doivent survivre à la perte d'Enable, acquittement conscient requis)
    // - StateAtError & Reason : snapshot d'état et cause de blocage préservés pour diagnostic IHM
    RETURN;
END_IF;

RequestedStepClamped := LIMIT(0, RequestedStep, 5);
// SafeStop reste une rampe rapide du FB_Winch (Enable maintenu, Partie 3 §1bis).
// La demande métier reste donc autorisée ici pendant la décélération ; l'interlock
// ne coupe qu'après sa retombée effective à zéro, sauf gate dur ou défaut final.
MotorRequest := (RequestedStepClamped > 0) AND (RequestedRelayFwd XOR RequestedRelayRev);

// ── 2bis. Arbitrage Error/ErrorId AVANT §3/§5 ────────────────────────────
// ErrorId bit0 (défaut frein) est mis à jour en FIN de scan (§6bis, après BrakeCmd) —
// la valeur lue ici est donc celle de la fin du scan PRÉCÉDENT (retard d'un scan sur une
// nouvelle occurrence, ~10ms, négligeable face au budget 500ms du watchdog). Voir bandeau.
Error := (ErrorId <> 16#0000);
ErrorEdge(CLK := Error);
IF ErrorEdge.Q THEN
    StateAtError := State;
END_IF;

// ── 3. Réarmement anti-redémarrage automatique ──────────────────────────
// Après timeout : cause disparue + front Reset, puis demande moteur à zéro,
// puis nouvelle demande distincte. La retombée + T#1000ms sera ensuite refaite.
IF RestartInhibit AND NOT Error AND NOT ResetRequired AND NOT ResetEdge.Q THEN
    IF NOT MotorRequest THEN
        NeutralRequestSeen := TRUE;
        RestartRequired := TRUE;
    ELSIF NeutralRequestSeen AND MotorRequest THEN
        RestartInhibit := FALSE;
        NeutralRequestSeen := FALSE;
        RestartRequired := TRUE;
        Reason := E_WinchFinalInterlockReason.NONE;
    END_IF;
END_IF;

// ── 4. Mémorisation de l'arrêt : temporisation d'anti-redémarrage chaud ──────────
// La temporisation RestartDelay (1500ms) s'arme à l'arrêt du mouvement (NOT MotorRequest) et
// s'écoule pendant la pause à l'arrêt (au lieu de repartir de zéro à chaque nouvel appui) —
// permet un redémarrage instantané si la pause dépasse 1.5s. Le décompte ne démarre/continue que
// si le moteur n'est plus demandé ET que le frein est physiquement confirmé fermé
// (NOT BrakeFeedback = bobine retombée, mâchoires serrées). Garde le comportement "purge pendant
// la pause" (RestartRequired ne s'arme qu'à NOT MotorRequest, indépendant de la tempo).
IF NOT MotorRequest THEN
    RestartRequired := TRUE;
END_IF;
RestartDelay(IN := RestartRequired AND NOT MotorRequest AND NOT BrakeFeedback AND NOT RestartInhibit, PT := T#1500ms);
RestartDelayElapsed := RestartDelay.ET;
IF RestartDelay.Q THEN
    RestartRequired := FALSE;
END_IF;

// ── 4bis. Temps mort directionnel (TC-P10-021/022) ─────────────────────────────
// Distinct du RestartDelay (anti-court-cycle uniforme) : temps mort de redémarrage
// après arrêt, dépendant du sens de la nouvelle demande. Même sens → DeadTimeSameDir ;
// inversion → DeadTimeOppositeDir. Neutre↔sens = immédiat (pas de temps mort à l'armement).
// Le décompte se fait pendant la pause ET pendant l'attente après nouvelle demande
// (purge pendant la pause, comme RestartDelay). PT = DeadTimeOppositeDir (le plus long) ;
// à la nouvelle demande, on vérifie si le temps mort requis pour le sens réel est déjà
// écoulé (cas même sens, plus court).
IF MotorRequest THEN
    IF RequestedRelayFwd THEN
        LastDirection := 1;
    ELSE
        LastDirection := -1;
    END_IF;
END_IF;

IF NOT MotorRequest THEN
    DeadTimePending := TRUE;
END_IF;

DeadTime(IN := DeadTimePending AND NOT BrakeFeedback AND NOT RestartInhibit, PT := DeadTimeOppositeDir);
DeadTimeElapsed := DeadTime.ET;
IF DeadTime.Q THEN
    DeadTimePending := FALSE;
END_IF;

IF MotorRequest AND DeadTimePending THEN
    IF RequestedRelayFwd THEN
        NewDirection := 1;
    ELSE
        NewDirection := -1;
    END_IF;
    IF NewDirection = LastDirection THEN
        RequiredDeadTime := DeadTimeSameDir;
    ELSE
        RequiredDeadTime := DeadTimeOppositeDir;
    END_IF;
    IF DeadTime.ET >= RequiredDeadTime THEN
        DeadTimePending := FALSE;
    END_IF;
END_IF;

// ── 5. Autorisation moteur ────────────────────────────────────────────────
RelayFwd := FALSE; RelayRev := FALSE;
Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
Busy := FALSE; Done := FALSE; Ready := NOT Error;

FUNCTION_BLOCK PUBLIC FB_Winch
VAR_INPUT
    // === COMMANDES ET CONSIGNES ===
    Enable                          : BOOL;                           // --> [CMD] Autorisation generale FB (TRUE=Autorise)
    Reset                           : BOOL;                           // --> [CMD] Acquittement defaut (front)
    StartStop                       : BOOL;                           // --> [CMD] Ordre de marche/arret mouvement
    Direction                       : INT;                            // --> [CMD] Sens demande (-1: Descente, 0: Stop, 1: Montee)
    SpeedTgt_Pct                    : REAL;                           // --> [CMD] Vitesse cible % (0.0 a 100.0)
    Mode                            : E_Mode;                         // --> [CMD] Mode d'exploitation actif

    // === SECOURS ET PERMIS DE SÉCURITÉ ===
    SafeStop                        : BOOL;                           // --> [SAFE] Arret rapide de securite
    DescendPermit                   : BOOL := TRUE;                  // --> [SAFE] Permis de descente (mou de cable)
    AscentPermit                    : BOOL := TRUE;                  // --> [SAFE] Permis de montee (fin de course haut)
    PowerContactorEngaged           : BOOL;                           // --> [SAFE] Statut contacteur de ligne / puissance

    // === REGLAGES ET CONFIGURATION ===
    SpeedStepTable                  : ST_SpeedStepTable;              // --> [CFG] Table de seuils des paliers de vitesse
    CfgMaxStepDescente              : INT := 3;                       // --> [CFG] Plafond palier en descente (1..5)
    MaxStepAscent                   : INT := 5;                       // --> [CFG] Plafond palier en montee (1..5)
    HomingApproachActive            : BOOL;                           // --> [CFG] Vitesse limitee au palier 1 (approche homing)
    SpeedStepHystMargin             : REAL := 2.0;                    // --> [CFG] Marge d'hysteresis palier (%)
    DirectionInterlockDelayDescent  : TIME := T#400ms;                // --> [CFG] Delai interlock changement de sens descente
    DirectionInterlockDelayAscent   : TIME := T#900ms;                // --> [CFG] Delai interlock changement de sens montee
    ContactorFeedbackTimeout        : TIME := T#500ms;                // --> [CFG] Delai d'attente retombee des contacteurs
    TopLimitM                       : REAL := 8.5;                    // --> [CFG] Position limite haute active (m)
    BottomLimitM                    : REAL := -20.0;                  // --> [CFG] Position limite basse active (m)
    CfgSlowdownDistanceM            : REAL := 1.0;                    // --> [CFG] Distance d'approche avant limite pour ralentir (m)
    CfgSlowdownMaxStep              : INT := 1;                       // --> [CFG] Plafond palier dans zone de ralentissement (1..5)

    // === MESURES ET FEEDBACKS SENSORS ===
    FwdRevSpeedFeedbackOff          : BOOL;                           // --> [HW] Retour unique contacteurs sens+vitesse retombes
    Homed                           : BOOL;                           // --> [HW] Statut treuil reference
    HomingSuspect                   : BOOL;                           // --> [HW] Incertitude / doute sur le referencement
    CablePosM                       : REAL;                           // --> [HW] Position actuelle du cable (m)
    MeasuredSpeedBand               : INT := 0;                       // --> [HW] Bande vitesse reelle validee
    SpeedGuardEnable                : BOOL := FALSE;                  // --> [HW] Activation garde-fou changement de palier
    SpeedGuardReady                 : BOOL := FALSE;                  // --> [HW] Conditions de stabilite validees

    // === BYPASSES ET SIMULATION ===
    BypassContactorCheck            : BOOL;                           // --> [TST] Bypass diagnostic contacteurs (simu)
    BypassGlobal                    : BOOL;                           // --> [TST] Force ErrorId=0, ignore erreurs
END_VAR
VAR_OUTPUT
    // === ETATS IEC ET SYNOPTIQUE ===
    Ready                           : BOOL;                           // <-- [STAT] FB pret a fonctionner
    // 🧩 Statut transverse (forme cible ST_Fault)
    Fault                           : ST_Fault;

    // === ORDRES ACTIONNEURS (CONTACTEURS) ===
    RelayFwd                        : BOOL;                           // <-- [ACT] Ordre contacteur sens montee
    RelayRev                        : BOOL;                           // <-- [ACT] Ordre contacteur sens descente
    Contactor1                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 1
    Contactor2                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 2
    Contactor3                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 3
    Contactor4                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 4

    // === DIAGNOSTICS ET MESURES ===
    RequestedStep                   : INT;                            // <-- [DIAG] Palier brut calcule par SpeedStep
    StepNumber                      : INT;                            // <-- [DIAG] Palier metier temporise actif
    BusinessStepDelayElapsed        : TIME;                           // <-- [DIAG] Ecoule tempo metier palier
    SpeedGuardLimited               : BOOL;                           // <-- [DIAG] Statut limitation par garde-fou
    ContactorsCheck                 : ST_ContactorCheck;              // <-- [DIAG] Diagnostic concorde contacteurs
    InTopSlowdownZone               : BOOL;                           // <-- [DIAG] Zone de ralentissement haut active
    InBottomSlowdownZone            : BOOL;                           // <-- [DIAG] Zone de ralentissement bas active
    CommandedDirection              : INT;                            // <-- [DIAG] Sens interne memorise
    DirectionChangeDelayElapsed     : TIME;                           // <-- [DIAG] Ecoule delai changement de sens
    DirectionChangePending          : BOOL;                           // <-- [DIAG] Changement de sens en attente
END_VAR
VAR
    // === SOUS-INSTANCES FB ===
    SpeedStep                       : FB_SpeedStep;                   // * [INST] Decodage paliers vitesse
    CycleTimeCalc                   : FB_CycleTime;                   // * [INST] Calcul temps de cycle

    // === DETECTEURS ET TIMERS INTERNES ===
    ResetEdge                       : R_TRIG;                         // . [LOC] Detecteur front montant Reset
    DirectionChangeDelay            : TON;                            // . [LOC] Temporisation changement de sens
    TonContactorsDropped            : TON;                            // . [LOC] Temporisation retombee contacteurs
    BusinessStepDelay               : TON;                            // . [LOC] Temporisation hausse de palier

    // === VARIABLES LOCALES CALCULÉES ===
    EffectiveDirectionInterlockDelay: TIME;                           // . [LOC] Delai interlock effectif selon sens
    EffectiveStepDelay              : TIME;                           // . [LOC] Delai palier effectif selon sens
    AllContactorsCommandedOff       : BOOL;                           // . [LOC] Tous contacteurs commandes a FALSE
    RampTargetPct                   : REAL;                           // . [LOC] Consigne cible envoyee au SpeedStep
    EffectiveSafeStop               : BOOL;                           // . [LOC] SafeStop effectif incluant permis
    ActiveMaxStep                   : INT;                            // . [LOC] Plafond palier actif dynamique
    FirstScanDone                   : BOOL;                           // . [LOC] Indicateur premier cycle passe

    // Latches défauts
    SpeedStepConfigFaultLatched     : BOOL;

    // Socle défaut
    instFault                       : FB_FaultCore;                   // * [INST] Socle défaut transverse
    instCauses                      : ARRAY[0..15] OF ST_FaultCause;  // . [LOC] Table des causes nommées
END_VAR

// === IMPLEMENTATION ===

{region "§1 Causes & Socle Défaut (évalué en continu)"}
// 🔑 1. FRONTS & REARMEMENT
ResetEdge(CLK := Reset);

IF ResetEdge.Q OR BypassGlobal THEN
    SpeedStepConfigFaultLatched := FALSE;
    ContactorsCheck.StuckClosed := FALSE;
END_IF;

IF Enable AND SpeedStep.ConfigError THEN
    SpeedStepConfigFaultLatched := TRUE;
END_IF;

IF NOT BypassGlobal AND NOT BypassContactorCheck THEN
    instCauses[1].Active   := ContactorsCheck.StuckClosed;
    instCauses[1].Latching := TRUE;
    instCauses[1].Texte    := 'Contacteurs sens ou vitesse bloques fermes';
ELSE
    instCauses[1].Active := FALSE;
END_IF;

instCauses[2].Active   := SpeedStepConfigFaultLatched;
instCauses[2].Latching := TRUE;
instCauses[2].Texte    := 'Configuration table des paliers vitesse invalide';

// Appel du socle de défaut
instFault(Enable := Enable, Reset := Reset, Causes := instCauses);
Fault := instFault.Fault;
{endregion}

{region "§2 Gate neutralisation"}
// 🛑 2. GATE
IF NOT Enable THEN
    SpeedStep(Enable := FALSE, Table := SpeedStepTable);
    RelayFwd := FALSE;
    RelayRev := FALSE;
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    RequestedStep := 0;
    StepNumber := 0;
    CommandedDirection := 0;
    DirectionChangePending := FALSE;
    BusinessStepDelay(IN := FALSE);
    DirectionChangeDelay(IN := FALSE);
    TonContactorsDropped(IN := FALSE);
    ContactorsCheck.Command := FALSE;
    ContactorsCheck.Feedback := NOT FwdRevSpeedFeedbackOff;
    InTopSlowdownZone := FALSE;
    InBottomSlowdownZone := FALSE;
    Ready := FALSE;
    RETURN;
END_IF;

Ready := Enable AND NOT Fault.Latched;
{endregion}

{region "§3 Premier scan et temps de cycle"}
IF NOT FirstScanDone THEN
    FirstScanDone := TRUE;
    CommandedDirection := Direction;
END_IF;

CycleTimeCalc();
{endregion}

{region "§4 Permis et zones de ralentissement"}
EffectiveSafeStop := SafeStop OR (Direction > 0 AND NOT AscentPermit) OR (Direction < 0 AND NOT DescendPermit);

InTopSlowdownZone := Homed AND NOT HomingSuspect AND (CablePosM >= (TopLimitM - CfgSlowdownDistanceM));
InBottomSlowdownZone := Homed AND NOT HomingSuspect AND (CablePosM <= (BottomLimitM + CfgSlowdownDistanceM));

IF Direction > 0 THEN
    ActiveMaxStep := MaxStepAscent;
    IF InTopSlowdownZone THEN
        ActiveMaxStep := MIN(ActiveMaxStep, CfgSlowdownMaxStep);
    END_IF;
ELSIF Direction < 0 THEN
    ActiveMaxStep := CfgMaxStepDescente;
    IF InBottomSlowdownZone THEN
        ActiveMaxStep := MIN(ActiveMaxStep, CfgSlowdownMaxStep);
    END_IF;
ELSE
    ActiveMaxStep := MaxStepAscent;
END_IF;

IF HomingApproachActive THEN
    ActiveMaxStep := 1;
END_IF;
{endregion}

{region "§5 Interlock direction"}
IF Direction > 0 THEN
    EffectiveDirectionInterlockDelay := DirectionInterlockDelayAscent;
ELSE
    EffectiveDirectionInterlockDelay := DirectionInterlockDelayDescent;
END_IF;

DirectionChangeDelay(
    IN := (Direction <> 0) AND (Direction <> CommandedDirection) AND (StepNumber = 0) AND FwdRevSpeedFeedbackOff,
    PT := EffectiveDirectionInterlockDelay
);
DirectionChangeDelayElapsed := DirectionChangeDelay.ET;

IF (CommandedDirection = 0) AND (Direction <> 0) THEN
    CommandedDirection := Direction;
    DirectionChangePending := FALSE;
ELSIF DirectionChangeDelay.Q THEN
    CommandedDirection := Direction;
    DirectionChangePending := FALSE;
ELSIF (Direction <> 0) AND (Direction <> CommandedDirection) THEN
    DirectionChangePending := TRUE;
ELSE
    DirectionChangePending := FALSE;
END_IF;

IF Direction = 0 THEN
    CommandedDirection := 0;
    DirectionChangePending := FALSE;
END_IF;
{endregion}

{region "§6 Paliers vitesse et contacteurs"}
IF DirectionChangePending OR (Direction = 0) OR EffectiveSafeStop OR NOT StartStop THEN
    RampTargetPct := 0.0;
ELSE
    RampTargetPct := LIMIT(0.0, SpeedTgt_Pct, 100.0);
END_IF;

SpeedStep(
    Enable := (RampTargetPct > 0.1),
    SpeedTgt_Pct := RampTargetPct,
    Table := SpeedStepTable,
    HystMargin := SpeedStepHystMargin,
    MaxStepNumber := ActiveMaxStep,
    MeasuredSpeedBand := MeasuredSpeedBand,
    SpeedGuardEnable := SpeedGuardEnable,
    SpeedGuardReady := SpeedGuardReady
);
RequestedStep := SpeedStep.StepNumber;

EffectiveStepDelay := SEL(CommandedDirection = -1, DirectionInterlockDelayAscent, DirectionInterlockDelayDescent) + T#100ms;

BusinessStepDelay(
    IN := (RequestedStep > StepNumber) AND (StepNumber > 0),
    PT := EffectiveStepDelay
);
BusinessStepDelayElapsed := BusinessStepDelay.ET;

IF RequestedStep = 0 THEN
    StepNumber := 0;
ELSIF RequestedStep < StepNumber THEN
    StepNumber := RequestedStep;
ELSIF RequestedStep > StepNumber THEN
    IF StepNumber = 0 THEN
        StepNumber := 1;
    ELSIF BusinessStepDelay.Q THEN
        IF SpeedGuardEnable AND SpeedGuardReady AND (MeasuredSpeedBand < StepNumber) THEN
            SpeedGuardLimited := TRUE;
        ELSE
            StepNumber := StepNumber + 1;
            SpeedGuardLimited := FALSE;
        END_IF;
    END_IF;
END_IF;

IF StepNumber > 0 AND CommandedDirection = 1 THEN
    RelayFwd := TRUE;
    RelayRev := FALSE;
ELSIF StepNumber > 0 AND CommandedDirection = -1 THEN
    RelayFwd := FALSE;
    RelayRev := TRUE;
ELSE
    RelayFwd := FALSE;
    RelayRev := FALSE;
END_IF;

Contactor1 := (StepNumber = 1);
Contactor2 := (StepNumber = 2);
Contactor3 := (StepNumber = 3);
Contactor4 := (StepNumber = 4);

AllContactorsCommandedOff := (NOT RelayFwd) AND (NOT RelayRev)
                             AND (NOT Contactor1) AND (NOT Contactor2) AND (NOT Contactor3) AND (NOT Contactor4);

ContactorsCheck.Command  := NOT AllContactorsCommandedOff;
ContactorsCheck.Feedback := NOT FwdRevSpeedFeedbackOff;
TonContactorsDropped(IN := AllContactorsCommandedOff AND NOT FwdRevSpeedFeedbackOff, PT := ContactorFeedbackTimeout);
IF TonContactorsDropped.Q THEN
    ContactorsCheck.StuckClosed := TRUE;
END_IF;
ContactorsCheck.StuckOpen := FALSE;

IF Fault.Error THEN
    RelayFwd := FALSE;
    RelayRev := FALSE;
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
END_IF;
{endregion}

END_FUNCTION_BLOCK
