(* =======================================================================
   🧩 FB_FaultCore — Socle de remplissage de ST_Fault (vue live + vue latchée)
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : à partir d'une LISTE de causes en clair (ARRAY[0..15] OF
      ST_FaultCause), produire la brique ST_Fault :
        • Error / ErrorId   : vue LIVE, suit les causes Active, retombe seule.
        • Latched / LatchedId: vue LATCHÉE, chaque bit figé à l'apparition d'une
          cause `Active AND Latching`, conservé jusqu'au front Reset.
   🔒 Sécurité :
      - interlock TOUJOURS sur `Causes[i].Active` (cause brute), jamais sur le latch.
      - Reset TOUJOURS effectif, jamais conditionné (std §9) — agit même Enable=FALSE.
      - Latching=FALSE → cause visible en LIVE seulement (fail-safe : reste visible).
      - un défaut laté non acquitté reste publié même FB désactivé (pas d'effacement
        silencieux sur bascule Enable OFF→ON sans Reset).
      - une cause à acquitter réapparue re-arme son bit (ré-alarme).
   🧬 Un seul socle, réutilisé par tout FB `standard` qui expose un ST_Fault.
      Pas de State, pas de Warning, pas de texte — hors périmètre de ce socle.
   📄 Doc : DOC/STDS/CODE_QUALITY_STANDARDS.md §2quinquies / §9
   ======================================================================= *)

FUNCTION_BLOCK FB_FaultCore
VAR_INPUT
    Enable  : BOOL;                          // --> [CMD] Autorisation générale du FB porteur
    Reset   : BOOL;                          // --> [CMD] Front d'acquittement (jamais conditionné, §9)
    Causes  : ARRAY[0..15] OF ST_FaultCause; // --> [CMD] Liste des causes en clair (Active / Latching / Texte)
END_VAR
VAR_OUTPUT
    Ready   : BOOL;                          // <-- [STAT] FB porteur autorisé (= Enable)
    Fault   : ST_Fault;                      // <-- [DIAG] Brique défaut (mappée 1:1 sur la sortie du FB porteur)
END_VAR
VAR
    ResetEdge : R_TRIG;                      // . [LOC] Front montant Reset
    Latch     : ARRAY[0..15] OF BOOL;        // . [LOC] Bit laté par cause, jusqu'au Reset
    i         : INT;                         // . [LOC] Index de parcours de la liste
END_VAR

{region "§1 Reset (front, toujours effectif)"}
// Front Reset : efface tous les latches. Jamais conditionné, agit même Enable=FALSE (§9).
Ready := Enable;
ResetEdge(CLK := Reset);
IF ResetEdge.Q THEN
    FOR i := 0 TO 15 DO Latch[i] := FALSE; END_FOR;
END_IF;
{endregion}

{region "§2 Vue LIVE — Error / ErrorId"}
// Bitfield des causes présentes MAINTENANT. Retombe seule à 0 quand les causes disparaissent.
// Non évaluée si Enable=FALSE (pas de lecture de cause hors autorisation).
Fault.ErrorId := 0;
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active THEN
            Fault.ErrorId := Fault.ErrorId OR SHL(WORD#1, INT_TO_DWORD(i));
        END_IF;
    END_FOR;
END_IF;
Fault.Error := (Fault.ErrorId <> 0);
{endregion}

{region "§3 Armement des latches (causes à acquitter)"}
// Une cause Active ET Latching arme son bit ; le bit reste jusqu'au front Reset (§1),
// même si la cause disparaît. Réapparition => ré-armement (ré-alarme).
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active AND Causes[i].Latching THEN
            Latch[i] := TRUE;
        END_IF;
    END_FOR;
END_IF;
{endregion}

{region "§4 Vue LATCHÉE — Latched / LatchedId"}
// Toujours publiée, y compris Enable=FALSE : un défaut non acquitté reste visible.
Fault.LatchedId := 0;
FOR i := 0 TO 15 DO
    IF Latch[i] THEN
        Fault.LatchedId := Fault.LatchedId OR SHL(WORD#1, INT_TO_DWORD(i));
    END_IF;
END_FOR;
Fault.Latched := (Fault.LatchedId <> 0);
{endregion}

END_FUNCTION_BLOCK

    MaintHomingRequiredM2 := BootHomingRequiredM2;

// Armé AVANT l'arbitrage de mode (§3) et non à la sortie du mode maintenance :
//   1. pas de fenêtre d'un scan où le re-homing serait perdu malgré un override effectif ;
//   2. pas de perte de l'obligation sur coupure / download pendant la maintenance
//      (la valeur est écrite dans le miroir PERSISTENT dès le scan courant, pas à la sortie).
// Le re-homing ne bloque PAS la sélection SEMI_AUTO (T207) : il est publié à l'IHM

END_IF;

    MaintHomingRequiredM2 := FALSE;
END_IF;

// la sélection n'est pas refusée, le mouvement est coupé en aval (FB_Cycle/FB_Safety_Winch).

instCauses[1].Texte    := 'Changement de mode refuse : arreter les treuils avant de changer de mode';

    Auth.HomingRequiredM2 := MaintHomingRequiredM2;

// ⚙️ 3. ARBITRAGE DU MODE

// 3.1 ARRÊT MÉCANIQUE CONFIRMÉ DES DEUX TREUILS (condition de bascule, AF-05 §4bis)
// Même composite que l'armement Méca B : contacteurs retombés + frein serré, redondé par
// la mesure absolue de vitesse. Vaut pour les DEUX treuils (M1 ET M2).
ModeChangeAllowed := M1ContactorsReleased AND NOT M1BrakeIsOpen
                     AND (ABS(M1SpeedAbsMps) < MovementSpeedThresholdMps)
                     AND M2ContactorsReleased AND NOT M2BrakeIsOpen
                     AND (ABS(M2SpeedAbsMps) < MovementSpeedThresholdMps);

// 3.2 ARBITRAGE — le mode suit la sélection IHM (découplé de l'AU/défaut, T207)
// La sélection de mode n'est JAMAIS bloquée par l'AU, l'arrêt d'urgence ou un défaut :
// le mode est un état de sélection, le mouvement est coupé en aval (FB_Cycle, FB_Safety_Winch,
// FB_Winch) par PowerContactorEngaged et les interlocks. Seule la bascule N1/N2/SEMI_AUTO
// reste soumise à l'arrêt mécanique confirmé des treuils (ModeChangeAllowed).

      AND (SelMode <> E_Mode.DISABLE) AND (PrevArbitratedMode <> E_Mode.DISABLE)
      AND NOT ModeChangeAllowed THEN
    // Bascule N1/N2/SEMI_AUTO refusée : le mode courant est MAINTENU.
    // ⚠️ Exception de sécurité (contrat D2) : une transition vers ou depuis DISABLE n'est
    // JAMAIS bloquée — sortir d'un mode maintenance ne doit jamais dépendre d'un treuil à l'arrêt.
    Auth.Mode := PrevArbitratedMode;
    Auth.ModeChangePendingBlocked := TRUE;
ELSE
    Auth.Mode := SelMode;
    Auth.ModeChangePendingBlocked := FALSE;
END_IF;

PrevArbitratedMode := Auth.Mode;

        Auth.InhibitM1 := FALSE;
    ELSE
        Auth.InhibitM1 := TRUE;
        Auth.InhibitM2 := FALSE;
        SelSyncEnable := FALSE;
    END_IF;
ELSIF InhibitM2Request AND NOT PrevInhibitM2 THEN
    IF PrevInhibitM1 THEN
        Auth.InhibitM2 := FALSE;
    ELSE
        Auth.InhibitM2 := TRUE;
        Auth.InhibitM1 := FALSE;
        SelSyncEnable := FALSE;
    END_IF;
ELSE
    IF InhibitM1Request AND InhibitM2Request THEN
        Auth.InhibitM1 := FALSE;
        Auth.InhibitM2 := FALSE;
    END_IF;
END_IF;

PrevInhibitM1 := InhibitM1Request;
PrevInhibitM2 := InhibitM2Request;

Auth.InhibitM1 := (Auth.Mode = E_Mode.MAINT_N2) AND Auth.InhibitM1;
Auth.InhibitM2 := (Auth.Mode = E_Mode.MAINT_N2) AND Auth.InhibitM2;

// 4ter. AUTORISATION ARRÊT HAUT (HomingApproach)
Auth.HomingApproachEnable := (Auth.Mode = E_Mode.MAINT_N2) AND SelHomingApproachEnable;

// 4quater. SÉLECTEUR JOYSTICK TREUIL

    Auth.JoystickWinchSelectArbitrated := JoystickWinchSelectRequest;
ELSE
    Auth.JoystickWinchSelectArbitrated := 0;  // Couplé nominal forcé
END_IF;

// 4quinquies. CIBLE TRANSLATION MAINTENANCE
Auth.MaintenanceM3TargetEnable := ((Auth.Mode = E_Mode.MAINT_N2) OR (Auth.Mode = E_Mode.MAINT_N1)) AND SelMaintenanceZoneAccess;

// Re-vérification sécurité si sortie MAINT
IF (Auth.Mode <> E_Mode.MAINT_N2) AND (Auth.Mode <> E_Mode.MAINT_N1) THEN
    Auth.InhibitM1 := FALSE;
    Auth.InhibitM2 := FALSE;
    Auth.HomingApproachEnable := FALSE;
    Auth.MaintenanceM3TargetEnable := FALSE;
    Auth.JoystickWinchSelectArbitrated := 0;
END_IF;
{endregion}
