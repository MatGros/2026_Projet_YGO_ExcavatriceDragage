FUNCTION_BLOCK FB_CycleTime
VAR_INPUT

END_VAR
VAR_OUTPUT

{endregion}

IF NOT InitDone THEN

    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).

    // === CONSIGNES ET TAUX DE RAMPE ===
    Target                  : REAL;               // --> [CMD] Valeur cible visee (-100.0..+100.0 %)
    AccelRate               : REAL;               // --> [CFG] Taux d'acceleration (%/s)
    DecelRate               : REAL;               // --> [CFG] Taux de deceleration (%/s)
    CycleTimeS              : REAL;               // --> [CFG] Temps de cycle reel en secondes (dt)
END_VAR
VAR_OUTPUT
    // === VALEUR RAMPEE ===
    Current                 : REAL;               // <-- [STAT] Valeur courante amortie (-100.0..+100.0 %)

    TargetLimited           : REAL;               // . [LOC] Cible bornee a [-100.0, +100.0]
END_VAR

// Détection du comportement (Accélération vs Décélération)
// Classification par CÔTÉ DE ZÉRO + sens de déplacement, plus par signe brut de Target seul
// (robuste à une traversée de zéro rapide).

TargetLimited := LIMIT(-100.0, Target, 100.0);
Current := LIMIT(-100.0, Current, 100.0);

IF (Current >= 0.0 AND TargetLimited >= Current) OR (Current <= 0.0 AND TargetLimited <= Current) THEN
    StepX := AccelRate * CycleTimeS; // S'éloigne du zéro (reste du même côté, va plus loin)
ELSE
    StepX := DecelRate * CycleTimeS; // Retourne vers zéro, ou le traverse vers l'autre signe
END_IF;

// Application de la rampe
IF Current < TargetLimited THEN
    Current := Current + StepX;
    IF Current > TargetLimited THEN Current := TargetLimited; END_IF;
ELSIF Current > TargetLimited THEN
    Current := Current - StepX;
    IF Current < TargetLimited THEN Current := TargetLimited; END_IF;
END_IF;

// 🛡️ Limitation finale : même une valeur d'état déjà corrompue reste bornée.
Current := LIMIT(-100.0, Current, 100.0);

(* =======================================================================
   ⚙️ FB_CycleTime — Mesure du temps de cycle réel de la tâche
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul du dt réel entre deux exécutions successives
   ⚙️ Calcul : Delta horloge système TIME() converti en secondes
   🛡️ Robustesse : dt borné (0 < dt <= CST_MaxCycleDeltaMs) sinon valeur de secours —
      neutralise les artefacts de rebouclage TIME() (49,7 j) et de reprise (online-change, debug)
   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md
   ======================================================================= *)
FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

FUNCTION_BLOCK PUBLIC FB_Translation_PositionEstimator

    Reset                : BOOL;   (* --> [CMD] Front d'acquittement / réinitialisation *)
    DriveActualFreqHz    : REAL;   (* --> [HW] Fréquence réelle variateur en Hz (ex: 5.0 Hz) *)
    ReqTremie            : BOOL;   (* --> [CMD] Demande explicite vers Trémie *)
    ReqMaintenance       : BOOL;   (* --> [CMD] Demande explicite vers Maintenance *)
    
    (* 5 Capteurs de position fixes *)
    SensorMaintenance    : BOOL;   (* --> [HW] Capteur Maintenance (30.0 m) *)
    SensorP1             : BOOL;   (* --> [HW] Capteur P1 (20.0 m) *)
    SensorP2             : BOOL;   (* --> [HW] Capteur P2 (15.0 m) *)
    SensorPV             : BOOL;   (* --> [HW] Capteur PV (5.0 m) *)
    SensorTremie         : BOOL;   (* --> [HW] Capteur Trémie (0.0 m) *)

    (* Positions de référence calibrées (mètres) depuis GVL_PERSISTENT *)
    PosMaintenanceM      : REAL := 30.0;   (* --> [CFG] Position Maintenance *)
    PosP1M               : REAL := 20.0;   (* --> [CFG] Position P1 *)
    PosP2M               : REAL := 15.0;   (* --> [CFG] Position P2 *)

    PosTremieM           : REAL := 0.0;   (* --> [CFG] Position Trémie *)

{region "§1 Gate et initialisation"}
IF NOT Enable THEN
    Recalibrated := FALSE;
    RecalibratedSensorId := -1;

    RETURN;
END_IF;

IF ColdStart THEN
    PositionEstimatedM := PersistedPositionM;
    Initialized := PersistedInitialized;
    ColdStart := FALSE;
END_IF;

CycleTimeCalc(DefaultValueS := 0.010);
Recalibrated := FALSE;

{region "§2 Fronts capteurs"}
// 1. Détection des fronts montants sur franchissement des capteurs fixes
TrigTremie(CLK := SensorTremie);
TrigPV(CLK := SensorPV);
TrigP2(CLK := SensorP2);
TrigP1(CLK := SensorP1);
TrigMaintenance(CLK := SensorMaintenance);
{endregion}

{region "§3 Recalage et intégration"}
// 2. Recalage absolu sur franchissement capteur — UNIQUEMENT après le premier scan utile.
//    Au 1er scan, un capteur "déjà présent" à l'initialisation (sim démarre à la trémie :
//    5 capteurs actifs) n'est pas un franchissement réel ; sinon le recalage IF/ELSIF
//    partirait de Maintenance (premier dans l'ordre) et écraserait la position à 30 m.
IF NOT FirstScanDone THEN
    FirstScanDone := TRUE;
ELSE
    IF TrigMaintenance.Q THEN
        PositionEstimatedM := PosMaintenanceM;
        Recalibrated := TRUE;
        RecalibratedSensorId := 0;
    ELSIF TrigP1.Q THEN
        PositionEstimatedM := PosP1M;
        Recalibrated := TRUE;
        RecalibratedSensorId := 1;
    ELSIF TrigP2.Q THEN
        PositionEstimatedM := PosP2M;
        Recalibrated := TRUE;
        RecalibratedSensorId := 2;
    ELSIF TrigPV.Q THEN
        PositionEstimatedM := PosPVM;
        Recalibrated := TRUE;
        RecalibratedSensorId := 3;
    ELSIF TrigTremie.Q THEN
        PositionEstimatedM := PosTremieM;
        Recalibrated := TRUE;
        RecalibratedSensorId := 4;
    END_IF;
END_IF;

// 3. Initialisation au tout premier démarrage sur le premier capteur actif vu (si non initialisé)
IF NOT Initialized THEN
    IF SensorTremie THEN PositionEstimatedM := PosTremieM; Initialized := TRUE;
    ELSIF SensorPV THEN PositionEstimatedM := PosPVM; Initialized := TRUE;
    ELSIF SensorP2 THEN PositionEstimatedM := PosP2M; Initialized := TRUE;
    ELSIF SensorP1 THEN PositionEstimatedM := PosP1M; Initialized := TRUE;
    ELSIF SensorMaintenance THEN PositionEstimatedM := PosMaintenanceM; Initialized := TRUE;
    END_IF;
END_IF;

// 4. Intégration temporelle continue (Odométrie cycle par cycle)
// Les bits explicites portent le sens sémantique. Une demande contradictoire est
// neutralisée : aucun déplacement estimé ne doit être inventé.
IF DriveActualFreqHz > 0.1 THEN
    IF ReqTremie AND NOT ReqMaintenance THEN
        PositionEstimatedM := PositionEstimatedM - (DriveActualFreqHz * GainMetersPerHzSec * CycleTimeCalc.CycleTimeS);
    ELSIF ReqMaintenance AND NOT ReqTremie THEN
        PositionEstimatedM := PositionEstimatedM + (DriveActualFreqHz * GainMetersPerHzSec * CycleTimeCalc.CycleTimeS);
    END_IF;
END_IF;
{endregion}

{region "§4 Bornage"}
// Bornage de sécurité pour éviter la dérive infinie hors rail
// Convention 0 m=Trémie, 30 m=Maintenance : bornage [Trémie-0.5, Maintenance+0.5].
PositionEstimatedM := LIMIT(PosTremieM - 0.5, PositionEstimatedM, PosMaintenanceM + 0.5);
{endregion}
