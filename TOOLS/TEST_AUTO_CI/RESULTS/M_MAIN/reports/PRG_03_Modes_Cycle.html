(* =======================================================================
   🧩 FB_FaultCore — Socle de remplissage de ST_Fault (vue live + vue latchée)
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : à partir d'une LISTE de causes en clair (ARRAY[0..15] OF
      ST_FaultCause), produire la brique ST_Fault :
        • Error / ErrorId   : vue LIVE, suit les causes Active, retombe seule.
        • Latched / LatchedId: vue LATCHÉE, chaque bit figé à l'apparition d'une
          cause `Active AND Latching`, conservé jusqu'au front Reset.
   🔒 Sécurité :
      - interlock TOUJOURS sur `Causes[i].Active` (cause brute), jamais sur le latch.
      - Reset TOUJOURS effectif, jamais conditionné (std §9) — agit même Enable=FALSE.
      - Latching=FALSE → cause visible en LIVE seulement (fail-safe : reste visible).

   🧬 Un seul socle, réutilisé par tout FB `standard` qui expose un ST_Fault.
      Pas de State, pas de Warning, pas de texte — hors périmètre de ce socle.
   📄 Doc : DOC/STDS/CODE_QUALITY_STANDARDS.md §2quinquies / §9

    Reset   : BOOL;                          // --> [CMD] Front d'acquittement (jamais conditionné, §9)
    Causes  : ARRAY[0..15] OF ST_FaultCause; // --> [CMD] Liste des causes en clair (Active / Latching / Texte)
END_VAR
VAR_OUTPUT
    Ready   : BOOL;                          // <-- [STAT] FB porteur autorisé (= Enable)
    Fault   : ST_Fault;                      // <-- [DIAG] Brique défaut (mappée 1:1 sur la sortie du FB porteur)
END_VAR

    ResetEdge : R_TRIG;                      // . [LOC] Front montant Reset
    Latch     : ARRAY[0..15] OF BOOL;        // . [LOC] Bit laté par cause, jusqu'au Reset
    i         : INT;                         // . [LOC] Index de parcours de la liste
END_VAR

{region "§1 Reset (front, toujours effectif)"}
// Front Reset : efface tous les latches. Jamais conditionné, agit même Enable=FALSE (§9).
Ready := Enable;
ResetEdge(CLK := Reset);
IF ResetEdge.Q THEN
    FOR i := 0 TO 15 DO Latch[i] := FALSE; END_FOR;
END_IF;
{endregion}

{region "§2 Vue LIVE — Error / ErrorId"}
// Bitfield des causes présentes MAINTENANT. Retombe seule à 0 quand les causes disparaissent.
// Non évaluée si Enable=FALSE (pas de lecture de cause hors autorisation).
Fault.ErrorId := 0;
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active THEN
            Fault.ErrorId := Fault.ErrorId OR SHL(WORD#1, INT_TO_DWORD(i));
        END_IF;
    END_FOR;
END_IF;
Fault.Error := (Fault.ErrorId <> 0);
{endregion}

{region "§3 Armement des latches (causes à acquitter)"}
// Une cause Active ET Latching arme son bit ; le bit reste jusqu'au front Reset (§1),
// même si la cause disparaît. Réapparition => ré-armement (ré-alarme).
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active AND Causes[i].Latching THEN
            Latch[i] := TRUE;
        END_IF;
    END_FOR;
END_IF;
{endregion}

{region "§4 Vue LATCHÉE — Latched / LatchedId"}
// Toujours publiée, y compris Enable=FALSE : un défaut non acquitté reste visible.
Fault.LatchedId := 0;
FOR i := 0 TO 15 DO
    IF Latch[i] THEN
        Fault.LatchedId := Fault.LatchedId OR SHL(WORD#1, INT_TO_DWORD(i));
    END_IF;
END_FOR;

{endregion}

   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md

FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

   ───────────────────────────────────────────────────────────────

   🔒 Polarité : Distingue l'éloignement de zéro (Accel) et le retour à zéro (Decel)
   📄 Doc métier : DOC/AF/AF_Partie-11_Fonction_Translation_v2.3.md §1 (TC-P11-040)

FUNCTION_BLOCK FB_Ramp
VAR_INPUT
    // === CONSIGNES ET TAUX DE RAMPE ===
    Target                  : REAL;               // --> [CMD] Valeur cible visee (-100.0..+100.0 %)
    AccelRate               : REAL;               // --> [CFG] Taux d'acceleration (%/s)
    DecelRate               : REAL;               // --> [CFG] Taux de deceleration (%/s)
    CycleTimeS              : REAL;               // --> [CFG] Temps de cycle reel en secondes (dt)
END_VAR
VAR_OUTPUT
    // === VALEUR RAMPEE ===
    Current                 : REAL;               // <-- [STAT] Valeur courante amortie (-100.0..+100.0 %)
END_VAR
VAR
    // === CALCULS INTERNES ===
    StepX                   : REAL;               // . [LOC] Pas de variation pour le cycle courant
    TargetLimited           : REAL;               // . [LOC] Cible bornee a [-100.0, +100.0]
END_VAR

// Détection du comportement (Accélération vs Décélération)
// Classification par CÔTÉ DE ZÉRO + sens de déplacement, plus par signe brut de Target seul
// (robuste à une traversée de zéro rapide).
// 🛡️ Garde-fou : la rampe ne sort jamais de la plage normalisée ±100 %.
TargetLimited := LIMIT(-100.0, Target, 100.0);
Current := LIMIT(-100.0, Current, 100.0);

IF (Current >= 0.0 AND TargetLimited >= Current) OR (Current <= 0.0 AND TargetLimited <= Current) THEN
    StepX := AccelRate * CycleTimeS; // S'éloigne du zéro (reste du même côté, va plus loin)
ELSE
    StepX := DecelRate * CycleTimeS; // Retourne vers zéro, ou le traverse vers l'autre signe
END_IF;

// Application de la rampe
IF Current < TargetLimited THEN
    Current := Current + StepX;
    IF Current > TargetLimited THEN Current := TargetLimited; END_IF;
ELSIF Current > TargetLimited THEN
    Current := Current - StepX;
    IF Current < TargetLimited THEN Current := TargetLimited; END_IF;
END_IF;

// 🛡️ Limitation finale : même une valeur d'état déjà corrompue reste bornée.
Current := LIMIT(-100.0, Current, 100.0);

// ⚙️ 3. ARBITRAGE DU MODE

// 3.1 ARRÊT MÉCANIQUE CONFIRMÉ DES DEUX TREUILS (condition de bascule, AF-05 §4bis)
// Même composite que l'armement Méca B : contacteurs retombés + frein serré, redondé par
// la mesure absolue de vitesse. Vaut pour les DEUX treuils (M1 ET M2).
ModeChangeAllowed := M1ContactorsReleased AND NOT M1BrakeIsOpen
                     AND (ABS(M1SpeedAbsMps) < MovementSpeedThresholdMps)
                     AND M2ContactorsReleased AND NOT M2BrakeIsOpen
                     AND (ABS(M2SpeedAbsMps) < MovementSpeedThresholdMps);

// Détail des 6 sous-conditions (diagnostic IHM) — TRUE = condition satisfaite.
// Permet à l'opérateur de voir EXACTEMENT ce qui bloque la bascule de mode.
Auth.M1ContactorsReleased := M1ContactorsReleased;
Auth.M1BrakeApplied       := NOT M1BrakeIsOpen;
Auth.M1SpeedLow           := ABS(M1SpeedAbsMps) < MovementSpeedThresholdMps;
Auth.M2ContactorsReleased := M2ContactorsReleased;
Auth.M2BrakeApplied       := NOT M2BrakeIsOpen;
Auth.M2SpeedLow           := ABS(M2SpeedAbsMps) < MovementSpeedThresholdMps;

// 3.2 ARBITRAGE — le mode suit la sélection IHM (découplé de l'AU/défaut)
// La sélection de mode n'est JAMAIS bloquée par l'AU, l'arrêt d'urgence ou un défaut :
// le mode est un état de sélection, le mouvement est coupé en aval (FB_Cycle, FB_Safety_Winch,
// FB_Winch) par PowerContactorEngaged et les interlocks. Seule la bascule N1/N2/SEMI_AUTO
// reste soumise à l'arrêt mécanique confirmé des treuils (ModeChangeAllowed).
// 3.2bis (AF-05 §4bis - referencement benne guide) : SEMI_AUTO est un mode "homed-only".
// Une selection SEMI_AUTO sans datum machine valide (MachineHomed) est REFUSEE et le mode
// est force vers MAINT_N1 (expulsion). Vaut aussi si MachineHomed retombe alors qu'on EST
// en SEMI_AUTO (perte de datum -> bascule MAINT_N1). Le SafeStop des treuils sur perte de
// datum en mouvement est assure en aval (PRG_04 via MachineHoming.HomingLoss). L'expulsion
// n'est jamais bloquee par l'arret mecanique (comme la sortie DISABLE, contrat D2).
IF (SelMode = E_Mode.SEMI_AUTO) AND NOT MachineHomed THEN
    Auth.Mode := E_Mode.MAINT_N1;
    Auth.ModeChangePendingBlocked := TRUE;
    // Motif distinct : perte de datum EN cours de SEMI_AUTO (arret controle assure en
    // aval par MachineHoming.HomingLoss si les treuils bougeaient) vs simple selection
    // refusee. Aucune coupure de puissance, aucun redemarrage automatique.
    IF PrevArbitratedMode = E_Mode.SEMI_AUTO THEN
        Auth.ModeChangeBlockReason := 'Datum machine perdu : arret controle';
    ELSE
        Auth.ModeChangeBlockReason := 'SEMI_AUTO refuse : datum machine absent';
    END_IF;
ELSIF (SelMode <> PrevArbitratedMode)
      AND (SelMode <> E_Mode.DISABLE) AND (PrevArbitratedMode <> E_Mode.DISABLE)
      AND NOT ModeChangeAllowed THEN
    // Bascule N1/N2/SEMI_AUTO refusée : le mode courant est MAINTENU.
    // ⚠️ Exception de sécurité (contrat D2) : une transition vers ou depuis DISABLE n'est
    // JAMAIS bloquée — sortir d'un mode maintenance ne doit jamais dépendre d'un treuil à l'arrêt.
    Auth.Mode := PrevArbitratedMode;
    Auth.ModeChangePendingBlocked := TRUE;
    // Motif du blocage (texte IHM court) — première condition échouée, ordre M1 puis M2.
    IF NOT M1ContactorsReleased THEN
        ModeChangeBlockReason := 'Chgt mode refuse : contacteur M1';
    ELSIF M1BrakeIsOpen THEN
        ModeChangeBlockReason := 'Chgt mode refuse : frein M1';
    ELSIF ABS(M1SpeedAbsMps) >= MovementSpeedThresholdMps THEN
        ModeChangeBlockReason := 'Chgt mode refuse : M1 en mvt';
    ELSIF NOT M2ContactorsReleased THEN
        ModeChangeBlockReason := 'Chgt mode refuse : contacteur M2';
    ELSIF M2BrakeIsOpen THEN
        ModeChangeBlockReason := 'Chgt mode refuse : frein M2';
    ELSIF ABS(M2SpeedAbsMps) >= MovementSpeedThresholdMps THEN
        ModeChangeBlockReason := 'Chgt mode refuse : M2 en mvt';
    ELSE
        ModeChangeBlockReason := '';
    END_IF;
    Auth.ModeChangeBlockReason := ModeChangeBlockReason;
ELSE
    Auth.Mode := SelMode;
    Auth.ModeChangePendingBlocked := FALSE;
    Auth.ModeChangeBlockReason := '';
END_IF;

PrevArbitratedMode := Auth.Mode;
{endregion}

PrevInhibitM2 := InhibitM2Request;

    Auth.InhibitM1 := FALSE;
    Auth.InhibitM2 := FALSE;
    Auth.HomingApproachEnable := FALSE;
    Auth.MaintenanceM3TargetEnable := FALSE;
    Auth.JoystickWinchSelectArbitrated := 0;
END_IF;
{endregion}

{region "§1 Causes & Socle Défaut"}
// 🔑 1. FRONTS ET CAUSES
ResetEdge(CLK := Reset);
StartEdge(CLK := StartCycle);
DeadmanArmedEdge(CLK := DeadmanArmed);
AbortEdge(CLK := AbortCycle);
SkipDrainEdge(CLK := SkipDrainRequest);

// ⏱️ Chronomètre du cycle : compte tant que le séquenceur est actif (Lifecycle.Busy).
// .ET = durée écoulée du cycle en cours ; figée dans LastCycleDuration à l'achèvement (X13).
CycleRunTimer(IN := Lifecycle.Busy, PT := T#12h);
CurrentCycleElapsed := CycleRunTimer.ET;

// Durée d'égouttage effective : config IHM si valide, sinon défaut interne.
DrainTimeEff := SEL(CfgDrainTime <= T#0s, CfgDrainTime, DrainingTime);

// ⏱️ Surveillance indépendante de la remontée lente
// X7_CTRL_ASCENT -> X8_ASCENT_LOADED : TON de stabilisation palier 1.
StabilizationTimer(IN := (State = E_CycleStep.X7_CTRL_ASCENT), PT := CtrlAscentTimeout);

// 📏 Comparaison vitesse réelle M1/M2
SpeedMismatchMps := ABS(M1_MeasuredSpeedMps - M2_MeasuredSpeedMps);
SpeedMismatchActive := (State = E_CycleStep.X7_CTRL_ASCENT)
                       AND JoystickDeflected
                       AND (SpeedMismatchThresholdMps > 0.0)
                       AND (SpeedMismatchTimeout > T#0ms)
                       AND (SpeedMismatchMps > SpeedMismatchThresholdMps);
// X7_CTRL_ASCENT -> STABILIZING : TON confirmation écart vitesse.
SpeedMismatchTimer(IN := SpeedMismatchActive, PT := SpeedMismatchTimeout);
SpeedMismatchConfirmed := SpeedMismatchTimer.Q;

// Tempo max d'étape (R5) — backstop durée d'état, lancé SEULEMENT si pilotage joystick actif.
// Xi -> STABILIZING sur dépassement. Réarmé à CHAQUE changement d'étape (IN=FALSE 1 scan) :
// chaque étape dispose de sa propre fenêtre StepMaxTimeout, la tempo n'est PAS cumulative
// sur l'ensemble du cycle (un cycle réel long ne tombe plus en STABILIZING intempestif).
StepMaxTimer(IN := (State = StepMaxPrevState)
                  AND JoystickDeflected
                  AND (State <> E_CycleStep.X0_PREPARATION)
                  AND (State <> E_CycleStep.X13_DONE_SYNC)
                  AND (State <> E_CycleStep.STABILIZING),
             PT := StepMaxTimeout);
StepMaxPrevState := State;

// Egouttage X9 : TON gaté sur l'étape (remis à 0 hors X9). Remplace l'ancien IN:=TRUE câblé
// qui pouvait laisser Q déjà vrai à l'entrée en X9 (temps d'égouttage non respecté).
DrainingTimer(IN := (State = E_CycleStep.X9_DRAIN_PAUSE), PT := DrainTimeEff);

// Table des causes de défaut
instCauses[0].Active   := LimitLegalReached AND (State = E_CycleStep.X4_DESCEND_OPEN);
instCauses[0].Latching := TRUE;
instCauses[0].Texte    := 'Limite legale de profondeur atteinte pendant descente';

instCauses[1].Active   := WinchSyncError AND (State <> E_CycleStep.X0_PREPARATION) AND (State <> E_CycleStep.STABILIZING);
instCauses[1].Latching := TRUE;
instCauses[1].Texte    := 'Desynchronisme treuils M1/M2 bloquant';

instCauses[2].Active   := SpeedMismatchConfirmed AND (State <> E_CycleStep.X0_PREPARATION) AND (State <> E_CycleStep.STABILIZING);
instCauses[2].Latching := TRUE;
instCauses[2].Texte    := 'Ecart de vitesse M1/M2 confirme en montee';

instCauses[3].Active   := NOT HeartbeatIhmOk AND (State <> E_CycleStep.X0_PREPARATION) AND (State <> E_CycleStep.STABILIZING);
instCauses[3].Latching := TRUE;
instCauses[3].Texte    := 'Perte communication IHM pendant cycle';

instCauses[4].Active   := StepMaxTimer.Q AND (State <> E_CycleStep.STABILIZING);
instCauses[4].Latching := TRUE;
instCauses[4].Texte    := 'Depassement tempo maximale d etape (blocage mouvement)';

instCauses[5].Active   := (StabilizationTimer.Q OR (ABS(M1_CablePosM - M2_CablePosM) > CtrlAscentToleranceM))
                          AND (State = E_CycleStep.X7_CTRL_ASCENT);
instCauses[5].Latching := TRUE;
instCauses[5].Texte    := 'Defaut de stabilisation / tolerance ecart codeurs en remontee controle';

// Backstop anti-telescopage : ecart de position M1/M2 excessif pendant une phase
// reellement couplee (X4 plongee, X5 remontee, X8 montee en charge). X7 a deja son
// propre controle (instCauses[5]). Seuil grossier (0,50 m) : filet local si
// FB_SyncDeviation (PRG_04) ne coupe pas.
instCauses[6].Active   := (ABS(M1_CablePosM - M2_CablePosM) > CST_CoupledPosBacklashM)
                          AND JoystickDeflected
                          AND ((State = E_CycleStep.X4_DESCEND_OPEN)
                            OR (State = E_CycleStep.X5_BOTTOM_CONFIRMED)
                            OR (State = E_CycleStep.X8_ASCENT_LOADED)
                            OR (State = E_CycleStep.X11C_DUMP_REPOSITION));
instCauses[6].Latching := TRUE;
instCauses[6].Texte    := 'Ecart de position M1/M2 excessif en phase couplee (anti-telescopage)';

// Appel du socle de défaut
instFault(Enable := Enable, Reset := Reset, Causes := instCauses);
Fault := instFault.Fault;
ErrorEdge(CLK := Fault.Error);
{endregion}

{region "§2 Gate securite & Bascule Mode"}
// 🛑 2. GATES DE SÉCURITÉ ET NEUTRALISATION
// Le mode SEMI_AUTO peut être sélectionné sous AU/défaut (FB_Modes), mais le cycle
// ne tourne JAMAIS sans puissance (PowerContactorEngaged) ni avec un codeur défaillant
// (EncoderFaultPresent) — les données de position invalides rendraient le Grafcet erroné.
IF NOT Enable OR NOT PowerContactorEngaged OR EncoderFaultPresent THEN
    WinchM1Cmd.RunRequest     := FALSE;
    WinchM1Cmd.ReqAscent      := FALSE;
    WinchM1Cmd.ReqDescend     := FALSE;
    WinchM1Cmd.StepTgt        := 0;

    WinchM2Cmd.RunRequest     := FALSE;
    WinchM2Cmd.ReqAscent      := FALSE;
    WinchM2Cmd.ReqDescend     := FALSE;
    WinchM2Cmd.StepTgt        := 0;

    TranslationCmd.ReqStart   := FALSE;
    TranslationCmd.PositionTgt  := 0;

    BucketCmd.ReqOpen         := FALSE;
    BucketCmd.ReqClose        := FALSE;
    BucketCmd.ReqKoboldMeasureEnable := FALSE;

    // Si on quitte le semi-auto alors qu'un cycle était en cours, on mémorise l'étape pour la pause
    IF (State <> E_CycleStep.X0_PREPARATION) AND (State <> E_CycleStep.X13_DONE_SYNC) AND (State <> E_CycleStep.STABILIZING) THEN
        PausedState := State;
        WaitingResume := TRUE;
    END_IF;

    CycleStep            := State;
    CycleStateStr        := 'Cycle desactive : securite/codeur/mode';
    Ready                := FALSE;
    Lifecycle.Busy       := FALSE;
    Lifecycle.Done       := FALSE;
    SpeedMismatchTimer(IN := FALSE, PT := SpeedMismatchTimeout);
    SpeedMismatchMps := 0.0;
    SpeedMismatchActive := FALSE;
    SpeedMismatchConfirmed := FALSE;
    OperatorActionId := 0;
    IF EncoderFaultPresent THEN
        OperatorAction := 'Cycle non autorise : defaut codeur present. Reinitialiser le defaut codeur.';
    ELSE
        OperatorAction := 'Passer en mode Semi-Auto et engager la securite pour activer le cycle.';
    END_IF;
    ExpectedAxis := E_OperatorAxis.NONE;
    ExpectedDirection := 0;
    WaitingForOperator := FALSE;
    WaitingForProcess := FALSE;
    RequestActive := FALSE;
    RETURN;
END_IF;

Ready := Enable AND NOT Fault.Latched;

// 🛡️ Gate continu homme-mort : le mouvement n'est autorisé que si le manche est
// défléchi (JoystickDeflected) ET l'homme-mort est armé (DeadmanArmed). Chaque
// commande de mouvement est gatée par ce permis à chaque scan (pas seulement au
// démarrage). Le séquenceur (transitions X0→X13) reste piloté par JoystickDeflected.
CycleMotionPermit := JoystickDeflected AND DeadmanArmed;

// Au retour en Semi-Auto, si une étape était en cours, on exige une reprise consciente
IF WaitingResume THEN
    // 🔒 Sécurité anti-redémarrage automatique (AF-04 §4.1) : le cycle ne reprend
    // PAS sur manche + homme-mort seuls. Il reste en attente d'un geste conscient
    // (StartCycle) avant de restaurer l'étape mémorisée.
    IF StartEdge.Q THEN
        State := PausedState;   // reprise consciente : restaure l'étape mémorisée
        WaitingResume := FALSE;
    ELSE
        // Hold : neutralise toute demande de mouvement et affiche l'attente.
        WinchM1Cmd.RunRequest := FALSE; WinchM1Cmd.ReqAscent := FALSE; WinchM1Cmd.ReqDescend := FALSE; WinchM1Cmd.StepTgt := 0;
        WinchM2Cmd.RunRequest := FALSE; WinchM2Cmd.ReqAscent := FALSE; WinchM2Cmd.ReqDescend := FALSE; WinchM2Cmd.StepTgt := 0;
        TranslationCmd.ReqStart := FALSE; TranslationCmd.PositionTgt := 0;
        BucketCmd.ReqOpen := FALSE; BucketCmd.ReqClose := FALSE; BucketCmd.ReqKoboldMeasureEnable := FALSE;
        CycleStep := PausedState;
        CycleStateStr := 'Cycle en pause : reprise consciente requise (StartCycle)';
        Lifecycle.Busy := FALSE;
        Lifecycle.Done := FALSE;
        ExpectedAxis := E_OperatorAxis.BUTTON;
        ExpectedDirection := 0;
        WaitingForOperator := TRUE;
        WaitingForProcess := FALSE;
        OperatorActionId := 16#0400;
        OperatorAction := 'Reprendre le cycle : appuyer sur StartCycle (geste conscient).';
        RequestActive := FALSE;
    END_IF;
END_IF;

// 🧊 R9 — capture l'étape spécifique AVANT la bascule vers STABILIZING
IF ErrorEdge.Q THEN
    CycleStepAtError := State;
    State := E_CycleStep.STABILIZING;
    WaitingResume := FALSE;   // un défaut abroge la reprise consciente : repli STABILIZING
END_IF;

IF ResetEdge.Q THEN
    SavedState := E_CycleStep.X0_PREPARATION;
    IF State = E_CycleStep.STABILIZING THEN
        State := E_CycleStep.X0_PREPARATION;
    END_IF;
END_IF;

// Gestion de l'Abandon explicite
IF AbortEdge.Q THEN
    State := E_CycleStep.X0_PREPARATION;
    WaitingResume := FALSE;
    PausedState := E_CycleStep.X0_PREPARATION;
END_IF;

// SEMI_AUTO : FB_Cycle est le SEUL producteur des demandes benne. Les champs
// assistants (DiveSearch/ExtractionAssist) sont inactifs en cycle automatique
// -> valeurs neutres déterministes (identiques à la branche SEMI_AUTO de PRG_03).
BucketCmd.DescendPermitDiveBucketOpen := TRUE;
BucketCmd.ExtractionControlActive := FALSE;
BucketCmd.MinStepDown := 0;
{endregion}

{region "§3 Machine etat cycle automatique"}
// 🔁 2. LOGIQUE DU SÉQUENCEUR (Machine d'état) — CASE sur enum unique (R1)
// Pendant une reprise consciente (WaitingResume), le séquenceur est gelé :
// l'étape mémorisée ne s'exécute pas tant que StartCycle n'a pas été appuyé.
IF NOT WaitingResume THEN
CASE State OF

    E_CycleStep.X0_PREPARATION:
        CycleStep := E_CycleStep.X0_PREPARATION;
        CycleStateStr := 'X0 - Verification coherence systeme';
        WinchM1Cmd.RunRequest := FALSE; WinchM1Cmd.ReqAscent := FALSE; WinchM1Cmd.ReqDescend := FALSE; WinchM1Cmd.StepTgt := 0;
        WinchM2Cmd.RunRequest := FALSE; WinchM2Cmd.ReqAscent := FALSE; WinchM2Cmd.ReqDescend := FALSE; WinchM2Cmd.StepTgt := 0;
        TranslationCmd.ReqStart := FALSE; TranslationCmd.PositionTgt := 0;
        BucketCmd.ReqOpen := FALSE; BucketCmd.ReqClose := FALSE; BucketCmd.ReqKoboldMeasureEnable := FALSE;
        Lifecycle.Busy := FALSE;
        Lifecycle.Done := FALSE;
        ExpectedAxis := E_OperatorAxis.BUTTON;
        ExpectedDirection := 0;
        WaitingForOperator := TRUE;
        WaitingForProcess := FALSE;
        OperatorActionId := 16#0400;
        OperatorAction := 'Demander le demarrage du cycle (Bouton IHM ou armement Homme-Mort).';
        SampleCountDone := FALSE;   // F2: réarmement du front de comptage à chaque entrée en X0

        // Transition si système sain et réarmé (front reset ou démarrage).
        // 🔫 D2 : démarrage par bouton IHM (StartEdge) OU armement homme-mort joystick
        IF (StartEdge.Q OR DeadmanArmedEdge.Q) AND NOT Fault.Latched THEN
            WaitingResume := FALSE;
            State := E_CycleStep.X1_HOMING;
        END_IF;

    E_CycleStep.X1_HOMING:
        CycleStep := E_CycleStep.X1_HOMING;
        Lifecycle.Busy := TRUE;
        Lifecycle.Done := FALSE;
        ExpectedDirection := 0;

        // 🎯 Le referencement machine (datum M1/M2 + etat benne) est la responsabilite
        // EXCLUSIVE de FB_MachineHomingCycle. SEMI_AUTO n'est cense etre accessible que si
        // MachineHomed (gate FB_Modes). X1 ne fait donc QUE verifier la condition : elle
        // n'emet AUCUN ordre de mouvement et ne contient aucun referencement.
        IF HomedM1 AND HomedM2 THEN
            CycleStateStr := 'X1 - Codeurs references : position validee';
            ExpectedAxis := E_OperatorAxis.NONE;
            WaitingForOperator := FALSE;
            WaitingForProcess := TRUE;
            OperatorActionId := 0;
            OperatorAction := '';
            State := E_CycleStep.X2_WORK_POS_SELECT;   // franchie sans mouvement
        ELSE
            CycleStateStr := 'X1 - Machine non referencee';
            ExpectedAxis := E_OperatorAxis.BUTTON;
            WaitingForOperator := TRUE;
            WaitingForProcess := FALSE;
            OperatorActionId := 16#0401;
            OperatorAction := 'Machine non referencee : passer en maintenance pour le referencement.';
        END_IF;

    E_CycleStep.X2_WORK_POS_SELECT:
        CycleStep := E_CycleStep.X2_WORK_POS_SELECT;
        CycleStateStr := 'X2 - Mise en position P1 (translation)';
        Lifecycle.Busy := TRUE;
        Lifecycle.Done := FALSE;
        ExpectedAxis := E_OperatorAxis.JOYSTICK_X;
        // C3.1 : le cycle part TOUJOURS de P1 (plus de selection de cible). Le treuil est a
