    Target                  : REAL;               // --> [CMD] Valeur cible visee (-100.0..+100.0 %)
    AccelRate               : REAL;               // --> [CFG] Taux d'acceleration (%/s)

    Current                 : REAL;               // <-- [STAT] Valeur courante amortie (-100.0..+100.0 %)

// (robuste à une traversée de zéro rapide).

    IF Current > TargetLimited THEN Current := TargetLimited; END_IF;

// === DECLARATION (à coller dans la partie déclaration du POU FB_SpeedStep) ===

FUNCTION_BLOCK PUBLIC FB_SpeedStep
VAR_INPUT
    // === COMMANDES ET CONSIGNES ===
    Enable                          : BOOL;                           // --> [CMD] Pilotage actif (FALSE = palier 0 / arret)
    SpeedRefPct                     : REAL;                           // --> [CMD] Consigne vitesse 0..100%

    // === REGLAGES ET CONFIGURATION ===
    Table                           : ST_SpeedStepTable;              // --> [CFG] Table des 5 paliers propre au treuil
    HystMargin                      : REAL := 2.0;                    // --> [CFG] Marge anti-battement (%)
    MaxStepNumber                   : INT := 5;                       // --> [CFG] Plafond palier atteignable (1..5)

    // === MESURES ET GARDE-FOUS ===
    MeasuredSpeedBand               : INT := 0;                       // --> [HW] Bande vitesse reelle validee
    SpeedGuardEnable                : BOOL := FALSE;                  // --> [HW] Garde-fou palier activable
    SpeedGuardReady                 : BOOL := FALSE;                  // --> [HW] Conditions de passage validees
END_VAR
VAR_OUTPUT
    // === ETATS IEC ET SYNOPTIQUE ===
    StepNumber                      : INT;                            // <-- [STAT] Numero de palier courant (0..5)

    // === ORDRES ACTIONNEURS (CONTACTEURS) ===
    Contactor1                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 1
    Contactor2                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 2
    Contactor3                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 3
    Contactor4                      : BOOL;                           // <-- [ACT] Ordre contacteur vitesse palier 4

    // === DIAGNOSTICS ET ERREURS ===
    SpeedGuardLimited               : BOOL;                           // <-- [DIAG] Statut limitation par garde-fou
    ConfigError                     : BOOL;                           // <-- [DIAG] Synthese erreur configuration
    ConfigErrorId                   : WORD;                           // <-- [DIAG] Bitfield erreur configuration
END_VAR
VAR
    // === SOUS-INSTANCES FB ===
    Hyst1                           : HYSTERESIS;                     // * [INST] Transition palier 1 vers 2
    Hyst2                           : HYSTERESIS;                     // * [INST] Transition palier 2 vers 3
    Hyst3                           : HYSTERESIS;                     // * [INST] Transition palier 3 vers 4
    Hyst4                           : HYSTERESIS;                     // * [INST] Transition palier 4 vers 5
    _ConfigErrorTrig                : R_TRIG;                         // * [INST] Front montant erreur configuration

    // === VARIABLES LOCALES INTERNES ===
    SpeedRefInt                     : INT;                            // . [LOC] Consigne convertie en INT pour Hysteresis
    _ConfigValid                    : BOOL;                           // . [LOC] Resultat validation table et MaxStep
    _ConfigValidPrev                : BOOL;                           // . [LOC] Memoire cycle precedent validation
    _ConfigErrorIdInt               : WORD;                           // . [LOC] Accumulateur interne defauts config
    _MaxStepClamped                 : INT;                            // . [LOC] MaxStepNumber borne dans [1..5]
END_VAR

// === IMPLEMENTATION (à coller dans le corps ST du POU FB_SpeedStep) ===

{region "§0 Validation configuration"}
// ─────────────────────────────────────────────────────────────────────────
// 🛡️ 0. VALIDATION CONFIGURATION (P0.2) — exécutée CHAQUE cycle pour détecter
//    toute corruption IHM/PERSISTENT, même siEnable=FALSE (sorties sûres forcée)
// ─────────────────────────────────────────────────────────────────────────

// 🔧 Bornage systématique MaxStepNumber ∈ [1..5]
IF MaxStepNumber < 1 THEN
    _MaxStepClamped := 1;
ELSIF MaxStepNumber > 5 THEN
    _MaxStepClamped := 5;
ELSE
    _MaxStepClamped := MaxStepNumber;
END_IF;

// 🔍 Validation table ST_SpeedStepTable
// Critères (Partie 9 §2) :
// 1. Seuils croissants stricts : StepThreshold_Pct[1] < [2] < [3] < [4] < [5]
// 2. Seuils dans [0..100] %
// 3. HystMargin > 0
// 4. Pour chaque palier 2.._MaxStepClamped : au moins 1 contacteur TRUE (cohérence câblage)
//    Le palier 1 peut légitimement avoir les 4 contacteurs FALSE : résistances rotoriques
//    toutes insérées = vitesse minimale. On ne valide pas les paliers jamais atteignables.
_ConfigValid := TRUE;
_ConfigErrorIdInt := 16#0000;

// Critère 1 : seuils croissants
IF NOT (Table.StepThreshold_Pct[1] < Table.StepThreshold_Pct[2] AND
        Table.StepThreshold_Pct[2] < Table.StepThreshold_Pct[3] AND
        Table.StepThreshold_Pct[3] < Table.StepThreshold_Pct[4] AND
        Table.StepThreshold_Pct[4] < Table.StepThreshold_Pct[5]) THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;  // bit0
END_IF;

// Critère 2 : seuils dans [0..100]
IF (Table.StepThreshold_Pct[1] < 0.0) OR (Table.StepThreshold_Pct[5] > 100.0) THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
END_IF;

// Critère 3 : hystérésis > 0
IF HystMargin <= 0.0 THEN
    _ConfigValid := FALSE;
    _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
END_IF;

// Critère 4 : cohérence contacteurs par palier (seulement jusqu'au plafond actif)
IF _ConfigValid THEN
    // Palier 1 : les 4 sorties FALSE sont une configuration valide de vitesse minimale.
    // Palier 2
    IF _MaxStepClamped >= 2 THEN
        IF NOT (Table.P2R1 OR Table.P2R2 OR Table.P2R3 OR Table.P2R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 3
    IF _MaxStepClamped >= 3 THEN
        IF NOT (Table.P3R1 OR Table.P3R2 OR Table.P3R3 OR Table.P3R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 4
    IF _MaxStepClamped >= 4 THEN
        IF NOT (Table.P4R1 OR Table.P4R2 OR Table.P4R3 OR Table.P4R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
    // Palier 5
    IF _MaxStepClamped >= 5 THEN
        IF NOT (Table.P5R1 OR Table.P5R2 OR Table.P5R3 OR Table.P5R4) THEN
            _ConfigValid := FALSE;
            _ConfigErrorIdInt := _ConfigErrorIdInt OR 16#0001;
        END_IF;
    END_IF;
END_IF;

// 🛡️ Gestion erreur config : front montant + reset sur front Reset (cause disparue)
// Note : pas d'entrée Reset dédiée (brique réduite) → on réarme quand config redevient valide
//        (front descendant de _ConfigValid). Conforme Partie 3 §5 : cause disparue + front.
_ConfigErrorTrig(CLK := NOT _ConfigValid);  // Front montant = apparition erreur

// Mémoire erreur config : reste TRUE tant que config invalide OU pas de réarmement
// Réarmement = front montant de _ConfigValid (cause disparue) — détecté via _ConfigValidPrev
IF _ConfigErrorTrig.Q THEN
    // Nouvelle erreur : on la mémorise
    _ConfigValidPrev := FALSE;  // Force détection front montant au cycle suivant si redevient valide
ELSIF _ConfigValid AND NOT _ConfigValidPrev THEN
    // Config redevenue valide : front montant → on efface l'erreur (reset sur front)
    _ConfigValidPrev := TRUE;
ELSIF NOT _ConfigValid THEN
    _ConfigValidPrev := FALSE;
END_IF;

// Sorties erreur config (miroir standard Partie 3)
ConfigErrorId := _ConfigErrorIdInt;
ConfigError   := (ConfigErrorId <> 16#0000);
{endregion}

{region "§1 Gate"}
// ─────────────────────────────────────────────────────────────────────────
// 🛑 1. GATE — Enable FALSE OU config invalide = palier 0, sorties sûres
// ─────────────────────────────────────────────────────────────────────────
IF (NOT Enable) OR (NOT _ConfigValid) THEN
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    StepNumber := 0;
    SpeedGuardLimited := FALSE;
    RETURN;
END_IF;
{endregion}

{region "§2 Sélection palier"}
// ─────────────────────────────────────────────────────────────────────────
// 🪜 2. SÉLECTION DU PALIER (HYSTERESIS lib Util : IN/HIGH/LOW → OUT)
//    OUT = TRUE tant que IN < LOW (régime "bas") ; OUT bascule à FALSE une fois
//    IN > HIGH (régime "haut" confirmé) ; maintien entre LOW et HIGH (anti-battement).
//    => NOT Hyst_n.OUT = "on a franchi le seuil n vers le haut" (palier n+1 atteint).
// ─────────────────────────────────────────────────────────────────────────
SpeedRefInt := REAL_TO_INT(SpeedRefPct);

Hyst1(IN := SpeedRefInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[1] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[1] - HystMargin));
Hyst2(IN := SpeedRefInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[2] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[2] - HystMargin));
Hyst3(IN := SpeedRefInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[3] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[3] - HystMargin));
Hyst4(IN := SpeedRefInt, HIGH := REAL_TO_INT(Table.StepThreshold_Pct[4] + HystMargin), LOW := REAL_TO_INT(Table.StepThreshold_Pct[4] - HystMargin));

StepNumber := 1;                          // Palier mini dès qu'Enable est actif
IF NOT Hyst1.OUT THEN StepNumber := 2; END_IF;
IF NOT Hyst2.OUT THEN StepNumber := 3; END_IF;
IF NOT Hyst3.OUT THEN StepNumber := 4; END_IF;
IF NOT Hyst4.OUT THEN StepNumber := 5; END_IF;

// ─────────────────────────────────────────────────────────────────────────
// 2bis. PLAFOND — appliqué APRÈS la sélection hystérésis,
// AVANT la lecture table : le 0..100% de la consigne peut réclamer le palier 5,
// mais ne dépassera jamais _MaxStepClamped (ex. limitation couple en descente)
// ─────────────────────────────────────────────────────────────────────────
IF StepNumber > _MaxStepClamped THEN
    StepNumber := _MaxStepClamped;
END_IF;
{endregion}

{region "§2ter Garde-fou vitesse"}
// ─────────────────────────────────────────────────────────────────────────
// 🛡️ 2ter. GARDE-FOU VITESSE RÉELLE — limite uniquement la consigne de palier.
// Si la vitesse n'est pas validée stable, seul le palier 1 reste autorisé.
// ─────────────────────────────────────────────────────────────────────────
SpeedGuardLimited := FALSE;
IF SpeedGuardEnable THEN
    IF NOT SpeedGuardReady THEN
        IF StepNumber > 1 THEN
            StepNumber := 1;
            SpeedGuardLimited := TRUE;
        END_IF;
    ELSIF (MeasuredSpeedBand >= 1) AND (MeasuredSpeedBand <= 5) AND (StepNumber > MeasuredSpeedBand) THEN
        StepNumber := MeasuredSpeedBand;
        SpeedGuardLimited := TRUE;
    END_IF;
END_IF;
{endregion}

{region "§3 Contacteurs"}
// ─────────────────────────────────────────────────────────────────────────
// 📤 3. CONTACTEURS — lus dans la table (donnée, jamais codé en dur)
//    CASE couvre explicitement 0..5 : StepNumber=0 → tous FALSE (sécurité)
// ─────────────────────────────────────────────────────────────────────────
CASE StepNumber OF
    0:  // Arrêt / config invalide / Enable=FALSE — sécurité positive
        Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    1:  Contactor1 := Table.P1R1; Contactor2 := Table.P1R2; Contactor3 := Table.P1R3; Contactor4 := Table.P1R4;
    2:  Contactor1 := Table.P2R1; Contactor2 := Table.P2R2; Contactor3 := Table.P2R3; Contactor4 := Table.P2R4;
    3:  Contactor1 := Table.P3R1; Contactor2 := Table.P3R2; Contactor3 := Table.P3R3; Contactor4 := Table.P3R4;
    4:  Contactor1 := Table.P4R1; Contactor2 := Table.P4R2; Contactor3 := Table.P4R3; Contactor4 := Table.P4R4;
    5:  Contactor1 := Table.P5R1; Contactor2 := Table.P5R2; Contactor3 := Table.P5R3; Contactor4 := Table.P5R4;
ELSE
    // Défense en profondeur : valeur StepNumber inattendue → sécurité
    Contactor1 := FALSE; Contactor2 := FALSE; Contactor3 := FALSE; Contactor4 := FALSE;
    StepNumber := 0;
END_CASE;
{endregion}

FUNCTION_BLOCK PUBLIC FB_DriftGuard
VAR_INPUT
    // === COMMANDES ET SEUILS ===
    Arm                             : BOOL;                           // --> [CMD] Condition d'armement de la surveillance
    PositionM                       : REAL;                           // --> [HW] Position courante a surveiller (m)
    ToleranceM                      : REAL;                           // --> [CFG] Tolerance de derive maximale autorisee (m)
END_VAR
VAR_OUTPUT
    // === MESURES ET ETATS ===
    RefPosM                         : REAL;                           // <-- [DIAG] Position de reference capturee a l'armement (m)
    DriftM                          : REAL;                           // <-- [DIAG] Derive absolue courante (m)
    Violation                       : BOOL;                           // <-- [STAT] Alarme derive maximale depassee
END_VAR
VAR
    // === VARIABLES LOCALES INTERNES ===
    WasArmed                        : BOOL;                           // . [LOC] Memoire etat armement cycle precedent
END_VAR

// === IMPLEMENTATION (à coller dans le corps ST du POU FB_DriftGuard) ===

    IF NOT WasArmed THEN
        RefPosM := PositionM;   // Capture au 1er cycle armé uniquement
        WasArmed := TRUE;
    END_IF;
    DriftM := ABS(PositionM - RefPosM);
    Violation := DriftM > ToleranceM;
ELSE
    WasArmed := FALSE;
    DriftM := 0.0;
    Violation := FALSE;
END_IF;

    StepDelay(IN := FALSE, PT := T#1s250ms);

    ResetRequired := TRUE;
    NeutralRequestSeen := FALSE;
    RestartRequired := TRUE;

    BrakeTimeoutAck := TRUE;     // toujours effectif, sans condition externe
    ResetRequired := FALSE;

IF BrakeTimeoutCause OR NOT BrakeTimeoutAck THEN
    ErrorId := ErrorId OR 16#0001;
ELSE
    ErrorId := ErrorId AND 16#FFFE;
END_IF;

// détectée pendant que c'est armé → roue libre / frein insuffisant.
// La chaîne codeur produit la mesure ; ce bloc safety la surveille seulement.
DriftGuardA(
    Arm        := MeasuredSpeedValid AND FwdRevSpeedFeedbackOff AND NOT BrakeFeedback AND NOT InReferencingMode,
    PositionM  := CablePosM,
    ToleranceM := UncommandedDriftToleranceM
);
MecaADriftM := DriftGuardA.DriftM;   // Diagnostic IHM

// 🕐 La vitesse Meca A est une derive de position sur fenetre glissante 50ms : a l'arret, elle met
// ~50ms a retomber a 0 alors que l'armement (contacts/frein) est instantane -> faux Meca a chaque
// arret. Exiger une confirmation "vitesse < seuil" maintenue >= 1 fenetre (100ms) avant d'armer la
// detection de VITESSE Meca. Impact securite negligeable : une vraie roue libre dure >> 100ms.
TonMecaStopSettled(IN := (MeasuredSpeedMps < UncommandedSpeedThresholdMps), PT := T#100ms);

IF NOT (BypassSafety OR BypassMecaA) AND MeasuredSpeedValid AND (DriftGuardA.Violation OR (FwdRevSpeedFeedbackOff AND NOT BrakeFeedback AND NOT InReferencingMode AND TonMecaStopSettled.Q AND (MeasuredSpeedMps > UncommandedSpeedThresholdMps))) THEN
    Status.ErrorId := Status.ErrorId OR 16#0080;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FF7F;
END_IF;

// 🔍 3nonies. MÉCA B — PILOTAGE ACTIF SANS COMMANDE OPÉRATEUR — bit8
// Indépendant de FB_Winch (défense en profondeur). Perte CAN (bit0) OU joystick au neutre ;
// si les contacteurs OU LE FREIN ne confirment pas être retombés/serrés dans PostRampTimeout, défaut.
// Un frein qui ne confirme pas son engagement dans ce délai est tout aussi dangereux qu'un
// contacteur qui ne retombe pas (risque de mouvement/roue libre non détecté).
MecaB_NoOperatorCmd := (NOT JoystickOnline) OR (NOT JoystickOperational)
                       OR (NOT HeartbeatIhmOk) OR JoystickYNeutral;
TonMecaB(IN := NOT (BypassSafety OR BypassMecaB) AND MecaB_NoOperatorCmd AND NOT (FwdRevSpeedFeedbackOff AND NOT BrakeFeedback), PT := PostRampTimeout);
MecaBElapsedTime := TonMecaB.ET;   // Diagnostic IHM
IF TonMecaB.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#0100;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FEFF;
END_IF;

// 🔍 3decies. MÉCA C — GLISSEMENT M1 PENDANT BENNE, ESCALADE — bit9
// Armé UNIQUEMENT pour l'instance M1 (BenneHoldStillActive câblé sur instBucket.Busy côté M1
// seulement, toujours FALSE côté M2). Tolérance plus large que celle de FB_Bucket (1.0 m) —
// n'intervient que si la couche Benne (SafeStop M1 seul) n'a pas suffi.
// Dérive surveillée via FB_DriftGuard.
DriftGuardC(
    Arm        := NOT (BypassSafety OR BypassMecaC) AND BenneHoldStillActive,
    PositionM  := CablePosM,
    ToleranceM := BenneSlipToleranceM
);
MecaCDriftM := DriftGuardC.DriftM;   // Diagnostic IHM

IF DriftGuardC.Violation THEN
    Status.ErrorId := Status.ErrorId OR 16#0200;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FDFF;
END_IF;

// 🔍 3undecies. THERMIQUE FREIN COMMUN M1/M2/M3 — bit10
// Signal commun (1 seul fil, câblé identiquement sur les 3 instances Safety) : sa perte peut
// signifier qu'un frein colle instantanément (frein à manque de courant) → SafeStop ET
// PowerCutOff (voir bandeau en-tête).
IF NOT (BypassSafety OR BypassBrakeThermal) AND BrakeThermalFeedback THEN
    Status.ErrorId := Status.ErrorId OR 16#0400;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FBFF;
END_IF;

// 🔍 3duodecies. MÉCA D — ARRIVÉE CAPTEUR HAUT HORS RÉFÉRENCEMENT — bit11
// Complète le bit5 (AscentPermit immédiat) : arme la MÊME condition (capteur haut atteint, hors
// homing) mais avec un délai de confirmation, comme Méca B. Si les contacteurs ET le frein ne
// confirment pas être retombés/serrés dans PostRampTimeout après l'arrivée au capteur → défaut
// → SafeStop ET PowerCutOff (la coupure AscentPermit seule ne suffit pas à arrêter réellement
// le treuil : roue libre ou contacteur/frein défaillant).
TonMecaD(
    IN := NOT (BypassSafety OR BypassMecaD) AND ((((NOT TopPositionSensor AND NOT InReferencingMode) 
            OR (Homed AND NOT HomingSuspect AND (CablePosM >= (TopLimitM + 0.10))))
           AND (Direction >= 0)) // Neutralise la butée haute en descente (Direction = -1)
          AND NOT (FwdRevSpeedFeedbackOff AND NOT BrakeFeedback)), 
    PT := PostRampTimeout
);
IF TonMecaD.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#0800;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#F7FF;
END_IF;

// 🔍 3terdecies. MÉCA E — ÉCART SYNCHRO M1/M2 CRITIQUE — bits 12/13
// Couche INDÉPENDANTE de FB_WinchSync (écart mineur, 1er niveau) — défense en profondeur pour un
// écart VRAIMENT important. ExpectedOtherWinchPosM est DÉJÀ corrigé de l'offset benne actif par
// l'appelant : comparaison directe valable que le benne soit ouvert, fermé, ou absent du calcul.
// Suppression pendant BenneBusy (trajet actif = écart transitoire normal, pas un défaut),
// InReferencingMode (position pas fiable pendant homing) et OtherWinchInhibited (l'autre treuil
// en maintenance a une position figée, non comparable).
// Bit12 : détection → SafeStop immédiat. Bit13 : escalade si pas confirmé arrêté → PowerCutOff.
IF NOT (BypassProcess OR BypassMecaE) AND SyncEnable AND NOT BenneBusy AND NOT InReferencingMode AND NOT OtherWinchInhibited
   AND (ABS(CablePosM - ExpectedOtherWinchPosM) > CriticalSyncToleranceM) THEN
    Status.ErrorId := Status.ErrorId OR 16#1000;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#EFFF;
END_IF;

TonMecaE(IN := NOT (BypassSafety OR BypassMecaE) AND (((Status.ErrorId AND 16#1000) <> 16#0000) AND NOT (FwdRevSpeedFeedbackOff AND NOT BrakeFeedback)),
         PT := PostRampTimeout);
IF TonMecaE.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#2000;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#DFFF;
END_IF;

// 🔍 3quaterdecies — SENS RÉEL OPPOSÉ À LA COMMANDE (bit14)
// Le signe de la vitesse câble est comparé à la direction commandée uniquement pendant un
// mouvement effectivement demandé. La temporisation filtre bruit codeur et inversion transitoire.
OppositeDirectionActive := NOT BypassProcess AND MeasuredSpeedValid AND (MovementCommanded AND NOT InReferencingMode
    AND (((Direction > 0) AND (MeasuredSpeedSignedMps < -MovementSpeedThresholdMps))
      OR ((Direction < 0) AND (MeasuredSpeedSignedMps > MovementSpeedThresholdMps))));
TonOppositeDirection(IN := OppositeDirectionActive, PT := OppositeDirectionTimeout);
IF TonOppositeDirection.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#4000;
ELSIF ResetEdge.Q AND NOT OppositeDirectionActive THEN
    Status.ErrorId := Status.ErrorId AND 16#BFFF;
END_IF;

// ⏱️ 3quindecies — ABSENCE DE MOUVEMENT MALGRÉ COMMANDE (bit15)
// Détecte charge coincée, contacteur inefficace ou chaîne mécanique non entraînée. SafeStop seul :
// aucun retour ne justifie une coupure puissance brutale immédiate ; l'escalade reste matérielle.
TonNoMovement(
    IN := NOT BypassProcess AND MeasuredSpeedValid AND (MovementCommanded AND NOT InReferencingMode
          AND (ABS(MeasuredSpeedSignedMps) < MovementSpeedThresholdMps)),
    PT := NoMovementTimeout
);
IF TonNoMovement.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#8000;
ELSIF ResetEdge.Q AND (NOT MeasuredSpeedValid OR NOT MovementCommanded OR (ABS(MeasuredSpeedSignedMps) >= MovementSpeedThresholdMps)) THEN
    Status.ErrorId := Status.ErrorId AND 16#7FFF;
END_IF;

END_IF; // fin IF NOT BypassGlobal

{endregion}

{region "§5 Arbitrage sorties securite"}
// 📤 4. ARBITRAGE DES SORTIES MAÎTRESSES
Status.Error := (Status.ErrorId <> 16#0000);       // 🧭 Partie3 §4 : miroir de TOUT le domaine, y compris mou de câble (visible IHM)
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;

// 🟧 SafeStop : bits 0/1/2/3/4 (joystick, codeur, thermique moteur, MOU CÂBLE, phases) + 7/8/9
// (Méca A/B/C) + 10 (thermique frein commun) + 11 (Méca D) + 12/13 (Méca E, écart synchro critique) ou si contacteur coupé
IF NOT PowerContactorEngaged THEN
    SafeStop := TRUE;
ELSIF NOT SyncEnable THEN
    SafeStop := (Status.ErrorId AND 16#FF97) <> 16#0000;   // bit3 exclu → SafeStop normal sans mou câble
ELSE
    SafeStop := (Status.ErrorId AND 16#FF9F) <> 16#0000;   // bit3 inclus → SafeStop complet avec mou câble
END_IF;

// ⬇️ DescendPermit : TRUE = descente autorisée (pas de limite basse, pas de mou câble en mode récup, contacteur engagé)
// Polarité positive Fail-Safe (TRUE = mouvement autorisé).
DescendPermit := NOT (
    (((Status.ErrorId AND 16#0040) <> 16#0000 OR NOT PowerContactorEngaged) AND NOT (BypassGlobal OR BypassCableLimitSwitch))
    OR (((CfgCableLimitDescentM < 0.0) AND (CablePosM <= CfgCableLimitDescentM) AND NOT InReferencingMode) AND NOT BypassCableLimitSwitch)
    OR (LimitLegalReached AND NOT (BypassProcess OR BypassLimitLegal))
    OR (((Status.ErrorId AND 16#0008) <> 16#0000 AND NOT SyncEnable) AND NOT (BypassProcess OR BypassGlobal))
);

// ⬆️ AscentPermit : TRUE = montée autorisée (pas de fin course haut, pas de dépassement position)
// Polarité positive Fail-Safe (TRUE = mouvement autorisé).
AscentPermit := NOT (
    (((Status.ErrorId AND 16#0020) <> 16#0000
    OR (NOT TopPositionSensor AND NOT InReferencingMode))
    AND NOT BypassTopLimitSwitch)
    OR ((Homed AND NOT HomingSuspect AND NOT InReferencingMode AND (CablePosM >= TopLimitM)) AND NOT BypassTopLimitSoftware)
);

// PowerCutOff — bits 7/8/9 (Méca A/B/C, voir bandeau en-tête).
// Coupure puissance amont : les contacteurs locaux sont déjà confirmés coupés/insuffisants,
// SafeStop seul ne sert à rien dans ces 3 cas précis.
// bit2 (surchauffe/perte thermique MOTEUR) et bit10 (surchauffe/perte thermique FREIN commun) :
// un frein à manque de courant peut coller instantanément sur perte de son retour thermique ;
// une rampe SafeStop seule ne protège pas la mécanique dans ce cas, coupure puissance immédiate
// nécessaire. Même raisonnement appliqué au thermique moteur par cohérence/défense en profondeur.
// bit11 (Méca D, capteur haut) : même raisonnement que Méca A/B — contacteurs/frein déjà censés
// être coupés (AscentPermit immédiat sur capteur), une confirmation manquante après délai signifie
// que la coupure locale n'a pas suffi.
// bit13 (Méca E escalade) : PAS bit12 seul (détection immédiate = SafeStop uniquement ;
// PowerCutOff seulement si l'arrêt n'est pas confirmé après délai).
PowerCutOff := (Status.ErrorId AND 16#2F84) <> 16#0000;

// Busy, Done non utilisés mais présents pour conformité (comme FB_Diag_CanOpen)
Status.Busy := FALSE;
Status.Done := FALSE;

{endregion}

(* =======================================================================

   📄 Doc métier : DOC/AF/AF_Partie-10_Fonction_Winch_v2.0.md
   🔌 Interface : Brique de diagnostic/mesure (1 instance par treuil)
   🔒 Sécurité : Aucune commande moteur ni action sur SafeStop (information IHM)

FUNCTION_BLOCK PUBLIC FB_WinchLoadEstimator

VAR_INPUT
    // === COMMANDES ET CONSIGNES ===
    Enable                          : BOOL;                           // --> [CMD] Autorisation generale FB
    Reset                           : BOOL;                           // --> [CMD] Acquittement defaut sur front
    Mode                            : E_Mode;                         // --> [CMD] Mode d'exploitation actif
    ActiveSpeedStep                 : INT;                            // --> [CMD] Palier commande actif (0..5)

    // === SECOURS ET HARDWARE ===
    PowerContactorEngaged           : BOOL;                           // --> [SAFE] Chaine AU rearmee et puissance OK
    MeasuredSpeedMps                : REAL;                           // --> [HW] Vitesse cable absolue (m/s)
    MeasuredSpeedSignedMps          : REAL;                           // --> [HW] Vitesse signee (+: montee, -: descente) (m/s)

    // === REGLAGES ET TABLES ===
    SpeedConfig                     : ST_WinchSpeedConfig;            // --> [CFG] Configuration des plages de vitesse
    LoadTable                       : ST_WinchLoadEstimateTable;      // --> [CFG] Table empirique d'estimation 5x5
END_VAR
VAR_OUTPUT
    // === ETATS IEC ET SYNOPTIQUE ===
    Ready                           : BOOL;                           // <-- [STAT] FB pret a fonctionner
    // 🧩 Statut transverse (forme cible) — regroupe Busy/Done/Error/ErrorId/State/StateAtError.
    // ErrorId bit0: "entrée/table invalide".
    Status                          : ST_FbStatus;

    // === DIAGNOSTICS ET ESTIMATIONS ===
    SpeedBand                       : INT;                            // <-- [DIAG] Bande de vitesse courante (0..5)
    EstimatedLoadPct                : REAL;                           // <-- [DIAG] Estimation de charge (0..100%)
    Configured                      : BOOL;                           // <-- [STAT] Table explicitement validee
    Ascending                       : BOOL;                           // <-- [STAT] Montee effective mesuree en cours
END_VAR
VAR
    // === DETECTEURS ET TIMERS INTERNES ===
    ResetEdge                       : R_TRIG;                         // . [LOC] Detecteur front montant Reset
    ErrorEdge                       : R_TRIG;                         // . [LOC] Detecteur front montant Erreur

    // === VARIABLES LOCALES CALCULÉES ===
    PreviousSpeedBand               : INT;                            // . [LOC] Memoire bande de vitesse cycle precedent
    HasPreviousBand                 : BOOL;                           // . [LOC] Indicateur premiere bande memorisee
    SelectedLoadPct                 : REAL;                           // . [LOC] Valeur de charge extraite de la table
END_VAR

{region "§1 Gate"}
// 🛑 1. GATE — aucune estimation lorsque le bloc est désactivé ou l'AU absent.
IF NOT Enable OR NOT PowerContactorEngaged THEN
    Ready := FALSE;
    Status.Busy := FALSE;
    Status.Done := FALSE;
    Status.Error := FALSE;
    Status.ErrorId := 16#0000;
    Status.State := E_State.DISABLED;
    SpeedBand := 0;
    EstimatedLoadPct := 0.0;
    Configured := FALSE;
    Ascending := FALSE;
    HasPreviousBand := FALSE;
    RETURN;
END_IF;

Ready := TRUE;
Status.Busy := FALSE;
Status.Done := FALSE;
Status.State := E_State.READY;
ResetEdge(CLK := Reset);
Configured := LoadTable.IsConfigured;
// 📈 L'estimation de charge n'a de sens qu'en montée.
// La bande de vitesse reste calculée pour le diagnostic et le garde-fou de palier,
// mais EstimatedLoadPct est forcée à zéro à l'arrêt et en descente.
Ascending := (MeasuredSpeedSignedMps > 0.0);
{endregion}

{region "§2 Classement vitesse"}
// 📏 2. CLASSEMENT DE LA VITESSE — hystérésis autour des 4 frontières.
IF (MeasuredSpeedMps < 0.0) OR (SpeedConfig.MaxMeasuredSpeedMps <= 0.0) THEN
    Status.ErrorId := Status.ErrorId OR 16#0001;
    SpeedBand := 0;
ELSIF NOT HasPreviousBand THEN
    IF MeasuredSpeedMps <= SpeedConfig.SpeedBandMaxMps[1] THEN
        PreviousSpeedBand := 1;
    ELSIF MeasuredSpeedMps <= SpeedConfig.SpeedBandMaxMps[2] THEN
        PreviousSpeedBand := 2;
    ELSIF MeasuredSpeedMps <= SpeedConfig.SpeedBandMaxMps[3] THEN
        PreviousSpeedBand := 3;
    ELSIF MeasuredSpeedMps <= SpeedConfig.SpeedBandMaxMps[4] THEN
        PreviousSpeedBand := 4;
    ELSE
        PreviousSpeedBand := 5;
    END_IF;
    HasPreviousBand := TRUE;
    SpeedBand := PreviousSpeedBand;
ELSE
    // 🔒 Valider l'état mémorisé avant toute indexation dynamique.
    IF PreviousSpeedBand < 1 OR PreviousSpeedBand > 5 THEN
        PreviousSpeedBand := 1;
        Status.ErrorId := Status.ErrorId OR 16#0001;
    END_IF;

    // CODESYS ne garantit pas le court-circuit des expressions AND :
    // chaque borne est donc testée dans une branche séparée avant accès.
    IF PreviousSpeedBand < 5 THEN
        IF MeasuredSpeedMps > (SpeedConfig.SpeedBandMaxMps[PreviousSpeedBand]
                               + SpeedConfig.SpeedBandHysteresisMps) THEN
            PreviousSpeedBand := PreviousSpeedBand + 1;
        END_IF;
    END_IF;
    IF PreviousSpeedBand > 1 THEN
        IF MeasuredSpeedMps < (SpeedConfig.SpeedBandMaxMps[PreviousSpeedBand - 1]
                               - SpeedConfig.SpeedBandHysteresisMps) THEN
            PreviousSpeedBand := PreviousSpeedBand - 1;
        END_IF;
    END_IF;
    SpeedBand := PreviousSpeedBand;
END_IF;
{endregion}

{region "§3 Lecture table"}
// 📊 3. LECTURE TABLE — bornes séparées avant l'accès au tableau 5x5.
EstimatedLoadPct := 0.0;
IF Configured THEN
    IF Ascending THEN
        IF ActiveSpeedStep >= 1 THEN
            IF ActiveSpeedStep <= 5 THEN
                IF SpeedBand >= 1 THEN
                    IF SpeedBand <= 5 THEN
                        SelectedLoadPct := LoadTable.LoadPctByStepAndSpeedBand[ActiveSpeedStep][SpeedBand];
                        IF (SelectedLoadPct < 0.0) OR (SelectedLoadPct > 100.0) THEN
                            Status.ErrorId := Status.ErrorId OR 16#0001;
                        ELSE
                            EstimatedLoadPct := SelectedLoadPct;
                        END_IF;
                    END_IF;
                END_IF;
            END_IF;
        END_IF;
    END_IF;
END_IF;
{endregion}

{region "§4 Sorties et acquittement"}
// 📤 4. SORTIES D'ÉTAT ET ACQUITTEMENT.
IF ResetEdge.Q AND (Status.ErrorId <> 16#0000) AND (MeasuredSpeedMps >= 0.0) THEN
    Status.ErrorId := 16#0000;
END_IF;
Status.Error := (Status.ErrorId <> 16#0000);
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;
{endregion}

(* =======================================================================

   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Observateur de symétrie et de synchronisme des treuils M1/M2

FUNCTION_BLOCK FB_Winch_Symmetry
VAR_INPUT
    // === COMMANDES ET CONSIGNES ===

    M1CommandActive                 : BOOL;                           // --> [CMD] Commande active sur treuil M1
    M2CommandActive                 : BOOL;                           // --> [CMD] Commande active sur treuil M2
    M1Direction                     : INT;                            // --> [CMD] Sens commande M1 (-1/0/+1)
    M2Direction                     : INT;                            // --> [CMD] Sens commande M2 (-1/0/+1)

    // === FEEDBACKS HARWARE ET FREINS ===
    M1BrakeCmd                      : BOOL;                           // --> [HW] Ordre ouverture frein M1
    M2BrakeCmd                      : BOOL;                           // --> [HW] Ordre ouverture frein M2
    M1BrakeApplied                  : BOOL;                           // --> [HW] Etat frein M1 serre
    M2BrakeApplied                  : BOOL;                           // --> [HW] Etat frein M2 serre

    // === MESURES POSITIONS ET VITESSES ===
    M1Position_M                    : REAL;                           // --> [HW] Position mesuree M1 (m)
    M2Position_M                    : REAL;                           // --> [HW] Position mesuree M2 (m)
    M1Speed_Mps                     : REAL;                           // --> [HW] Vitesse mesuree M1 (m/s)
    M2Speed_Mps                     : REAL;                           // --> [HW] Vitesse mesuree M2 (m/s)

    // === CONFIGURATION ===

END_VAR
VAR_OUTPUT
    // === ETATS IEC ET SYNOPTIQUE ===
    SymmetryOk                      : BOOL;                           // <-- [STAT] Symetrie globale conforme aux criteres
    SymmetryValid                   : BOOL;                           // <-- [STAT] Mesure de symetrie valide et calculee
END_VAR
VAR_IN_OUT
    // === STRUCTURES DE DONNÉES PARTAGÉES ===
    Data                            : ST_Winch_SymmetryData;          // <-> [BUS] Donnees de mesures de symetrie
END_VAR
VAR
    // === SOUS-INSTANCES FB ===
    ResetEdge                       : R_TRIG;                         // * [INST] Detecteur front reinitialisation
    CycleTime                       : FB_CycleTime;                   // * [INST] Calcul du temps de cycle

    // === TIMERS INTERNES ===
    CommandTon                      : TON;                            // . [LOC] Tempo qualification commande stable
    MotionTimer                     : TON;                            // . [LOC] Chronometre de mouvement
    StopTimer                       : TON;                            // . [LOC] Chronometre d'arret

    // === VARIABLES LOCALES CALCULÉES ===
    Comparable                      : BOOL;                           // . [LOC] Conditions de comparaison remplies
    Qualified                       : BOOL;                           // . [LOC] Phase de mesure qualifiee
    PreviousQualified               : BOOL;                           // . [LOC] Etat qualifie cycle precedent
    Stopping                        : BOOL;                           // . [LOC] Phase d'arret en cours
    StartCapturedM1                 : BOOL;                           // . [LOC] Top depart capture M1
    StartCapturedM2                 : BOOL;                           // . [LOC] Top depart capture M2
    StopCapturedM1                  : BOOL;                           // . [LOC] Top arret capture M1
    StopCapturedM2                  : BOOL;                           // . [LOC] Top arret capture M2
    StartMsM1                       : UDINT;                          // . [LOC] Horodatage d'ouverture M1 (ms)
    StartMsM2                       : UDINT;                          // . [LOC] Horodatage d'ouverture M2 (ms)
    ReleaseMsM1                     : UDINT;                          // . [LOC] Temps de desserrage frein M1 (ms)

    ApplyMsM2                       : UDINT;                          // . [LOC] Temps de serrage frein M2 (ms)

    StopMsM2                        : UDINT;                          // . [LOC] Horodatage d'arret M2 (ms)

    StopStartM2_M                   : REAL;                           // . [LOC] Position debut d'arret M2 (m)

CommandTon(IN := Comparable, PT := UDINT_TO_TIME(Config.MinimumCommandTime_Ms));
Qualified := CommandTon.Q AND (M1Speed_Mps >= Config.MinimumSpeed_Mps) AND (M2Speed_Mps >= Config.MinimumSpeed_Mps);

IF Comparable AND (M1Speed_Mps >= Config.MinimumSpeed_Mps) AND NOT StartCapturedM1 THEN StartMsM1:=TIME_TO_UDINT(MotionTimer.ET); StartCapturedM1:=TRUE; END_IF;
IF Comparable AND (M2Speed_Mps >= Config.MinimumSpeed_Mps) AND NOT StartCapturedM2 THEN StartMsM2:=TIME_TO_UDINT(MotionTimer.ET); StartCapturedM2:=TRUE; END_IF;

    M1SlipDetected := FALSE;
    // Les requêtes internes sont réinitialisées
    CloseReq := FALSE;
    OpenReq := FALSE;
    RETURN;
END_IF;

// 🟢 ÉTAT INITIAL READY SI DISABLED PRÉCÉDEMMENT
IF Status.State = E_State.DISABLED THEN
    Status.State := E_State.READY;
END_IF;

{endregion}

{region "§2 Reset et bypass"}
// 🔑 2. GESTION DU RESET (FRONT MONTANT)
ResetEdge(CLK := Reset);

// 🌐 BYPASS GLOBAL : force ErrorId = 0 (ignore toutes les erreurs benne)
IF BypassGlobal THEN
    Status.ErrorId := 16#0000;
END_IF;

IF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FFFA; // Efface les bits 0 et 2 (Timeout, Limites)
END_IF;

{endregion}

{region "§3 Coherence et etat mecanique benne"}
// 📏 3. DÉTECTION D'INCOHÉRENCE AU BOOT (PREMIER CYCLE API)
IF FirstCycle THEN
    FirstCycle := FALSE;
    IF Config.CoherenceLimitM > 0.01 THEN
        IF BucketState.IsOpen AND NOT BucketState.IsClosed THEN
            IF ABS(CablePosM2 - BucketState.LastPosM2Open) > Config.CoherenceLimitM THEN
                BucketState.StateIncoherent := TRUE;
            END_IF;
        ELSIF BucketState.IsClosed AND NOT BucketState.IsOpen THEN
            IF ABS(CablePosM2 - BucketState.LastPosM2Close) > Config.CoherenceLimitM THEN
                BucketState.StateIncoherent := TRUE;
            END_IF;
        ELSE
            // Si l'état n'a jamais été initialisé (les deux FALSE/TRUE), on signale une incohérence
            BucketState.StateIncoherent := TRUE;
        END_IF;
    ELSE
        // Limite non configurée ou invalide, par précaution on force l'incohérence
        BucketState.StateIncoherent := TRUE;
    END_IF;
END_IF;

// Met à jour le bit1 d'erreur si incohérence détectée
IF BucketState.StateIncoherent THEN
    Status.ErrorId := Status.ErrorId OR 16#0002;
    // Force l'état inconnu (ni ouvert ni fermé) si incohérence active
    BucketState.IsOpen := FALSE;
    BucketState.IsClosed := FALSE;
ELSE
    Status.ErrorId := Status.ErrorId AND 16#FFFD;
END_IF;

// 🧲 3bis. GARDE-FOU CODEURS NON RÉFÉRENCÉS (bit 3)
// CablePosM1/CablePosM2 ne sont fiables que si le homing (Partie10) a été fait sur les 2 treuils.
// Continu (pas de front Reset) : se lève/efface tout seul selon l'état réel du homing.
IF NOT HomedM1 OR NOT HomedM2 THEN
    Status.ErrorId := Status.ErrorId OR 16#0008;
ELSE
    Status.ErrorId := Status.ErrorId AND 16#FFF7;
END_IF;

// 3bis-bis. CONFIRMATION MANUELLE POSITION OUVERTE — voir bandeau
// Action de CONFIRMATION (aucun mouvement, ne pilote pas M2) : acquitte les défauts et réinitialise
// les demandes. L'état IsOpen/IsClosed reste STRICTEMENT évalué en continu (§3bis-quater) selon la position codeur réelle.
ConfirmOpenEdge(CLK := ConfirmOpenPosition);
IF ConfirmOpenEdge.Q
   AND (Mode = E_Mode.MAINT_N1 OR Mode = E_Mode.MAINT_N2)
   AND NOT M1_Busy AND NOT M2_Busy THEN
    BucketState.LastPosM2Open := CablePosM2;
    BucketState.LastPosM2Close := CablePosM2 - Config.OffsetOpenM + Config.OffsetCloseM;
    Status.ErrorId := 16#0000;      // Réinitialisation complète des défauts benne lors du référencement manuel
    CloseReq := FALSE;              // Reset des demandes en attente pour éviter les démarrages intempestifs
    OpenReq := FALSE;
END_IF;

// 3bis-ter. CONFIRMATION MANUELLE POSITION FERMÉE
// Même principe que ConfirmOpenPosition pour acquitter et réinitialiser.
ConfirmCloseEdge(CLK := ConfirmClosePosition);
IF ConfirmCloseEdge.Q
   AND (Mode = E_Mode.MAINT_N1 OR Mode = E_Mode.MAINT_N2)
   AND NOT M1_Busy AND NOT M2_Busy THEN
    BucketState.LastPosM2Close := CablePosM2;
    BucketState.LastPosM2Open := CablePosM2 - Config.OffsetCloseM + Config.OffsetOpenM;
    Status.ErrorId := 16#0000;      // Réinitialisation complète des défauts benne lors du référencement manuel
    CloseReq := FALSE;              // Reset des demandes en attente pour éviter les démarrages intempestifs
    OpenReq := FALSE;
END_IF;

// 3bis-quater. DÉTECTION AUTOMATIQUE EN CONTINU (5 ZONES CLAIRES OPÉRATEUR) :
// Recalcule IsOpen, IsClosed, TooOpen, TooClosed, IsIntermediate en PERMANENCE à partir de
// l'écart réel Delta = CablePosM2 - CablePosM1 et de la tolérance Config.CoherenceLimitM.
// Actif à l'arrêt (NOT M1_Busy/M2_Busy), hors séquence active (State <> BUSY), codeurs référencés.
IF (Status.State <> E_State.BUSY) AND NOT M1_Busy AND NOT M2_Busy AND HomedM1 AND HomedM2
   AND (Config.CoherenceLimitM > 0.01) THEN

    // 1. Zone Benne Ouverte (Delta proche de OffsetOpenM)
    IF ABS((CablePosM2 - CablePosM1) - Config.OffsetOpenM) <= Config.CoherenceLimitM THEN
        BucketState.IsOpen := TRUE;
        BucketState.IsClosed := FALSE;
        BucketState.TooOpen := FALSE;
        BucketState.TooClosed := FALSE;
        BucketState.IsIntermediate := FALSE;
        BucketState.LastPosM2Open := CablePosM2;
        BucketState.StateIncoherent := FALSE;

    // 2. Zone Benne Fermée (Delta proche de OffsetCloseM)
    ELSIF ABS((CablePosM2 - CablePosM1) - Config.OffsetCloseM) <= Config.CoherenceLimitM THEN
        BucketState.IsClosed := TRUE;
        BucketState.IsOpen := FALSE;
        BucketState.TooOpen := FALSE;
        BucketState.TooClosed := FALSE;
        BucketState.IsIntermediate := FALSE;
        BucketState.LastPosM2Close := CablePosM2;
        BucketState.StateIncoherent := FALSE;

    // 3. Zone Trop Ouvert (Dépassement en ouverture : Delta < OffsetOpenM - Tolerance)
    ELSIF (CablePosM2 - CablePosM1) < (Config.OffsetOpenM - Config.CoherenceLimitM) THEN
        BucketState.IsOpen := FALSE;
        BucketState.IsClosed := FALSE;
        BucketState.TooOpen := TRUE;
        BucketState.TooClosed := FALSE;
        BucketState.IsIntermediate := FALSE;
        BucketState.StateIncoherent := TRUE;

    // 4. Zone Trop Fermé (Dépassement en fermeture : Delta > OffsetCloseM + Tolerance)
    ELSIF (CablePosM2 - CablePosM1) > (Config.OffsetCloseM + Config.CoherenceLimitM) THEN
        BucketState.IsOpen := FALSE;
        BucketState.IsClosed := FALSE;
        BucketState.TooOpen := FALSE;
        BucketState.TooClosed := TRUE;
        BucketState.IsIntermediate := FALSE;
        BucketState.StateIncoherent := TRUE;

    // 5. Zone Intermédiaire / Entrebâillée (entre OffsetOpenM + Tol et OffsetCloseM - Tol)
    ELSE
        BucketState.IsOpen := FALSE;
        BucketState.IsClosed := FALSE;
        BucketState.TooOpen := FALSE;
        BucketState.TooClosed := FALSE;
        BucketState.IsIntermediate := TRUE;
        BucketState.StateIncoherent := FALSE;
    END_IF;
END_IF;

// 🔍 3ter. MÉCA C COUCHE 1 — GLISSEMENT M1 PENDANT BUSY (bit4)
// M1 doit rester immobile pendant que M2 bouge seul. Référence prise à l'entrée en Busy.
BusyEdge(CLK := (Status.State = E_State.BUSY));
IF BusyEdge.Q THEN
    M1RefPosM := CablePosM1;
    M2StartPosM := CablePosM2;   // Borne le recul autorisé (voir §6 CASE BUSY)
    // Mémorise l'état d'origine (restauré si recul complet jusqu'à M2StartPosM, voir §6 CASE BUSY)
    // puis passe en position transitoire dès le démarrage du mouvement — ni ouvert ni fermé pendant
    // le trajet, quel que soit l'état mémorisé avant.
    WasOpenAtStart := BucketState.IsOpen;
    WasClosedAtStart := BucketState.IsClosed;
    LeftStartSinceArm := FALSE;
    BucketState.IsOpen := FALSE;
    BucketState.IsClosed := FALSE;
END_IF;

IF (Status.State = E_State.BUSY) AND (ABS(CablePosM1 - M1RefPosM) > M1SlipToleranceM) AND NOT InhibitM1 THEN
    Status.ErrorId := Status.ErrorId OR 16#0010;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FFEF;
END_IF;
M1SlipDetected := (Status.ErrorId AND 16#0010) <> 16#0000;

{endregion}

{region "§4 Surveillance limites timeout et arret"}
// 🚨 4. SURVEILLANCE DES LIMITES DE PLACEMENT (Limites dépassées - bit 2)
// On s'assure que l'écart réel M1/M2 ne dépasse pas les offsets de configuration + marge de sécurité de 1m
MaxAllowedOffset := MAX(ABS(Config.OffsetOpenM), ABS(Config.OffsetCloseM)) + 1.0;
IF (ABS(CablePosM1 - CablePosM2) > MaxAllowedOffset) AND NOT InhibitM1 AND NOT InhibitM2 THEN
    Status.ErrorId := Status.ErrorId OR 16#0004;
ELSIF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FFFB;
END_IF;

// ⏲️ 5. SURVEILLANCE TIMEOUT MOUVEMENT (bit 0)
// Déclenche si le mouvement actif (M2_StartStop = TRUE) dure plus longtemps que le timeout
TonTimeout(IN := (Status.State = E_State.BUSY AND M2_StartStop), PT := CfgTimeoutDuration);
IF TonTimeout.Q THEN
    Status.ErrorId := Status.ErrorId OR 16#0001;
END_IF;

// ❌ Évaluation des erreurs graves (Timeout, Limites dépassées, Codeur(s) non référencé(s), ou Glissement M1)
SevereError := (Status.ErrorId AND 16#001D) <> 16#0000;

IF SevereError THEN
    IF (Status.State = E_State.BUSY) OR CloseReq OR OpenReq THEN
        BucketState.StateIncoherent := TRUE;
    END_IF;
    M2_StartStop := FALSE;
    M2_Direction := 0;
    M2_ForceSlowSpeed := FALSE;
    CloseReq := FALSE;
    OpenReq := FALSE;
    // On repasse en READY et attend un Reset
    Status.State := E_State.READY;
END_IF;

{endregion}

{region "§5 Machine etat commande"}
// 🎛️ 6. MACHINE À ÉTATS DE COMMANDE
CASE Status.State OF
    E_State.READY:
        Ready := TRUE;
        Status.Busy := FALSE;
        Status.Done := FALSE;
        M2_StartStop := FALSE;
        M2_Direction := 0;
        M2_ForceSlowSpeed := FALSE;
        RemainingTravelM := 0.0; // Pas de mouvement en cours

        // Demande benne acceptée UNIQUEMENT si M1 ET M2 sont à l'arrêt (tous
        // modes) — évite qu'un mouvement en cours sur l'un des deux treuils "traverse" l'armement
        // du benne. Ni mémorisation de la requête, ni démarrage tant que ce
        // n'est pas vrai — l'opérateur doit refaire la demande une fois les treuils arrêtés.
        IF NOT M1_Busy AND NOT M2_Busy THEN
            // Mémorisation des requêtes IHM (l'une annule l'autre)
            IF CmdClose_IHM AND NOT SevereError THEN
                CloseReq := TRUE;
                OpenReq := FALSE;
            ELSIF CmdOpen_IHM AND NOT SevereError THEN
                OpenReq := TRUE;
                CloseReq := FALSE;
            END_IF;

            // Condition d'activation du mouvement (Requête IHM + Validation Joystick dans le bon sens)
            // Sens inversés (enroulage ferme, déroulage ouvre)
            IF CloseReq AND MotionRequestActive AND (MotionDirection = 1) THEN
                Status.State := E_State.BUSY;
            ELSIF OpenReq AND MotionRequestActive AND (MotionDirection = -1) THEN
                Status.State := E_State.BUSY;
            END_IF;
        END_IF;

    E_State.BUSY:
        Ready := TRUE;
        Status.Busy := TRUE;
        Status.Done := FALSE;
        M2_ForceSlowSpeed := TRUE; // Vitesse lente forcée sans contacteurs

        // Distance restante avant cible, pour affichage IHM (aiguille/jauge)
        // Toujours positive, décroît vers 0 à mesure que M2 approche la cible.
        IF CloseReq THEN
            RemainingTravelM := MAX(0.0, (CablePosM1 + Config.OffsetCloseM) - CablePosM2);
        ELSIF OpenReq THEN
            RemainingTravelM := MAX(0.0, CablePosM2 - (CablePosM1 + Config.OffsetOpenM));
        ELSE
            RemainingTravelM := 0.0;
        END_IF;

        // Interlock Homme-mort joystick
        // Sens inverse autorisé (recul/annulation partielle) pendant CloseReq/OpenReq, mais BORNÉ
        // à la position de départ (M2StartPosM, mémorisée à l'entrée en Busy) — ne permet jamais de
        // dépasser cette position dans le sens opposé au sens principal.
        IF NOT MotionRequestActive THEN
            // Si relâchement de l'intention opérateur, arrêt temporaire du mouvement.
            M2_StartStop := FALSE;
            M2_Direction := 0;
        ELSIF CloseReq THEN
            IF (MotionDirection = 1) THEN
                // Sens principal (fermeture) : M2 enroule (monte)
                M2_StartStop := TRUE;
                M2_Direction := 1;
            ELSIF (MotionDirection = -1) AND (CablePosM2 > M2StartPosM) THEN
                // Sens inverse : recul autorisé UNIQUEMENT tant qu'on n'est pas revenu à la position
                // de départ (pas au-delà, donc pas de ré-ouverture complète depuis ici)
                M2_StartStop := TRUE;
                M2_Direction := -1;
            ELSE
                // Mauvais sens, ou position de départ déjà atteinte en reculant : arrêt
                M2_StartStop := FALSE;
                M2_Direction := 0;
            END_IF;
        ELSIF OpenReq THEN
            IF (MotionDirection = -1) THEN
                // Sens principal (ouverture) : M2 déroule (descend)
                M2_StartStop := TRUE;
                M2_Direction := -1;
            ELSIF (MotionDirection = 1) AND (CablePosM2 < M2StartPosM) THEN
                // Sens inverse : recul autorisé UNIQUEMENT tant qu'on n'est pas revenu à la position
                // de départ (pas au-delà, donc pas de re-fermeture complète depuis ici)
                M2_StartStop := TRUE;
                M2_Direction := 1;
            ELSE
                M2_StartStop := FALSE;
                M2_Direction := 0;
            END_IF;
        ELSE
            M2_StartStop := FALSE;
            M2_Direction := 0;
        END_IF;

        // Détecte qu'on a réellement quitté la position de départ (évite que le test de retour
        // ci-dessous ne se déclenche instantanément au 1er cycle de Busy, où CablePosM2 =
        // M2StartPosM par définition).
        IF (CloseReq AND (CablePosM2 > M2StartPosM)) OR (OpenReq AND (CablePosM2 < M2StartPosM)) THEN
            LeftStartSinceArm := TRUE;
        END_IF;

        // Condition d'arrêt automatique de la fermeture
        IF CloseReq THEN
            // Compense l'inertie mécanique (coast-down) : coupe la commande
            // CloseAnticipationM avant OffsetCloseM pour amener l'arrêt réel près du
            // vrai offset au lieu de le dépasser.
            IF CablePosM2 >= (CablePosM1 + Config.OffsetCloseM - Config.CloseAnticipationM) THEN
                M2_StartStop := FALSE;
                M2_Direction := 0;

                // Mémorisation de l'état physique fermé
                BucketState.IsOpen := FALSE;
                BucketState.IsClosed := TRUE;
                BucketState.LastPosM2Close := CablePosM2;
                BucketState.StateIncoherent := FALSE; // recalage réussi

                CloseReq := FALSE;
                Status.State := E_State.DONE;
            // Recul complet jusqu'à la position de départ — restaure l'état d'origine
            // (physiquement, on est revenu là où on a commencé)
            ELSIF LeftStartSinceArm AND (CablePosM2 <= M2StartPosM) THEN
                M2_StartStop := FALSE;
                M2_Direction := 0;

                BucketState.IsOpen := WasOpenAtStart;
                BucketState.IsClosed := WasClosedAtStart;

                CloseReq := FALSE;
                Status.State := E_State.DONE;
            END_IF;
        END_IF;

        // Condition d'arrêt automatique de l'ouverture
        IF OpenReq THEN
            // Même logique que la fermeture ci-dessus, inertie différente en ouverture
            // (OpenAnticipationM distinct).
            IF CablePosM2 <= (CablePosM1 + Config.OffsetOpenM + Config.OpenAnticipationM) THEN
                M2_StartStop := FALSE;
                M2_Direction := 0;

                // Mémorisation de l'état physique ouvert
                BucketState.IsOpen := TRUE;
                BucketState.IsClosed := FALSE;
                BucketState.LastPosM2Open := CablePosM2;
                BucketState.StateIncoherent := FALSE; // recalage réussi

                OpenReq := FALSE;
                Status.State := E_State.DONE;
            // Recul complet jusqu'à la position de départ — restaure l'état d'origine
            // (physiquement, on est revenu là où on a commencé)
            ELSIF LeftStartSinceArm AND (CablePosM2 >= M2StartPosM) THEN
                M2_StartStop := FALSE;
                M2_Direction := 0;

                BucketState.IsOpen := WasOpenAtStart;
                BucketState.IsClosed := WasClosedAtStart;

                OpenReq := FALSE;
                Status.State := E_State.DONE;
            END_IF;
        END_IF;

    E_State.DONE:
        Ready := TRUE;
        Status.Busy := FALSE;
        Status.Done := TRUE;
        M2_StartStop := FALSE;
        M2_Direction := 0;
        M2_ForceSlowSpeed := FALSE;
        RemainingTravelM := 0.0; // Cible atteinte

        // Attente du relâchement de la validation joystick pour repasser à READY
        IF NOT MotionRequestActive THEN
            Status.State := E_State.READY;
        END_IF;
END_CASE;

// 📤 7. GESTION DES SORTIES D'ÉTAT GLOBALES
Status.Error := (Status.ErrorId <> 16#0000);
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;

// Calcul continu de l'écart réel signé M2 - M1 pour l'affichage IHM et l'aide au pilotage
DeltaPosition_M := CablePosM2 - CablePosM1;

// Détermination de l'offset actif pour le synchronisme de FB_WinchSync ET pour l'affichage IHM
// (M2PositionCorrected, côté PRG_SUPERVISION_CFC).
// CloseReq/OpenReq sont pris en compte (pas seulement IsClosed/IsOpen, qui sont FALSE pendant
// TOUT le trajet depuis l'état transitoire) pour que la position corrigée de M2 converge
// progressivement vers M1 pendant l'enroulage.
IF Status.State = E_State.BUSY THEN
    // Pendant le déplacement, l'offset actif suit la différence réelle de câble (M2 - M1) afin que
    // la position corrigée de M2 reste stable (égale à M1, cad la hauteur du benne) sans à-coup,
    // tandis que le câble brut de M2 montre son enroulement.
    ActiveOffsetM := CablePosM2 - CablePosM1;
    HoldOffsetM := ActiveOffsetM;   // Mémorise l'écart réel pour le gel en position intermédiaire
ELSIF CloseReq OR BucketState.IsClosed THEN
    ActiveOffsetM := Config.OffsetCloseM;
    HoldOffsetM := ActiveOffsetM;
ELSIF OpenReq OR BucketState.IsOpen THEN
    ActiveOffsetM := Config.OffsetOpenM;
    HoldOffsetM := ActiveOffsetM;
ELSE
    // Position intermédiaire non référencée : gèle l'écart réel mémorisé (HoldOffsetM) au lieu de
    // sauter vers la config. Évite l'à-coup sur la position corrigée M2 / la synchro.
    ActiveOffsetM := HoldOffsetM;
END_IF;

{endregion}
