   📊 FB_Acquisition_Preflight — Verdict passif machine arrêtée
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Verdict passif de qualification E/S machine arrêtée (16 contrôles)
   🔒 Polarité : Observateur pur (aucun SafeStop, PowerCutOff ou ordre machine)
   📄 Doc métier : DOC/AF/AF_Partie-06_Acquisition_Qualification_IO_v2.4.md §6
   ======================================================================= *)
FUNCTION_BLOCK FB_Acquisition_Preflight
VAR_INPUT
    // === COMMANDES & ETATS MACHINE ===
    Execute                 : BOOL; // --> [CMD] Lancement du verdict (front montant)
    MachineIsStill          : BOOL; // --> [HW]  Machine a l'arret complet (vitesse nulle)

    // === ETATS FREINS & CONTACTEURS ===
    M1BrakeApplied          : BOOL; // --> [HW]  Frein M1 serre
    M2BrakeApplied          : BOOL; // --> [HW]  Frein M2 serre
    M3BrakeApplied          : BOOL; // --> [HW]  Frein M3 serre
    M1ContactorsReleased    : BOOL; // --> [HW]  Contacteurs M1 retombes
    M2ContactorsReleased    : BOOL; // --> [HW]  Contacteurs M2 retombes

    // === PROTECTION THERMIQUE & RESEAU ===
    M1ThermalOk             : BOOL; // --> [HW]  Thermique moteur M1 OK
    M2ThermalOk             : BOOL; // --> [HW]  Thermique moteur M2 OK
    BrakeThermalOk          : BOOL; // --> [HW]  Thermique freins OK
    PhaseRotationOk         : BOOL; // --> [HW]  Ordre des phases reseau OK

    // === CAPTEURS & CHAINE DE SECURITE ===
    SlackCableTensioned     : BOOL; // --> [HW]  Cable M2 tendu (anti-mou)
    M3SensorWordIncoherent  : BOOL; // --> [HW]  Incoherence mot capteurs M3
    EmergencyChainClosed    : BOOL; // --> [SAFE] Chaine d'arret d'urgence fermee
    PowerContactorEngaged   : BOOL; // --> [HW]  Contacteur de puissance enclenche

    // === CODEURS & HOMING ===
    EncoderM1Operational    : BOOL; // --> [HW]  Codeur M1 operationnel
    EncoderM2Operational    : BOOL; // --> [HW]  Codeur M2 operationnel
    HomedM1                 : BOOL; // --> [HW]  Homing M1 effectue
    HomedM2                 : BOOL; // --> [HW]  Homing M2 effectue
    M1PositionInBounds      : BOOL; // --> [HW]  Position M1 dans les bornes
    M2PositionInBounds      : BOOL; // --> [HW]  Position M2 dans les bornes
END_VAR
VAR_OUTPUT
    // === ETATS ET DIAGNOSTIC ===
    PreflightOk             : BOOL; // <-- [STAT] Verdict preflight OK (aucun defaut)
    PreflightDone           : BOOL; // <-- [STAT] Verdict preflight termine
    PreflightBusy           : BOOL; // <-- [STAT] Diagnostic preflight en cours
    PreflightErrorId        : WORD; // <-- [DIAG] Masque de defauts preflight (16 bits)
END_VAR
VAR
    // === TRIGGERS & ETATS INTERNES ===
    ExecuteEdge             : R_TRIG;       // . [LOC] Trigger front montant Execute
    RequestPending          : BOOL;         // . [LOC] Demande de verdict en attente
END_VAR

{region "§1 Gate et demande"}
ExecuteEdge(CLK := Execute);
RequestPending := RequestPending OR ExecuteEdge.Q;
PreflightBusy := RequestPending AND NOT MachineIsStill;
IF NOT MachineIsStill THEN PreflightDone := FALSE; RETURN; END_IF;
IF NOT RequestPending THEN RETURN; END_IF;
{endregion}

{region "§2 Calcul verdict"}
PreflightErrorId := 0;
(* bit0..2 frein M1/M2/M3 serré; bit3..4 contacteurs M1/M2 retombés;
   bit5..6 thermique M1/M2 OK; bit7 thermique frein OK; bit8 phases OK;
   bit9 câble M2 tendu; bit10 mot M3 cohérent; bit11 contacteur sans chaîne;
   bit12..13 codeur M1/M2 opérationnel; bit14..15 homé et position bornée M1/M2. *)
IF NOT M1BrakeApplied THEN PreflightErrorId := PreflightErrorId OR 16#0001; END_IF;
IF NOT M2BrakeApplied THEN PreflightErrorId := PreflightErrorId OR 16#0002; END_IF;
IF NOT M3BrakeApplied THEN PreflightErrorId := PreflightErrorId OR 16#0004; END_IF;
IF NOT M1ContactorsReleased THEN PreflightErrorId := PreflightErrorId OR 16#0008; END_IF;
IF NOT M2ContactorsReleased THEN PreflightErrorId := PreflightErrorId OR 16#0010; END_IF;
IF NOT M1ThermalOk THEN PreflightErrorId := PreflightErrorId OR 16#0020; END_IF;
IF NOT M2ThermalOk THEN PreflightErrorId := PreflightErrorId OR 16#0040; END_IF;
IF NOT BrakeThermalOk THEN PreflightErrorId := PreflightErrorId OR 16#0080; END_IF;
IF NOT PhaseRotationOk THEN PreflightErrorId := PreflightErrorId OR 16#0100; END_IF;
IF NOT SlackCableTensioned THEN PreflightErrorId := PreflightErrorId OR 16#0200; END_IF;
IF M3SensorWordIncoherent THEN PreflightErrorId := PreflightErrorId OR 16#0400; END_IF;
IF PowerContactorEngaged AND NOT EmergencyChainClosed THEN PreflightErrorId := PreflightErrorId OR 16#0800; END_IF;
IF NOT EncoderM1Operational THEN PreflightErrorId := PreflightErrorId OR 16#1000; END_IF;
IF NOT EncoderM2Operational THEN PreflightErrorId := PreflightErrorId OR 16#2000; END_IF;
IF NOT HomedM1 OR NOT M1PositionInBounds THEN PreflightErrorId := PreflightErrorId OR 16#4000; END_IF;
IF NOT HomedM2 OR NOT M2PositionInBounds THEN PreflightErrorId := PreflightErrorId OR 16#8000; END_IF;
{endregion}

{region "§3 Sortie"}
PreflightOk := (PreflightErrorId = 0); PreflightDone := TRUE; RequestPending := FALSE;
{endregion}

(* =======================================================================
   🧩 FB_FaultCore — Socle de remplissage de ST_Fault (vue live + vue latchée)
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : à partir d'une LISTE de causes en clair (ARRAY[0..15] OF
      ST_FaultCause), produire la brique ST_Fault :
        • Error / ErrorId   : vue LIVE, suit les causes Active, retombe seule.
        • Latched / LatchedId: vue LATCHÉE, chaque bit figé à l'apparition d'une
          cause `Active AND Latching`, conservé jusqu'au front Reset.
   🔒 Sécurité :
      - interlock TOUJOURS sur `Causes[i].Active` (cause brute), jamais sur le latch.
      - Reset TOUJOURS effectif, jamais conditionné (std §9) — agit même Enable=FALSE.
      - Latching=FALSE → cause visible en LIVE seulement (fail-safe : reste visible).
      - un défaut laté non acquitté reste publié même FB désactivé (pas d'effacement
        silencieux sur bascule Enable OFF→ON sans Reset).
      - une cause à acquitter réapparue re-arme son bit (ré-alarme).
   🧬 Un seul socle, réutilisé par tout FB `standard` qui expose un ST_Fault.
      Pas de State, pas de Warning, pas de texte — hors périmètre de ce socle.
   📄 Doc : DOC/STDS/CODE_QUALITY_STANDARDS.md §2quinquies / §9
   ======================================================================= *)

FUNCTION_BLOCK FB_FaultCore
VAR_INPUT
    Enable  : BOOL;                          // --> [CMD] Autorisation générale du FB porteur
    Reset   : BOOL;                          // --> [CMD] Front d'acquittement (jamais conditionné, §9)
    Causes  : ARRAY[0..15] OF ST_FaultCause; // --> [CMD] Liste des causes en clair (Active / Latching / Texte)
END_VAR
VAR_OUTPUT
    Ready   : BOOL;                          // <-- [STAT] FB porteur autorisé (= Enable)
    Fault   : ST_Fault;                      // <-- [DIAG] Brique défaut (mappée 1:1 sur la sortie du FB porteur)
END_VAR
VAR
    ResetEdge : R_TRIG;                      // . [LOC] Front montant Reset
    Latch     : ARRAY[0..15] OF BOOL;        // . [LOC] Bit laté par cause, jusqu'au Reset
    i         : INT;                         // . [LOC] Index de parcours de la liste
END_VAR

{region "§1 Reset (front, toujours effectif)"}
// Front Reset : efface tous les latches. Jamais conditionné, agit même Enable=FALSE (§9).
Ready := Enable;
ResetEdge(CLK := Reset);
IF ResetEdge.Q THEN
    FOR i := 0 TO 15 DO Latch[i] := FALSE; END_FOR;
END_IF;
{endregion}

{region "§2 Vue LIVE — Error / ErrorId"}
// Bitfield des causes présentes MAINTENANT. Retombe seule à 0 quand les causes disparaissent.
// Non évaluée si Enable=FALSE (pas de lecture de cause hors autorisation).
Fault.ErrorId := 0;
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active THEN
            Fault.ErrorId := Fault.ErrorId OR SHL(WORD#1, INT_TO_DWORD(i));
        END_IF;
    END_FOR;
END_IF;
Fault.Error := (Fault.ErrorId <> 0);
{endregion}

{region "§3 Armement des latches (causes à acquitter)"}
// Une cause Active ET Latching arme son bit ; le bit reste jusqu'au front Reset (§1),
// même si la cause disparaît. Réapparition => ré-armement (ré-alarme).
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active AND Causes[i].Latching THEN
            Latch[i] := TRUE;
        END_IF;
    END_FOR;
END_IF;
{endregion}

{region "§4 Vue LATCHÉE — Latched / LatchedId"}
// Toujours publiée, y compris Enable=FALSE : un défaut non acquitté reste visible.
Fault.LatchedId := 0;
FOR i := 0 TO 15 DO
    IF Latch[i] THEN
        Fault.LatchedId := Fault.LatchedId OR SHL(WORD#1, INT_TO_DWORD(i));
    END_IF;
END_FOR;
Fault.Latched := (Fault.LatchedId <> 0);
{endregion}

END_FUNCTION_BLOCK

(* =======================================================================
   ⚙️ FB_CycleTime — Mesure du temps de cycle réel de la tâche
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul du dt réel entre deux exécutions successives
   ⚙️ Calcul : Delta horloge système TIME() converti en secondes
   🛡️ Robustesse : dt borné (0 < dt <= CST_MaxCycleDeltaMs) sinon valeur de secours —
      neutralise les artefacts de rebouclage TIME() (49,7 j) et de reprise (online-change, debug)
   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md
   ======================================================================= *)
FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

(* =======================================================================
   🧩 FB_FaultCore — Socle de remplissage de ST_Fault (vue live + vue latchée)
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : à partir d'une LISTE de causes en clair (ARRAY[0..15] OF
      ST_FaultCause), produire la brique ST_Fault :
        • Error / ErrorId   : vue LIVE, suit les causes Active, retombe seule.
        • Latched / LatchedId: vue LATCHÉE, chaque bit figé à l'apparition d'une
          cause `Active AND Latching`, conservé jusqu'au front Reset.
   🔒 Sécurité :
      - interlock TOUJOURS sur `Causes[i].Active` (cause brute), jamais sur le latch.
      - Reset TOUJOURS effectif, jamais conditionné (std §9) — agit même Enable=FALSE.
      - Latching=FALSE → cause visible en LIVE seulement (fail-safe : reste visible).
      - un défaut laté non acquitté reste publié même FB désactivé (pas d'effacement
        silencieux sur bascule Enable OFF→ON sans Reset).
      - une cause à acquitter réapparue re-arme son bit (ré-alarme).
   🧬 Un seul socle, réutilisé par tout FB `standard` qui expose un ST_Fault.
      Pas de State, pas de Warning, pas de texte — hors périmètre de ce socle.
   📄 Doc : DOC/STDS/CODE_QUALITY_STANDARDS.md §2quinquies / §9
   ======================================================================= *)

FUNCTION_BLOCK FB_FaultCore
VAR_INPUT
    Enable  : BOOL;                          // --> [CMD] Autorisation générale du FB porteur
    Reset   : BOOL;                          // --> [CMD] Front d'acquittement (jamais conditionné, §9)
    Causes  : ARRAY[0..15] OF ST_FaultCause; // --> [CMD] Liste des causes en clair (Active / Latching / Texte)
END_VAR
VAR_OUTPUT
    Ready   : BOOL;                          // <-- [STAT] FB porteur autorisé (= Enable)
    Fault   : ST_Fault;                      // <-- [DIAG] Brique défaut (mappée 1:1 sur la sortie du FB porteur)
END_VAR
VAR
    ResetEdge : R_TRIG;                      // . [LOC] Front montant Reset
    Latch     : ARRAY[0..15] OF BOOL;        // . [LOC] Bit laté par cause, jusqu'au Reset
    i         : INT;                         // . [LOC] Index de parcours de la liste
END_VAR

{region "§1 Reset (front, toujours effectif)"}
// Front Reset : efface tous les latches. Jamais conditionné, agit même Enable=FALSE (§9).
Ready := Enable;
ResetEdge(CLK := Reset);
IF ResetEdge.Q THEN
    FOR i := 0 TO 15 DO Latch[i] := FALSE; END_FOR;
END_IF;
{endregion}

{region "§2 Vue LIVE — Error / ErrorId"}
// Bitfield des causes présentes MAINTENANT. Retombe seule à 0 quand les causes disparaissent.
// Non évaluée si Enable=FALSE (pas de lecture de cause hors autorisation).
Fault.ErrorId := 0;
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active THEN
            Fault.ErrorId := Fault.ErrorId OR SHL(WORD#1, INT_TO_DWORD(i));
        END_IF;
    END_FOR;
END_IF;
Fault.Error := (Fault.ErrorId <> 0);
{endregion}

{region "§3 Armement des latches (causes à acquitter)"}
// Une cause Active ET Latching arme son bit ; le bit reste jusqu'au front Reset (§1),
// même si la cause disparaît. Réapparition => ré-armement (ré-alarme).
IF Enable THEN
    FOR i := 0 TO 15 DO
        IF Causes[i].Active AND Causes[i].Latching THEN
            Latch[i] := TRUE;
        END_IF;
    END_FOR;
END_IF;
{endregion}

{region "§4 Vue LATCHÉE — Latched / LatchedId"}
// Toujours publiée, y compris Enable=FALSE : un défaut non acquitté reste visible.
Fault.LatchedId := 0;
FOR i := 0 TO 15 DO
    IF Latch[i] THEN
        Fault.LatchedId := Fault.LatchedId OR SHL(WORD#1, INT_TO_DWORD(i));
    END_IF;
END_FOR;
Fault.Latched := (Fault.LatchedId <> 0);
{endregion}

(* =======================================================================
   ⚙️ FB_Ramp — Rampe d'accélération et décélération asymétrique
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Limitation de variation anti-à-coups (%/s) sur signal analogique
   🔒 Polarité : Distingue l'éloignement de zéro (Accel) et le retour à zéro (Decel)
   📄 Doc métier : DOC/AF/AF_Partie-11_Fonction_Translation_v2.3.md §1 (TC-P11-040)
   ======================================================================= *)
FUNCTION_BLOCK FB_Ramp
VAR_INPUT
    // === CONSIGNES ET TAUX DE RAMPE ===
    Target                  : REAL;               // --> [CMD] Valeur cible visee (-100.0..+100.0 %)
    AccelRate               : REAL;               // --> [CFG] Taux d'acceleration (%/s)
    DecelRate               : REAL;               // --> [CFG] Taux de deceleration (%/s)
    CycleTimeS              : REAL;               // --> [CFG] Temps de cycle reel en secondes (dt)
END_VAR
VAR_OUTPUT
    // === VALEUR RAMPEE ===
    Current                 : REAL;               // <-- [STAT] Valeur courante amortie (-100.0..+100.0 %)
END_VAR
VAR
    // === CALCULS INTERNES ===
    StepX                   : REAL;               // . [LOC] Pas de variation pour le cycle courant
    TargetLimited           : REAL;               // . [LOC] Cible bornee a [-100.0, +100.0]
END_VAR

// Détection du comportement (Accélération vs Décélération)
// Classification par CÔTÉ DE ZÉRO + sens de déplacement, plus par signe brut de Target seul
// (robuste à une traversée de zéro rapide).
// 🛡️ Garde-fou : la rampe ne sort jamais de la plage normalisée ±100 %.
TargetLimited := LIMIT(-100.0, Target, 100.0);
Current := LIMIT(-100.0, Current, 100.0);

IF (Current >= 0.0 AND TargetLimited >= Current) OR (Current <= 0.0 AND TargetLimited <= Current) THEN
    StepX := AccelRate * CycleTimeS; // S'éloigne du zéro (reste du même côté, va plus loin)
ELSE
    StepX := DecelRate * CycleTimeS; // Retourne vers zéro, ou le traverse vers l'autre signe
END_IF;

// Application de la rampe
IF Current < TargetLimited THEN
    Current := Current + StepX;
    IF Current > TargetLimited THEN Current := TargetLimited; END_IF;
ELSIF Current > TargetLimited THEN
    Current := Current - StepX;
    IF Current < TargetLimited THEN Current := TargetLimited; END_IF;
END_IF;

// 🛡️ Limitation finale : même une valeur d'état déjà corrompue reste bornée.
Current := LIMIT(-100.0, Current, 100.0);
