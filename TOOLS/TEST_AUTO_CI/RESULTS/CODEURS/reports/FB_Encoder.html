{region "§1 Gate"}
// 🛑 1. GATE
IF NOT Enable THEN
    EncoderAvailable := FALSE;
    Status.State := E_State.DISABLED;
    Ready := FALSE; Status.Busy := FALSE; Status.Done := FALSE;
    PresetTriggerCmd := 16#0000; CodeSeqTriggerCmd := 16#0000; PresetValueOut := 0;
    PresetSeqStep := 0;
    PresetTimer(IN := FALSE, PT := PresetTimeout);
    PresetTimerVisual(IN := FALSE, PT := T#500MS);
    RETURN;
END_IF;

Ready := TRUE;
Status.State := E_State.READY;
{endregion}

{region "§2 Reset front"}
// 🔑 2. RESET FRONT
ResetEdge(CLK := Reset);
{endregion}

{region "§3 Diagnostic bus et alarme"}
// 🔍 3. DIAGNOSTIC BUS / ALARME
IF (AlarmsIn <> 0) OR NOT SlaveOperational THEN
    Status.ErrorId := Status.ErrorId OR 16#0001;
ELSE
    Status.ErrorId := Status.ErrorId AND 16#FFFE; (* Auto-clearing dès que le bus et l'esclave sont OK *)
END_IF;
EncoderAvailable := (Status.ErrorId AND 16#0001) = 0;
{endregion}

{region "§4 Lecture position"}
// 📥 4. LECTURE POSITION — GELER SUR DOUTE (Partie10 §6)
// Si EncoderAvailable=FALSE : RawPos/AngleRaw/TurnCount conservent leur dernière valeur (pas de RETURN
// global ici pour ne pas geler aussi la séquence preset/l'état — seule la lecture est gelée).
IF EncoderAvailable THEN
    RawPos := RawPosIn;   // Sens de comptage géré côté codeur (CoE 6000h bit0, startup parameter CODESYS) — voir Partie10 §3bis. Plus de calcul ici.
    AngleRaw := UDINT_TO_UINT(RawPos MOD PointsPerRev);
    TurnCount := RawPos / PointsPerRev;
END_IF;
{endregion}

{region "§5 Séquence preset"}
// 🔧 5. SÉQUENCE PRESET (écriture bus)
// PresetTriggerCmd := 2 déclenche le référencement à PresetValue (valeur de commande, pas un
// masque de bits). CodeSeqTriggerCmd : rôle non identifié — laissé à 0, PAS piloté tant que sa
// fonction n'est pas confirmée. Ne jamais lui donner une valeur arbitraire.
PresetReqEdge(CLK := PresetRequest);
PresetAck := FALSE; PresetNak := FALSE; // pulses 1 cycle uniquement
CodeSeqTriggerCmd := 16#0000;           // non piloté — rôle inconnu, voir note ci-dessus
CASE PresetSeqStep OF
    0: // IDLE
        PresetTriggerCmd := 0;
        IF PresetReqEdge.Q AND EncoderAvailable THEN
            PresetValueOut := PresetValue;
            PresetSeqStep := 1;
            PresetTimer(IN := TRUE, PT := PresetTimeout);
        END_IF;
    1: // Déclenchement preset (laissé actif pendant 0.5s pour confort visuel en simulation/bus)
        PresetTriggerCmd := 2;   // Référence le codeur à PresetValueOut
        RawDiff := UDINT_TO_DINT(RawPos) - UDINT_TO_DINT(PresetValueOut);
        IF ABS(RawDiff) <= UDINT_TO_DINT(PresetTolerancePts) THEN
            PresetTimer(IN := FALSE, PT := PresetTimeout);
            PresetTimerVisual(IN := TRUE, PT := T#500MS);
            IF PresetTimerVisual.Q THEN
                PresetTriggerCmd := 0;
                PresetAck := TRUE;
                PresetSeqStep := 0;
                PresetTimerVisual(IN := FALSE, PT := T#500MS);
            END_IF;
        ELSIF PresetTimer.Q THEN
            PresetTriggerCmd := 0;
            PresetNak := TRUE;
            Status.ErrorId := Status.ErrorId OR 16#0002;
            PresetSeqStep := 0;
            PresetTimer(IN := FALSE, PT := PresetTimeout);
            PresetTimerVisual(IN := FALSE, PT := T#500MS);
        END_IF;
END_CASE;
IF ResetEdge.Q AND (PresetSeqStep = 0) THEN
    Status.ErrorId := Status.ErrorId AND 16#FFFD;
END_IF;
{endregion}

{region "§6 Error et StateAtError"}
// 🧾 6. ERROR / STATEATERROR
Status.Error := (Status.ErrorId <> 16#0000);
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;

Status.Busy := (PresetSeqStep <> 0);
Status.Done := PresetAck;
{endregion}

   ⚙️ FB_Encoder_Homing — Référencement codeur nominal ou unitaire
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Séquence de référencement nominal (capteur haut) ou unitaire (MAINT_N2)
   🔒 Polarité : Homed=TRUE dès qu'un référencement valide est enregistré
   🔌 Architecture : Expose ST_Encoder_Calib en VAR_IN_OUT
   📄 Doc métier : DOC/AF/AF_Partie-09_Fonction_Encoder_v2.1.md
   ======================================================================= *)

// === DECLARATION (à coller dans la partie déclaration du POU FB_Encoder_Homing) ===

FUNCTION_BLOCK PUBLIC FB_Encoder_Homing
VAR_INPUT
    // === COMMANDES & AUTORISATIONS ===
    Enable                 : BOOL;        // --> [CMD] Autorisation generale FB (TRUE=Autorise)
    Reset                  : BOOL;        // --> [CMD] Acquittement defaut (front montant)
    Home                   : BOOL;        // --> [CMD] Demande de referencement (front)
    HomingPermit           : BOOL;        // --> [CMD] Autorisation de homer (calculee par le caller)
    BtnConfirmCoherence    : BOOL;        // --> [CMD] Levee de doute d'incoherence au boot (front)

    // === CIBLES & CONFIGURATION HOMING ===
    CfgHomingTargetM       : REAL;        // --> [CFG] Cible libre homing unitaire (m)
    UseDynamicTarget       : BOOL := FALSE; // --> [CFG] Activation cible dynamique
    DynamicHomingTargetM   : REAL := 0.0; // --> [CFG] Cible dynamique de referencement (m)
    CfgTopSensorPosM       : REAL := 8.5; // --> [CFG] Cible homing nominal capteur haut (m)

    // === RETOURS HW & CAPTEURS ===
    TopPositionSensor      : BOOL;        // --> [HW]  Capteur physique position haute unique
    FwdRevSpeedFeedbackOff : BOOL;        // --> [HW]  Contacteurs sens et vitesse retombes
    BrakeFeedback          : BOOL;        // --> [HW]  Contacteur frein ouvert/desserre

    // === MESURES & RETOURS FB CODEUR ABSOLU ===
    RawPos                 : UDINT;       // --> [HW]  Position brute codeur (sortie FB_Encoder_Abs)
    EncoderAvailable       : BOOL;        // --> [HW]  Codeur absolu disponible et valide
    PresetAck              : BOOL;        // --> [HW]  Pulse confirmation ecriture preset reussie
    PresetNak              : BOOL;        // --> [HW]  Pulse echec / timeout ecriture preset

    // === CARACTERISTIQUES MECANIQUES & BUS ===
    PointsPerRev           : UDINT := 8192; // --> [CFG] Resolution codeur (pts/tour)
    MultiTurnRevsMax       : UDINT := 4096; // --> [CFG] Capacite multitour max codeur (tours)
    CableM_PerRev          : REAL := 2.0;   // --> [CFG] Perimetre tambour / developpement cable (m/tour)

    // === TESTS & BYPASS ===
    BypassGlobal           : BOOL := FALSE; // --> [TST] Neutralisation securites pour mise en service
END_VAR
VAR_OUTPUT
    // === ETATS IEC & SYNOPTIQUE ===
    Ready                  : BOOL;        // <-- [STAT] FB pret a fonctionner
    // 🧩 Statut transverse (forme cible) — regroupe Busy/Done/Error/ErrorId/State/StateAtError.
    // ErrorId : masque de bits de défaut (bit0: mode, bit1: treuil, bit4: cible, bit5: preset, bit9: incoherence boot).
    Status                 : ST_FbStatus;

    // === COMMANDES VERS FB_ENCODER_ABS ===
    PresetRequest          : BOOL;        // <-- [ACT]  Pulse demande d'ecriture preset SDO
    PresetValue            : UDINT;       // <-- [ACT]  Valeur de cible brute pour preset SDO

    // === RESULTATS & MEMOIRE CALIBRATION ===
    Homed                  : BOOL;        // <-- [STAT] Codeur pagine et referencie
    HomingSuspect          : BOOL;        // <-- [DIAG] Incoherence redemarrage a lever
    HomingRefRaw           : UDINT;       // <-- [DIAG] Offset de reference brut calcule
END_VAR
VAR_IN_OUT
    // === STRUCTURE DE CALIBRATION PERSISTANTE ===
    Calib                  : ST_Encoder_Calib; // <-> [BUS] Structure de calibration RETAIN
END_VAR
VAR
    // === DETECTEURS DE FRONT ===
    ResetEdge              : R_TRIG;      // . [LOC] Detecteur front montant Reset
    ErrorEdge              : R_TRIG;      // . [LOC] Detecteur front montant Error
    HomeEdge               : R_TRIG;      // . [LOC] Detecteur front montant Home
    ConfirmCoherenceEdge   : R_TRIG;      // . [LOC] Detecteur front montant BtnConfirmCoherence
    TopSensorEdge          : R_TRIG;      // . [LOC] Detecteur front montant TopPositionSensor
    DynamicTargetEdge      : R_TRIG;      // . [LOC] Detecteur front montant UseDynamicTarget

    // === ETATS & DRAPEAUX INTERNES ===
    InitDone               : BOOL;        // . [LOC] Controle incoherence boot effectue
    ArretConfirme          : BOOL;        // . [LOC] Condition d'arret moteur confirmee
    HomingModeOk           : BOOL;        // . [LOC] Mode de fonctionnement valide pour homing
    NominalHomingTrigger   : BOOL;        // . [LOC] Condition de declenchement homing nominal
    UnitaryHomingTrigger   : BOOL;        // . [LOC] Condition de declenchement homing unitaire
    DynamicHomingTrigger   : BOOL;        // . [LOC] Condition de declenchement homing dynamique
    HomingTrigger          : BOOL;        // . [LOC] Synthese declenchement homing

    // === VARIABLES DE CALCUL ===
    TargetPoints           : DINT;        // . [LOC] Ecart cible converti en points
    TargetPositionM        : REAL;        // . [LOC] Cible retenue en metres
    RawDiffRestart         : DINT;        // . [LOC] Ecart brut de position au redemarrage
END_VAR

// === IMPLEMENTATION (à coller dans le corps ST du POU FB_Encoder_Homing) ===

{region "§1 Gate neutralisation"}
// 🛑 1. GATE — neutralisation ; on continue toutefois d'exposer le dernier état RETAIN connu
// (FB_Encoder_Scale a besoin de HomingRefRaw même si ce FB est désactivé)
// PowerContactorEngaged retiré du GATE (le codeur 24V / EtherCAT peut être homé disjoncteurs coupés)
IF NOT Enable THEN
    PresetRequest := FALSE;
    PresetValue := 0;
    Status.State := E_State.DISABLED;
    Ready := FALSE; Status.Busy := FALSE; Status.Done := FALSE;
    InitDone := FALSE;   // Force un nouveau contrôle §3.7 à la prochaine activation
    Homed := Calib.Homed;
    HomingSuspect := Calib.HomingSuspect;
    HomingRefRaw := Calib.HomingRefRaw;
    RETURN;
END_IF;

Ready := TRUE;

{endregion}

{region "§2 Fronts et coherence demarrage"}
// 🔑 2. FRONTS
ResetEdge(CLK := Reset);
HomeEdge(CLK := Home);
ConfirmCoherenceEdge(CLK := BtnConfirmCoherence);

// 🔍 3. COHÉRENCE AU REDÉMARRAGE (§3.7) — une fois par session Enable, dès que le codeur répond
IF NOT InitDone THEN
    Status.State := E_State.INIT;
    IF EncoderAvailable THEN
        RawDiffRestart := UDINT_TO_DINT(RawPos) - UDINT_TO_DINT(Calib.LastKnownRawPos);
        IF Calib.Homed AND (ABS(RawDiffRestart) > UDINT_TO_DINT(Calib.RestartCoherenceTolerancePts)) THEN
            Calib.HomingSuspect := TRUE;
            Status.ErrorId := Status.ErrorId OR 16#0200;   // bit9 : incohérence codeur au redémarrage
        END_IF;
        InitDone := TRUE;
    END_IF;
ELSE
    IF Status.State = E_State.INIT THEN Status.State := E_State.READY; END_IF;
END_IF;

{endregion}

{region "§3 Confirmation coherence et demande homing"}
// ✅ 4. LEVÉE DU DOUTE (BtnConfirmCoherence, §3.7) — indépendant du Reset générique
IF ConfirmCoherenceEdge.Q AND HomingPermit THEN
    Calib.HomingSuspect := FALSE;
    Status.ErrorId := Status.ErrorId AND 16#FDFF;   // efface bit9
END_IF;

PresetRequest := FALSE;
TopSensorEdge(CLK := TopPositionSensor);
DynamicTargetEdge(CLK := UseDynamicTarget);

// Gates rejetées = Warning auto-effaçable (§9/TC-P03-007), pas un Fault : recalculées à
// chaque scan, jamais latchées, aucun Reset requis pour les effacer.
Status.ErrorId := Status.ErrorId AND 16#FFEC;   // efface bits 0/1/4 (garde bit5, bit9 — inchangés)

HomingModeOk := HomingPermit;

IF HomeEdge.Q AND NOT HomingModeOk THEN
    Status.ErrorId := Status.ErrorId OR 16#0001;   // bit0 : homing non autorise
END_IF;

// Nominal : capture AU FRONT du capteur haut, pas après arrêt confirmé — AscentPermit
// (FB_Safety_Winch) coupe déjà la montée indépendamment dès que TopPositionSensor est actif ;
// attendre l'arrêt casserait la répétabilité recherchée à vitesse d'accostage constante.
// Couvre les deux cas : Home déjà maintenu quand le capteur déclenche, ou capteur déjà actif
// quand Home est pressé.
NominalHomingTrigger := HomingModeOk AND NOT UseDynamicTarget
                         AND ((Home AND TopSensorEdge.Q) OR (HomeEdge.Q AND TopPositionSensor));

// Unitaire : cible libre (CfgHomingTargetM), déclenchée par la demande (front).
UnitaryHomingTrigger := HomingModeOk AND NOT UseDynamicTarget AND HomeEdge.Q;

// Dynamique (benne) : HomingModeOk vérifié ici en défense.
DynamicHomingTrigger := HomingModeOk AND (DynamicTargetEdge.Q OR (UseDynamicTarget AND HomeEdge.Q));

HomingTrigger := NominalHomingTrigger OR UnitaryHomingTrigger OR DynamicHomingTrigger;

IF HomingTrigger THEN
    IF UseDynamicTarget THEN
        TargetPositionM := DynamicHomingTargetM;
    ELSIF NominalHomingTrigger THEN
        TargetPositionM := CfgTopSensorPosM;
    ELSE
        TargetPositionM := CfgHomingTargetM;
    END_IF;

    // Bornage obligatoire avant écriture (CODE_QUALITY_STANDARDS §6) — toute cible, quelle
    // que soit son origine (config, calcul benne, capteur haut), est vérifiée dans la plage
    // physique documentée [-99;+99] m avant d'être gravée dans la référence codeur.
    IF (TargetPositionM < -99.0) OR (TargetPositionM > 99.0) THEN
        Status.ErrorId := Status.ErrorId OR 16#0010;   // bit4 : cible hors plage, homing refusé
    ELSE
        TargetPoints := REAL_TO_DINT(TargetPositionM * UDINT_TO_REAL(PointsPerRev) / CableM_PerRev);
        Calib.HomingRefRaw := DINT_TO_UDINT(UDINT_TO_DINT(RawPos) - TargetPoints);
        Calib.Homed := TRUE;
        Calib.HomingSuspect := FALSE;
        Calib.LastKnownRawPos := RawPos;
        HomingRefRaw := Calib.HomingRefRaw;
        Homed := TRUE;
        HomingSuspect := FALSE;
        Status.Done := TRUE;
        Status.ErrorId := 16#0000;
        Status.State := E_State.READY;

        // Envoi de la demande de Preset SDO au matériel : la valeur brute injectee
        // correspond a la cible (calcul inverse) pour que CablePosM = cible.
        PresetValue := DINT_TO_UDINT(UDINT_TO_DINT(Calib.HomingRefRaw) + TargetPoints);
        PresetRequest := TRUE;
    END_IF;
END_IF;

{endregion}

{region "§4 Suivi SDO reset et sorties etat"}
// ⏳ 6. SUIVI DE LA SÉQUENCE SDO OPTIONNELLE
IF Status.State = E_State.BUSY THEN
    IF PresetAck THEN
        Status.Done := TRUE;
        Status.State := E_State.READY;
        // Un homing réussi efface l'intégralité de ErrorId
        Status.ErrorId := 16#0000;
    ELSIF PresetNak THEN
        Status.ErrorId := Status.ErrorId OR 16#0020;        // bit5 : preset refusé/timeout (voir FB_Encoder_Abs)
        Status.State := E_State.READY;
    END_IF;
END_IF;

// 🧾 7. RESET (front) — efface les bits « dernière tentative rejetée » (pas de cause persistante
// à revérifier ici : le prochain Home réévalue toutes les conditions de zéro, voir §5)
IF ResetEdge.Q THEN
    Status.ErrorId := Status.ErrorId AND 16#FFC0;   // efface bits 0..5 (garde bit9, levé seulement par BtnConfirmCoherence/Home)
END_IF;

// 📤 8. SORTIES ÉTAT
Status.Error := (Status.ErrorId <> 16#0000);
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;

// Mise à jour continue du point brut valide connu (hors séquence de homing elle-même)
IF EncoderAvailable AND (Status.State <> E_State.BUSY) THEN
    Calib.LastKnownRawPos := RawPos;
END_IF;

Status.Busy := (Status.State = E_State.BUSY);
Homed := Calib.Homed AND NOT Calib.HomingSuspect; // Perd l'état homed si suspect (démarrage ou fonctionnement), restauré après BtnConfirmCoherence
HomingSuspect := Calib.HomingSuspect;
HomingRefRaw := Calib.HomingRefRaw;

{endregion}

(* =======================================================================
   ⚙️ FB_Encoder_Scale — Mise à l'échelle points → mètres (1 par treuil)
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul de la position câble en mètres par conversion arithmétique signée
   🔒 Polarité : CablePosM signée (+ enroulé, - sous l'eau)
   🔌 Architecture : Convertisseur de mesure unitaire par treuil
   📄 Doc métier : DOC/AF/AF_Partie-09_Fonction_Encoder_v2.1.md
   ======================================================================= *)

// === DECLARATION (à coller dans la partie déclaration du POU FB_Encoder_Scale) ===

FUNCTION_BLOCK PUBLIC FB_Encoder_Scale
VAR_INPUT
    // === DONNEES ENTREE & REFERENCES BRUTES ===
    RawPos          : UDINT;       // --> [HW]  Position brute codeur (sortie FB_Encoder_Abs, gelee sur doute)
    HomingRefRaw    : UDINT;       // --> [CFG] Reference brute figee au dernier homing (ST_Encoder_Calib)

    // === PARAMETRES & RATIOS DE CONVERSION ===
    CableM_PerRev   : REAL := 2.0; // --> [CFG] Developpement cable par tour tambour (m/tour)
    PointsPerRev    : UDINT := 8192;// --> [CFG] Resolution codeur (pts/tour)
END_VAR
VAR_OUTPUT
    // === MESURE CONVERTIE ===
    CablePosM       : REAL;        // <-- [DIAG] Position cable en metres (+ enroule, - sous l'eau)
END_VAR
VAR
    // === CALCULS INTERNES ===
    RawDiff         : DINT;        // . [LOC] Ecart signe RawPos - HomingRefRaw
END_VAR

// === IMPLEMENTATION (à coller dans le corps ST du POU FB_Encoder_Scale) ===

// 🧮 Écart signé sûr : conversion DINT AVANT soustraction (§3.1) — jamais l'inverse
RawDiff := UDINT_TO_DINT(RawPos) - UDINT_TO_DINT(HomingRefRaw);

// 📏 Formule fixe points → mètres (Partie10 §2), garde division par zéro
IF PointsPerRev > 0 THEN
    CablePosM := DINT_TO_REAL(RawDiff) * (CableM_PerRev / UDINT_TO_REAL(PointsPerRev));
ELSE
    CablePosM := 0.0;
END_IF;

(* =======================================================================
   🛡️ FB_Encoder_Safety — Bornage physique & surveillance cohérence
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Bornage physique absolu [-99; +99]m et relais d'incohérence redémarrage
   🔒 Polarité : EncoderIncoherent=TRUE en cas de hors bornes ou HomingSuspect

   📄 Doc métier : DOC/AF/AF_Partie-09_Fonction_Encoder_v2.1.md

FUNCTION_BLOCK PUBLIC FB_Encoder_Safety
VAR_INPUT
    // === COMMANDES & AUTORISATIONS ===
    Enable                : BOOL;        // --> [CMD] Autorisation generale FB (TRUE=Autorise)
    Reset                 : BOOL;        // --> [CMD] Acquittement defaut (front montant)

    // === MESURES & ALERTE HOMING ===
    CablePosM             : REAL;        // --> [HW]  Position calculee (m, sortie FB_Encoder_Scale)
    HomingSuspect         : BOOL;        // --> [SAFE] Alerte incoherence redemarrage (FB_Encoder_Homing)

    // === REGLAGES & BORNES PHYSIQUES ===
    PositionMinM          : REAL := -99.0; // --> [CFG] Plafond bornage physique bas (m)
    PositionMaxM          : REAL := 99.0;  // --> [CFG] Plafond bornage physique haut (m)

    // === TESTS & BYPASS ===

END_VAR
VAR_OUTPUT
    // === ETATS IEC & SYNOPTIQUE ===
    Ready                 : BOOL;        // <-- [STAT] FB pret a fonctionner
    // 🧩 Statut transverse (forme cible) — regroupe Busy/Done/Error/ErrorId/State/StateAtError.
    // ErrorId bit0: "Bornes", bit1: "Boot suspect".
    Status                : ST_FbStatus;

    // === MESURES SURE & SECURITE ===
    CablePosMSafe         : REAL;        // <-- [DIAG] Position cable instantanee transmise (m)
    EncoderIncoherent     : BOOL;        // <-- [SAFE] Incoherence ou hors bornes (TRUE = defaut)
END_VAR
VAR
    // === DETECTEURS DE FRONT ===
    ResetEdge             : R_TRIG;      // . [LOC] Detecteur front montant Reset
    ErrorEdge             : R_TRIG;      // . [LOC] Detecteur front montant Error
END_VAR

{region "§1 Gate"}
// 🛑 1. GATE
IF NOT Enable THEN
    Ready := FALSE; Status.Busy := FALSE; Status.Done := FALSE;
    Status.State := E_State.DISABLED;
    CablePosMSafe := CablePosM;
    EncoderIncoherent := FALSE;
    RETURN;
END_IF;

Ready := TRUE;
Status.State := E_State.READY;
{endregion}

{region "§2 Reset front"}
// 🔑 2. RESET FRONT
ResetEdge(CLK := Reset);
{endregion}

{region "§3 Bypass global"}
// 🌐 3. BYPASS GLOBAL
IF BypassGlobal THEN
    Status.ErrorId := 16#0000;
END_IF;
{endregion}

{region "§4 Surveillance bornage"}
// 🔍 4. SURVEILLANCE BORNAGE PHYSIQUE ABSOLU (bit0)
IF NOT BypassGlobal AND ((CablePosM < PositionMinM) OR (CablePosM > PositionMaxM)) THEN
    Status.ErrorId := Status.ErrorId OR 16#0001;
ELSE
    Status.ErrorId := Status.ErrorId AND 16#FFFE; (* Auto-clearing dès retour dans la plage *)
END_IF;
{endregion}

{region "§5 Relais incohérence redémarrage"}
// 🔍 5. RELAIS INCOHÉRENCE REDÉMARRAGE (bit1)
IF NOT BypassGlobal THEN
    IF HomingSuspect THEN
        Status.ErrorId := Status.ErrorId OR 16#0002;
    ELSE
        Status.ErrorId := Status.ErrorId AND 16#FFFD;
    END_IF;
END_IF;
{endregion}

{region "§6 Transmission mesure"}
// 📏 6. TRANSMISSION DE LA MESURE (Toujours réelle, non figée)
CablePosMSafe := CablePosM;
{endregion}

{region "§7 Sorties état et sécurité"}
// 📤 7. SORTIES ÉTAT & SÉCURITÉ
Status.Error := (Status.ErrorId <> 16#0000);
ErrorEdge(CLK := Status.Error);
IF ErrorEdge.Q THEN
    Status.StateAtError := Status.State;
END_IF;

EncoderIncoherent := Status.Error;
Status.Busy := FALSE;
Status.Done := FALSE;
{endregion}

(* =======================================================================
   🛡️ FB_EncoderReliability — Gates de fiabilite de la chaine codeur
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Synthetise les gates de fiabilite d'un treuil a partir des
      faits produits par la chaine codeur (Abs/Homing/Safety), pur calcul
      combinatoire, sans memoire ni machine d'etat.
   🔒 Polarité :
      EncoderFault     = NOT EncoderAvailable OR EncoderIncoherent   (gate GENERAL, SANS Homed)
      HomedAndReliable = EncoderAvailable AND Homed AND NOT EncoderIncoherent (gate STRICT M3)
   🧩 Responsabilite scindee (D-A2bis) : Homing = referencement, Safety = coherence,
      ce FB = « le codeur est fiable ». Aucune recomposition chez les consommateurs.
   📄 Doc : DOC/AF/AF_Partie-09_Fonction_Encoder_v2.1.md
   ======================================================================= *)

FUNCTION_BLOCK PUBLIC FB_EncoderReliability
VAR_INPUT
    // === FAITS DE LA CHAINE CODEUR ===
    EncoderAvailable  : BOOL;        // --> [HW]  Codeur operationnel sur bus (FB_Encoder_Abs)
    Homed             : BOOL;        // --> [STAT] Codeur reference (FB_Encoder_Homing)
    EncoderIncoherent : BOOL;        // --> [SAFE] Incoherence ou hors bornes (FB_Encoder_Safety)
END_VAR
VAR_OUTPUT
    // === GATES DE FIABILITE ===
    EncoderFault      : BOOL;        // <-- [SAFE] Gate general fiabilite (SANS Homed) — vitesse/mouvements
    HomedAndReliable  : BOOL;        // <-- [STAT] Gate strict M3 : disponible ET reference ET pas incoherent
END_VAR

// 🛡️ 1. GATE GENERAL (D-A11, C1) — PAS de Homed : non-reference ≠ incoherent.
//    Ce gate sert a la fiabilite de la mesure (vitesse, mouvements, PositionsValid).
EncoderFault := (NOT EncoderAvailable) OR EncoderIncoherent;

// 🎯 2. GATE STRICT — reserve a l'interlock hauteur M3 (C1) : on ne translate pas
//    si la hauteur est inconnue (jamais home) ou douteuse.
HomedAndReliable := EncoderAvailable AND Homed AND (NOT EncoderIncoherent);

(* =======================================================================
   ⚙️ FB_Encoder_SpeedMeasure — Mesure de vitesse câble horodatée
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul de la vitesse absolue et signée sur fenêtre glissante horodatée
   🔒 Polarité : Valid=TRUE après 6 échantillons couvrant au moins 50ms
   📄 Doc métier : DOC/AF/AF_Partie-09_Fonction_Encoder_v2.1.md
   ======================================================================= *)

FUNCTION_BLOCK PUBLIC FB_Encoder_SpeedMeasure
VAR_INPUT
    // === COMMANDES & AUTORISATIONS ===
    Enable              : BOOL;        // --> [CMD] Autorisation generale FB (TRUE=Autorise)
    Reset               : BOOL;        // --> [CMD] Purge immediate de l'historique (front montant)

    // === MESURE SURES EN ENTREE ===
    Position_M          : REAL;        // --> [HW]  Position cable sure en metres (FB_Encoder_Safety)
    PositionValid       : BOOL;        // --> [SAFE] Validite de la chaine codeur
END_VAR
VAR_OUTPUT
    // === VITESSE CALCULÉE & VALIDITE ===
    Speed_Mps           : REAL;        // <-- [DIAG] Vitesse absolue sur la fenetre glissante (m/s)
    SignedSpeed_Mps     : REAL;        // <-- [DIAG] Vitesse signee (+ montee, - descente) (m/s)
    Valid               : BOOL;        // <-- [STAT] Vitesse valide (6 ech. sur 50 ms)
END_VAR
VAR
    // === HISTORIQUE & ECHANTILLONS ===
    Positions_M         : ARRAY[0..5] OF REAL; // . [LOC] Tableau des 6 positions enregistrees
    Timestamps          : ARRAY[0..5] OF TIME; // . [LOC] Horodatage associe aux 6 positions

    // === HORODATAGE & TEMPORISATION ===
    CurrentTimestamp    : TIME;        // . [LOC] Horodatage courant du cycle
    LastTimestamp       : TIME;        // . [LOC] Horodatage de l'echantillon precedent
    ElapsedSinceSample  : TIME;        // . [LOC] Temps ecoule depuis le dernier echantillon
    WindowElapsed       : TIME;        // . [LOC] Temps total ecoule sur la fenetre de mesure

    // === COMPTEURS & ETATS INTERNES ===
    CollectedSamples    : INT;         // . [LOC] Nombre d'echantillons collectes dans la fenetre
    Index               : INT;         // . [LOC] Index de boucle de decalage
    TimestampInitialized: BOOL;        // . [LOC] Indicateur d'initialisation du premier horodatage
END_VAR
VAR CONSTANT
    // === PARAMETRES DE LA FENETRE GLISSANTE ===
    WindowTime          : TIME := T#50MS; // Fenetre glissante de mesure
    SamplePeriod        : TIME := T#10MS; // Periode minimale entre deux echantillons
    SampleCount         : INT := 6;       // Nombre total d'echantillons requis
END_VAR

{region "§1 Horodatage et purge"}
// ⏱️ Horodatage natif : le calcul utilise le temps réellement écoulé, jamais un cycle supposé.
CurrentTimestamp := TIME();

// Purge des six positions sur rupture de confiance : aucune source antérieure ne peut contaminer la mesure.
IF NOT Enable OR Reset OR NOT PositionValid THEN
    Speed_Mps := 0.0;
    SignedSpeed_Mps := 0.0;
    Valid := FALSE;
    CollectedSamples := 0;
    TimestampInitialized := FALSE;
    RETURN;
END_IF;
{endregion}

{region "§2 Première position"}
// 🕒 Première position de la nouvelle fenêtre.
IF NOT TimestampInitialized THEN
    Positions_M[0] := Position_M;
    Timestamps[0] := CurrentTimestamp;
    LastTimestamp := CurrentTimestamp;
    CollectedSamples := 1;
    TimestampInitialized := TRUE;
    Speed_Mps := 0.0;
    SignedSpeed_Mps := 0.0;
    Valid := FALSE;
    RETURN;
END_IF;
{endregion}

{region "§3 Rebouclage TIME"}
// 🛡️ Rebouclage de TIME détecté localement : purge sans réutiliser le delta aberrant.
