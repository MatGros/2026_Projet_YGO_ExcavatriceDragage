(* =======================================================================
   ⚙️ FB_CycleTime — Mesure du temps de cycle réel de la tâche
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul du dt réel entre deux exécutions successives
   ⚙️ Calcul : Delta horloge système TIME() converti en secondes
   🛡️ Robustesse : dt borné (0 < dt <= CST_MaxCycleDeltaMs) sinon valeur de secours —
      neutralise les artefacts de rebouclage TIME() (49,7 j) et de reprise (online-change, debug)
   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md
   ======================================================================= *)
FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

(* ═══════════════════════════════════════════════════════════════
   🧪 FB_Sim_Translation — Progression capteurs simulés M3
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Produit les cinq capteurs de position M3 par progression continue
      de Trémie vers Maintenance. Le modèle ne publie que les six mots valides
      de FB_Translation_PositionDecoder : 11111 → 01111 → 00111 → 00011 →
      00001 → 00000.
   📏 Convention : PositionProgress en mètres, 0 m = Trémie … 30 m = Maintenance.
      Distances NON-linéaires : Trémie(0)→PV(5)=5 m, PV(5)→P2(15)=10 m,
      P2(15)→P1(20)=5 m, P1(20)→Maintenance(30)=10 m.
   🔒 Le modèle est une source d'entrée confinée derrière HwIn ; il ne pilote
      aucune sortie réelle. Un override manuel valide est appliqué par
      FB_SimBench, pas par ce FB.
   📄 Docs : DOC/AF/AF_Partie-13_Fonction_Simulation_v2.4.md
             DOC/AF/AF_Partie-11_Fonction_Translation_v2.3.md
   ═══════════════════════════════════════════════════════════════ *)

FUNCTION_BLOCK PUBLIC FB_Sim_Translation
VAR_INPUT
    Enable          : BOOL;              (* --> [CMD] TRUE = modèle actif *)
    ReqTremie       : BOOL;              (* --> [CMD] Demande explicite vers Trémie *)
    ReqMaintenance  : BOOL;              (* --> [CMD] Demande explicite vers Maintenance *)
    SpeedTgt_Pct    : REAL;              (* --> [CMD] Magnitude commande 0..100 %, bornée avant calcul *)
    FullTravelTimeS : REAL := 8.0;       (* --> [CFG] Temps complet Trémie→Maintenance à 100 %, simulation seulement *)
END_VAR
VAR_OUTPUT
    PosTremie      : BOOL;               (* <-- [STAT] Bit4 du mot capteurs *)
    PosPV          : BOOL;               (* <-- [STAT] Bit3 du mot capteurs *)
    PosP2          : BOOL;               (* <-- [STAT] Bit2 du mot capteurs (point matériel PVP2) *)
    PosP1          : BOOL;               (* <-- [STAT] Bit1 du mot capteurs *)
    PosMaintenance : BOOL;               (* <-- [STAT] Bit0 du mot capteurs *)
END_VAR
VAR
    instCycleTime              : FB_CycleTime;   (* * [INST] Mesure temps de cycle *)
    PositionProgress           : REAL;   (* . [LOC] 0 m=Trémie ... 30 m=Maintenance *)
    SpeedPctBounded            : REAL;   (* . [LOC] Vitesse bornée *)
    PositionIncrement          : REAL;   (* . [LOC] Incrément de progression *)
    Initialized                : BOOL := FALSE;   (* . [LOC] Initialisation *)
END_VAR
VAR CONSTANT
    // 📏 Positions réelles des 5 capteurs (mètres) — convention 0 m=Trémie, 30 m=Maintenance.
    // Distances NON-linéaires : 0→5 (PV), 5→15 (P2), 15→20 (P1), 20→30 (Maintenance).
    CST_PositionAtTremie       : REAL := 0.0;
    CST_PositionAtPV           : REAL := 5.0;
    CST_PositionAtP2           : REAL := 15.0;
    CST_PositionAtP1           : REAL := 20.0;
    CST_PositionAtMaintenance  : REAL := 30.0;
    CST_TotalDistanceM         : REAL := 30.0; // Distance totale Trémie→Maintenance (m)
    CST_MinFullTravelTimeS     : REAL := 0.1; // Protection division : modèle sans dynamique sous ce seuil
END_VAR

// === IMPLEMENTATION ===

{region "§1 Gate"}
IF NOT Enable THEN
    // Prochain démarrage déterministe sur la Trémie (0 m), pour un état de départ propre.
    PositionProgress := CST_PositionAtTremie;
    Initialized := FALSE;
    PosTremie := FALSE;
    PosPV := FALSE;
    PosP2 := FALSE;
    PosP1 := FALSE;
    PosMaintenance := FALSE;
    RETURN;
END_IF;
{endregion}

{region "§2 Initialisation"}
IF NOT Initialized THEN
    PositionProgress := CST_PositionAtTremie;
    Initialized := TRUE;
END_IF;
{endregion}

{region "§3 Progression"}
instCycleTime(DefaultValueS := 0.010);
SpeedPctBounded := LIMIT(0.0, ABS(SpeedTgt_Pct), 100.0);

// Une vitesse à 100 % parcourt les 30 m (Trémie→Maintenance) pendant FullTravelTimeS.
// Les demandes de sens sont des bits explicites ; une demande contradictoire est neutre.
IF FullTravelTimeS >= CST_MinFullTravelTimeS THEN
    PositionIncrement := (SpeedPctBounded / 100.0)
                         * (CST_TotalDistanceM / FullTravelTimeS)
                         * instCycleTime.CycleTimeS;
ELSE
    PositionIncrement := 0.0;
END_IF;

IF ReqTremie AND NOT ReqMaintenance THEN
    PositionProgress := MAX(CST_PositionAtTremie, PositionProgress - PositionIncrement);
ELSIF ReqMaintenance AND NOT ReqTremie THEN
    PositionProgress := MIN(CST_PositionAtMaintenance, PositionProgress + PositionIncrement);
END_IF;
{endregion}

{region "§4 Publication capteurs"}
// Réinitialisation explicite : aucune valeur transitoire héritée d'une zone précédente.
PosTremie := FALSE;
PosPV := FALSE;
PosP2 := FALSE;
PosP1 := FALSE;
PosMaintenance := FALSE;

// Chaque intervalle correspond à un seul mot thermomètre valide.
IF PositionProgress <= CST_PositionAtTremie THEN
    PosTremie := TRUE;
    PosPV := TRUE;
    PosP2 := TRUE;
    PosP1 := TRUE;
    PosMaintenance := TRUE;
ELSIF PositionProgress <= CST_PositionAtPV THEN
    PosPV := TRUE;
    PosP2 := TRUE;
    PosP1 := TRUE;
    PosMaintenance := TRUE;
ELSIF PositionProgress <= CST_PositionAtP2 THEN
    PosP2 := TRUE;
    PosP1 := TRUE;
    PosMaintenance := TRUE;
ELSIF PositionProgress <= CST_PositionAtP1 THEN
    PosP1 := TRUE;
    PosMaintenance := TRUE;
ELSIF PositionProgress < CST_PositionAtMaintenance THEN
    PosMaintenance := TRUE;
END_IF;
{endregion}
