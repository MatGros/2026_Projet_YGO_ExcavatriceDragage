(* =======================================================================
   ⚙️ FB_CycleTime — Mesure du temps de cycle réel de la tâche
   ───────────────────────────────────────────────────────────────
   🎯 Rôle : Calcul du dt réel entre deux exécutions successives
   ⚙️ Calcul : Delta horloge système TIME() converti en secondes
   🛡️ Robustesse : dt borné (0 < dt <= CST_MaxCycleDeltaMs) sinon valeur de secours —
      neutralise les artefacts de rebouclage TIME() (49,7 j) et de reprise (online-change, debug)
   📄 Doc métier : DOC/AF/AF_Partie-03_Contrats_Composants/FB_CycleTime_v1.0.md
   ======================================================================= *)
FUNCTION_BLOCK FB_CycleTime
VAR_INPUT
    // === REGLAGES DE SECOURS ===
    DefaultValueS           : REAL := 0.004;      // --> [CFG] Valeur de secours par defaut (ex: 4ms = 0.004s)
END_VAR
VAR_OUTPUT
    // === MESURE ET DIAGNOSTIC ===
    CycleTimeS              : REAL;               // <-- [DIAG] Temps ecoule reel calcule (secondes)
END_VAR
VAR
    // === HORLOGE ET MEMOIRE TEMPORELLE ===
    TimeCurrent             : TIME;               // . [LOC] Temps systeme actuel
    TimeLast                : TIME;               // . [LOC] Empreinte de temps du cycle precedent
    DeltaTimeMs             : UDINT;              // . [LOC] Ecart brut calcule (ms)
    InitDone                : BOOL;               // . [LOC] Drapeau d'initialisation du premier cycle
END_VAR
VAR CONSTANT
    // === BORNE HAUTE DU DELTA DE CYCLE ===
    CST_MaxCycleDeltaMs     : UDINT := 1000;      // Plafond physique du dt entre deux scans (ms) : au-dela -> valeur de secours
END_VAR

{region "§1 Acquisition temps"}
// 1. ACQUISITION DU TEMPS SYSTÈME
TimeCurrent := TIME();
{endregion}

{region "§2 Initialisation"}
// 2. INITIALISATION UNIQUE AU PREMIER CYCLE
IF NOT InitDone THEN
    TimeLast := TimeCurrent;
    InitDone := TRUE;
    CycleTimeS := DefaultValueS;
    RETURN;
END_IF;
{endregion}

{region "§3 Calcul delta"}
// 3. CALCUL DU DELTA EN MILLISECONDES
DeltaTimeMs := TIME_TO_UDINT(TimeCurrent - TimeLast);
{endregion}

{region "§4 Conversion, secours et bornage"}
// 4. CONVERSION EN SECONDES, AVEC DOUBLE BORNAGE DU DELTA
//
// TIME() est un compteur systeme 32 bits en millisecondes depuis le demarrage de
// l'automate. Il n'est PAS absolu : il reboucle a 0 tous les 2^32 ms (~49,7 jours).
// La soustraction TimeCurrent - TimeLast est modulo 2^32 : au passage du rebouclage,
// tant que l'ecart reel entre deux scans reste < 49,7 j (toujours vrai a 10 ms),
// DeltaTimeMs reste correct. Le rebouclage seul n'est donc pas le risque principal.
//
// Le vrai risque est un DeltaTimeMs REELLEMENT > CST_MaxCycleDeltaMs : reprise apres
// un online-change, sortie d'un point d'arret / pas-a-pas debug, ou fenetre du
// rebouclage TIME(). Le chien de garde tache (200 ms, sensibilite 1) borne deja les
// gels en fonctionnement ; ce plafond couvre les cas hors surveillance.
// Un dt aberrant publie tel quel donnerait un CycleTimeS enorme, propage en un seul
// scan dans les rampes de vitesse et les integrateurs de position/frein -> saut
// physique. Fail-safe : hors de la plage nominale (0 < dt <= plafond), on publie la
// valeur de secours DefaultValueS au lieu de l'artefact. Pas de redemarrage auto.
//
// Borne basse dt = 0 (anti-zero, ex. premier appel du scan) et borne haute
// dt > CST_MaxCycleDeltaMs (anti-artefact) partagent la meme branche de secours.
IF (DeltaTimeMs > 0) AND (DeltaTimeMs <= CST_MaxCycleDeltaMs) THEN
    CycleTimeS := UDINT_TO_REAL(DeltaTimeMs) / 1000.0;
ELSE
    CycleTimeS := DefaultValueS;
END_IF;
{endregion}

{region "§5 Archivage"}
// 5. ARCHIVAGE POUR LE PROCHAIN CYCLE
TimeLast := TimeCurrent;
{endregion}

FUNCTION_BLOCK PUBLIC FB_Sim_TranslationDrive
VAR_INPUT
    Enable : BOOL;
    FrequencyTgt_Hz : REAL;
    AccelRate_HzPerS : REAL := 20.0;
    DecelRate_HzPerS : REAL := 20.0;
    TimeConstant_S : REAL := 0.15;
    LoadAngularVelocity_RadPerS : REAL;
    LoadFrequencyDroopGain_Hz : REAL := 0.50;
    CycleTime_S : REAL := 0.01;
END_VAR
VAR_OUTPUT
    Ready : BOOL;
    FrequencyCmd_Hz : REAL;
    ElectricalFrequency_Hz : REAL;
    ActualFrequency_Hz : REAL;
    ResponseAlpha : REAL;
END_VAR
VAR
    FrequencyDelta_Hz : REAL;
    FrequencyStep_Hz : REAL;
END_VAR
VAR CONSTANT
    CST_MinTime_S : REAL := 0.001;
    CST_MaxFrequency_Hz : REAL := 40.0;
END_VAR

IF NOT Enable THEN
    Ready := FALSE;
    FrequencyCmd_Hz := 0.0;
    ElectricalFrequency_Hz := 0.0;
    ActualFrequency_Hz := 0.0;
    ResponseAlpha := 0.0;
    RETURN;
END_IF;

FrequencyDelta_Hz := LIMIT(0.0, FrequencyTgt_Hz, CST_MaxFrequency_Hz) - FrequencyCmd_Hz;
IF FrequencyDelta_Hz >= 0.0 THEN
    FrequencyStep_Hz := MAX(0.0, AccelRate_HzPerS) * CycleTime_S;
ELSE
    FrequencyStep_Hz := MAX(0.0, DecelRate_HzPerS) * CycleTime_S;
END_IF;
FrequencyCmd_Hz := FrequencyCmd_Hz + LIMIT(-FrequencyStep_Hz, FrequencyDelta_Hz, FrequencyStep_Hz);
IF TimeConstant_S > CST_MinTime_S THEN
    ResponseAlpha := CycleTime_S / (TimeConstant_S + CycleTime_S);
ELSE
    ResponseAlpha := 1.0;
END_IF;
ElectricalFrequency_Hz := ElectricalFrequency_Hz
    + ResponseAlpha * (FrequencyCmd_Hz - ElectricalFrequency_Hz);
ActualFrequency_Hz := MAX(0.0, ElectricalFrequency_Hz
    - MAX(0.0, LoadFrequencyDroopGain_Hz) * ABS(LoadAngularVelocity_RadPerS));
IF ActualFrequency_Hz < 0.005 THEN
    ActualFrequency_Hz := 0.0;
END_IF;
Ready := TRUE;

    ElectricalOutput : BOOL;
    EpisodeActive : BOOL;
    TransitionCount : UDINT;
    EpisodeTogglePeriodS : REAL;

    Initialized : BOOL;
    PreviousRawDetected : BOOL;
    EpisodeTarget : BOOL;
    DelayTimerS : REAL;
    EpisodeTimerS : REAL;
    ToggleTimerS : REAL;
    RandomState : UDINT;

    Initialized := FALSE;
    RawDetected := FALSE;
    ElectricalOutput := FALSE;
    PreviousRawDetected := FALSE;
    EpisodeActive := FALSE;
    DelayTimerS := 0.0;
    EpisodeTimerS := 0.0;
    ToggleTimerS := 0.0;
    TransitionCount := 0;
    EpisodeTogglePeriodS := 0.0;
    RETURN;

(* Hysteresis sur une frontiere cumulative : les cinq DI restent compatibles avec
   le mot thermometre du decodeur M3, contrairement a cinq fenetres independantes. *)
IF NOT Initialized THEN
    IF ActiveBelowThreshold THEN
        RawDetected := Position_M <= Threshold_M;
    ELSE
        RawDetected := Position_M >= Threshold_M;
    END_IF;
    ElectricalOutput := RawDetected;
    PreviousRawDetected := RawDetected;
    RandomState := ScenarioSeed MOD 1000000;
    IF RandomState = 0 THEN
        RandomState := 1;
    END_IF;
    EpisodeTogglePeriodS := MAX(0.01, IntermittencePeriodS);
    Initialized := TRUE;
    RETURN;
END_IF;

IF ActiveBelowThreshold THEN
    IF RawDetected THEN
        RawDetected := Position_M <= (Threshold_M + MAX(0.0, Hysteresis_M));
    ELSE
        RawDetected := Position_M <= (Threshold_M - MAX(0.0, Hysteresis_M));
    END_IF;
ELSE
    IF RawDetected THEN
        RawDetected := Position_M >= (Threshold_M - MAX(0.0, Hysteresis_M));
    ELSE
        RawDetected := Position_M >= (Threshold_M + MAX(0.0, Hysteresis_M));
    END_IF;
END_IF;

   propres bascules electriques. Les fronts restent donc bornes et reproductibles. *)

    PreviousRawDetected := RawDetected;
    EpisodeTarget := RawDetected;
    (* Une bascule mecanique pendant l'episode modifie seulement la cible finale.
       Elle ne remet jamais son chronometre a zero : l'episode reste borne. *)
    IF NOT EpisodeActive THEN
        DelayTimerS := 0.0;
        EpisodeTimerS := 0.0;
        ToggleTimerS := 0.0;
        (* Tirage unique par episode : reproductible pour une graine et sans bruit
           blanc scan-par-scan. La periode varie de 70 % a 130 % du nominal. *)
        RandomState := RandomState + 7919;
        IF RandomState >= 1000000 THEN
            RandomState := RandomState - 1000000;
        END_IF;
        EpisodeTogglePeriodS := MAX(0.01, IntermittencePeriodS
            * (0.7 + UDINT_TO_REAL(RandomState MOD 6000) / 10000.0));
        EpisodeActive := IntermittenceActive AND (IntermittenceDurationS > 0.0);
    END_IF;
END_IF;

IF EpisodeActive THEN
    EpisodeTimerS := EpisodeTimerS + CycleTimeS;
    ToggleTimerS := ToggleTimerS + CycleTimeS;
    IF ToggleTimerS >= EpisodeTogglePeriodS THEN
        ElectricalOutput := NOT ElectricalOutput;
        ToggleTimerS := 0.0;
        TransitionCount := TransitionCount + 1;
    END_IF;
    IF EpisodeTimerS >= IntermittenceDurationS THEN
        EpisodeActive := FALSE;
        IF ElectricalOutput <> EpisodeTarget THEN
            ElectricalOutput := EpisodeTarget;
            TransitionCount := TransitionCount + 1;
        END_IF;
        DelayTimerS := 0.0;
    END_IF;
ELSIF ElectricalOutput <> RawDetected THEN
    DelayTimerS := DelayTimerS + CycleTimeS;
    IF DelayTimerS >= MAX(0.0, ElectricalDelayS) THEN
        ElectricalOutput := RawDetected;
        DelayTimerS := 0.0;
        TransitionCount := TransitionCount + 1;
    END_IF;
ELSE
    DelayTimerS := 0.0;
END_IF;
